prontera,159,157,4	script	Tutorial Quest 3::QuestTuto3	4_F_VALKYRIE,{
	mes "[Tutorial Quest 3]";

	if (Tuto3_State == 0) {
		mes "Do you know the game?";
		next;

		if (select("Yes:No") == 1) {
			mes "[Tutorial Quest 3]";
			mes "Good. Go to 163 157 and read the board.";
			Tuto3_State = 1;
			close;
		}

		mes "[Tutorial Quest 3]";
		mes "Then go read the board anyway. Education, sadly, remains legal.";
		Tuto3_State = 1;
		close;
	}

	if (Tuto3_State == 1) {
		mes "Did you read what I asked you to read?";
		next;

		switch(select("^0000FFBlue^000000:^FF00002^000000:^FF0000Red^000000:^00AA00Bard^000000")) {

		case 1:
		case 3:
		case 4:
			mes "[Tutorial Quest 3]";
			mes "Wrong answer.";
			close;

		case 2:
			mes "[Tutorial Quest 3]";
			mes "Well done!";
			Zeny += 10000;
			Tuto3_State = 2;
			next;
			mes "[Tutorial Quest 3]";
			mes "Now go see Tutorial Quest 3-1.";
			close;
		}
	}

	if (Tuto3_State == 2) {
		mes "[Tutorial Quest 3]";
		mes "Go talk to the NPC at 164 171.";
		close;
	}

	if (Tuto3_State == 3) {
		mes "[Tutorial Quest 3]";
		mes "Did your trip go well?";
		next;
		select("Yes:No");

		mes "[Tutorial Quest 3]";
		mes "Do you want your reward?";
		next;

		if (select("^FF0000Yes^000000:^00AA00No^000000") == 1) {
			Zeny += 10000;
			getexp 10000,10000;
			Tuto3_State = 4;
			mes "[Tutorial Quest 3]";
			mes "See you next time, companion!";
			close;
		}

		Tuto3_State = 4;
		mes "[Tutorial Quest 3]";
		mes "All right, see you next time then...";
		close;
	}

	mes "[Tutorial Quest 3]";
	mes "You already completed this quest.";
	close;
}
