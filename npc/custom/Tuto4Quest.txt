//===== Hercules Script ======================================
// Main Quest Tuto 4
// Valkyrie gives a Boarding Halter, auto-mounts the player,
// sends them on an autopilot route through Prontera, then rewards
// them when they return and speak to her again.
//============================================================

prontera,159,160,4	script	Taxi 2 - Swordman Class	4_F_VALKYRIE,{
	mes "[Valkyrie]";

	if (Tuto4_State == 1 && !#Tuto4_RewardClaimed) {
		mes "Did the trip go well?";
		next;

		if (select("Yes:No") == 2) {
			mes "[Valkyrie]";
			mes "All right. Come back when you want to receive your reward.";
			close;
		}

		Zeny += 100000;
		getexp 100000,100000;
		getitem 607,1;
		#Tuto4_RewardClaimed = 1;
		Tuto4_State = 0;

		mes "[Valkyrie]";
		mes "Excellent. Here is your reward.";
		mes "100,000 Zeny, 100,000 Base EXP, 100,000 Job EXP and a Yggdrasil Berry.";
		close;
	}

	if (Tuto4_State == 1 && #Tuto4_RewardClaimed)
		Tuto4_State = 0;

	if (Tuto4_State != 1) {
		mes "Do you want me to take you to the Swordman class master?";
		next;

		if (select("Yes:No") == 2) {
			mes "[Valkyrie]";
			mes "Very well. Come back when you want to meet him.";
			close;
		}

		if ($Tuto4_RiderBusy) {
			mes "[Valkyrie]";
			mes "The mount is already out.";
			mes "Wait a few seconds, then try again.";
			close;
		}

		if (checkmount() || hascashmount()) {
			mes "[Valkyrie]";
			mes "Dismount from your current mount first.";
			close;
		}

		if (!checkweight(12622,1)) {
			mes "[Valkyrie]";
			mes "Your inventory is too heavy to receive the Boarding Halter.";
			close;
		}

		$Tuto4_RiderBusy = getcharid(CHAR_ID_ACCOUNT);
		getitem 12622,1;
		consumeitem 12622;
		sleep2 300;

		if (!hascashmount()) {
			if (countitem(12622) > 0)
				delitem 12622,1;
			$Tuto4_RiderBusy = 0;
			mes "[Valkyrie]";
			mes "The mount refuses to leave for now.";
			mes "Try again in a few moments.";
			close;
		}

		mes "[Valkyrie]";
		mes "Perfect. I will take you to the Swordman master.";
		close2;

		sleep2 400;

		callsub S_MoveTo, 161, 175, 2800;
		callsub S_MoveTo, 165, 190, 2800;
		callsub S_MoveTo, 167, 204, 2600;

		callsub S_MoveTo, 164, 218, 3000;
		callsub S_MoveTo, 161, 232, 3000;
		callsub S_MoveTo, 158, 246, 3000;
		callsub S_MoveTo, 155, 260, 3000;
		callsub S_MoveTo, 152, 274, 3000;
		callsub S_MoveTo, 150, 287, 2800;

		callsub S_MoveTo, 150, 294, 500;
		callsub S_MoveTo, 149, 302, 900;
		callsub S_MoveTo, 149, 308, 1400;
		callsub S_MoveTo, 148, 314, 1600;
		callsub S_MoveTo, 148, 318, 1200;

		callsub S_MoveTo, 142, 318, 2200;
		callsub S_MoveTo, 136, 318, 2200;
		callsub S_MoveTo, 130, 318, 2200;
		callsub S_MoveTo, 123, 318, 2200;

		callsub S_MoveTo, 120, 309, 2400;
		callsub S_MoveTo, 117, 300, 2400;
		callsub S_MoveTo, 114, 292, 2400;
		callsub S_MoveTo, 107, 292, 2200;
		callsub S_MoveTo, 100, 292, 2200;

		callsub S_MoveTo, 90, 302, 2800;
		callsub S_MoveTo, 80, 312, 2800;
		callsub S_MoveTo, 70, 322, 2800;
		callsub S_MoveTo, 60, 333, 2800;
		callsub S_MoveTo, 47, 344, 3600;
		sleep2 1000;

		sc_end(SC_ALL_RIDING);
		if (countitem(12622) > 0)
			delitem 12622,1;

		if (#Tuto4_RewardClaimed)
			Tuto4_State = 0;
		else
			Tuto4_State = 1;

		$Tuto4_RiderBusy = 0;
		if (#Tuto4_RewardClaimed)
			message strcharinfo(PC_NAME), "Valkyrie Tuto 4: you arrived in front of the Swordman building. You can repeat the trip whenever you want.";
		else
			message strcharinfo(PC_NAME), "Valkyrie Tuto 4: you arrived in front of the Swordman building. Return to the Valkyrie.";
		end;
	}

	mes "Come back to me after your trip.";
	close;

S_MoveTo:
	.@pc_gid = getcharid(CHAR_ID_ACCOUNT);
	.@timeout = getarg(2);
	.@elapsed = 0;

	unitwalk(.@pc_gid, getarg(0), getarg(1));
	sleep2 75;
	.@elapsed += 75;

	while (.@elapsed < .@timeout) {
		getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
		.@dx = getarg(0) - .@x;
		.@dy = getarg(1) - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@dx <= 6 && .@dy <= 6)
			return;

		if (unitiswalking(.@pc_gid) != 1 && .@elapsed > 250)
			unitwalk(.@pc_gid, getarg(0), getarg(1));

		sleep2 75;
		.@elapsed += 75;
	}

	return;

OnPCLogoutEvent:
	if ($Tuto4_RiderBusy == getcharid(CHAR_ID_ACCOUNT)) {
		sc_end(SC_ALL_RIDING);
		if (countitem(12622) > 0)
			delitem 12622,1;
		setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND|PCBLOCK_NPC, false);
		$Tuto4_RiderBusy = 0;
	}
	end;
}
