//===== Hercules Script ======================================
// Main Quest Tuto 4
// Valkyrie gives a Boarding Halter, auto-mounts the player,
// sends them on an autopilot route through Prontera, then rewards
// them when they return and speak to her again.
//============================================================

prontera,159,160,4	script	Valkyrie Tuto 4	4_F_VALKYRIE,{
	mes "[Valkyrie]";

	if (Tuto4_State == 1 && !#Tuto4_RewardClaimed) {
		mes "Le voyage a ete bon ?";
		next;

		if (select("Oui:Non") == 2) {
			mes "[Valkyrie]";
			mes "D'accord. Reviens me voir quand tu voudras recevoir ta recompense.";
			close;
		}

		Zeny += 100000;
		getexp 100000,100000;
		getitem 607,1;
		#Tuto4_RewardClaimed = 1;
		Tuto4_State = 0;

		mes "[Valkyrie]";
		mes "Excellent. Voici ta recompense.";
		mes "100 000 Zeny, 100 000 Base EXP, 100 000 Job EXP et une Yggdrasil Berry.";
		close;
	}

	if (Tuto4_State == 1 && #Tuto4_RewardClaimed)
		Tuto4_State = 0;

	if (Tuto4_State != 1) {
		mes "Veux-tu que je t'emmene au maitre de classe Swordman ?";
		next;

		if (select("Oui:Non") == 2) {
			mes "[Valkyrie]";
			mes "Tres bien. Reviens me voir quand tu voudras le rencontrer.";
			close;
		}

		if ($Tuto4_RiderBusy) {
			mes "[Valkyrie]";
			mes "La monture est deja en balade.";
			mes "Attends quelques secondes puis reessaie.";
			close;
		}

		if (checkmount() || hascashmount()) {
			mes "[Valkyrie]";
			mes "Descends d'abord de ta monture actuelle.";
			close;
		}

		if (!checkweight(12622,1)) {
			mes "[Valkyrie]";
			mes "Ton inventaire est trop lourd pour recevoir le Boarding Halter.";
			close;
		}

		$Tuto4_RiderBusy = getcharid(CHAR_ID_ACCOUNT);
		getitem 12622,1;
		consumeitem 12622;
		sleep2 300;

		if (!hascashmount()) {
			if (countitem(12622) > 0)
				delitem 12622,1;
			$Tuto4_RiderBusy = 0;
			mes "[Valkyrie]";
			mes "La monture refuse de partir pour le moment.";
			mes "Reessaie dans quelques instants.";
			close;
		}

		mes "[Valkyrie]";
		mes "Parfait. Je t'emmene au maitre Swordman.";
		close2;

		sleep2 400;

		callsub S_MoveTo, 161, 175, 2800;
		callsub S_MoveTo, 165, 190, 2800;
		callsub S_MoveTo, 167, 204, 2600;

		callsub S_MoveTo, 164, 218, 3000;
		callsub S_MoveTo, 161, 232, 3000;
		callsub S_MoveTo, 158, 246, 3000;
		callsub S_MoveTo, 155, 260, 3000;
		callsub S_MoveTo, 152, 274, 3000;
		callsub S_MoveTo, 150, 287, 2800;

		callsub S_MoveTo, 149, 296, 1800;
		callsub S_MoveTo, 149, 300, 1200;
		callsub S_MoveTo, 149, 308, 1700;
		callsub S_MoveTo, 148, 314, 1600;
		callsub S_MoveTo, 148, 318, 1200;

		callsub S_MoveTo, 142, 318, 2200;
		callsub S_MoveTo, 136, 318, 2200;
		callsub S_MoveTo, 130, 318, 2200;
		callsub S_MoveTo, 123, 318, 2200;

		callsub S_MoveTo, 120, 309, 2400;
		callsub S_MoveTo, 117, 300, 2400;
		callsub S_MoveTo, 114, 292, 2400;
		callsub S_MoveTo, 107, 292, 2200;
		callsub S_MoveTo, 100, 292, 2200;

		callsub S_MoveTo, 90, 302, 2800;
		callsub S_MoveTo, 80, 312, 2800;
		callsub S_MoveTo, 70, 322, 2800;
		callsub S_MoveTo, 60, 333, 2800;
		callsub S_MoveTo, 47, 344, 3600;

		sc_end(SC_ALL_RIDING);
		if (countitem(12622) > 0)
			delitem 12622,1;

		if (#Tuto4_RewardClaimed)
			Tuto4_State = 0;
		else
			Tuto4_State = 1;

		$Tuto4_RiderBusy = 0;
		if (#Tuto4_RewardClaimed)
			message strcharinfo(PC_NAME), "Valkyrie Tuto 4: tu es arrive devant le batiment Swordman. Tu peux refaire le trajet quand tu veux.";
		else
			message strcharinfo(PC_NAME), "Valkyrie Tuto 4: tu es arrive devant le batiment Swordman. Retourne parler a la Valkyrie.";
		end;
	}

	mes "Retourne me voir apres ton voyage.";
	close;

S_MoveTo:
	.@pc_gid = getcharid(CHAR_ID_ACCOUNT);
	.@timeout = getarg(2);
	.@elapsed = 0;

	unitwalk(.@pc_gid, getarg(0), getarg(1));
	sleep2 75;
	.@elapsed += 75;

	while (.@elapsed < .@timeout) {
		getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
		.@dx = getarg(0) - .@x;
		.@dy = getarg(1) - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@dx <= 5 && .@dy <= 5)
			return;

		if (unitiswalking(.@pc_gid) != 1 && .@elapsed > 250)
			unitwalk(.@pc_gid, getarg(0), getarg(1));

		sleep2 75;
		.@elapsed += 75;
	}

	return;

OnPCLogoutEvent:
	if ($Tuto4_RiderBusy == getcharid(CHAR_ID_ACCOUNT)) {
		sc_end(SC_ALL_RIDING);
		if (countitem(12622) > 0)
			delitem 12622,1;
		setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND|PCBLOCK_NPC, false);
		$Tuto4_RiderBusy = 0;
	}
	end;
}
