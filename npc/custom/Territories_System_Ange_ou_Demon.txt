//===== Hercules Script ======================================
//= Territories System Ange ou Demon
//= Affiche les territoires controles par les Anges et Demons.
//============================================================

prontera,152,154,4	script	Territoires Ange/Demon	10387,{
	while (true) {
		mes "[Gardienne des Territoires]";
		mes "Je veille sur les terres marquees par les Anges et les Demons.";
		next;

		.@menu = select("Voir les territoires:Comprendre le systeme:Quitter");

		if (.@menu == 1) {
			callsub S_SelectFaction;
			next;
		} else if (.@menu == 2) {
			callsub S_ShowInfo;
			next;
		} else {
			close;
		}
	}

S_SelectFaction:
	mes "[Gardienne des Territoires]";
	mes "Quel camp veux-tu consulter ?";
	next;

	.@faction = select("Anges:Demons:Retour");
	if (.@faction == 3)
		return;

	callsub S_SelectCategory, .@faction;
	return;

S_SelectCategory:
	.@faction = getarg(0);
	.@faction_name$ = (.@faction == 1) ? "Anges" : "Demons";

	mes "[Gardienne des Territoires]";
	mes "Territoires controles par les " + .@faction_name$ + ".";
	mes "Choisis une categorie.";
	next;

	.@category = select("Villes:Fields:Dungeons:Retour");
	if (.@category == 4)
		return;

	callsub S_ShowTerritories, .@faction, .@category;
	return;

S_ShowTerritories:
	.@faction = getarg(0);
	.@category = getarg(1);
	.@faction_name$ = (.@faction == 1) ? "Anges" : "Demons";
	.@category_name$ = "Villes";
	if (.@category == 2)
		.@category_name$ = "Fields";
	else if (.@category == 3)
		.@category_name$ = "Dungeons";

	mes "[Gardienne des Territoires]";
	mes .@category_name$ + " des " + .@faction_name$ + " :";

	.@count = 0;

	if (.@category == 2) {
		if ($Territory_Prt08_Alignment == .@faction) {
			mes "- prt_fild08";
			.@count++;
		}
		if ($Territory_Prt06_Alignment == .@faction) {
			mes "- prt_fild06";
			.@count++;
		}
		if ($Territory_Prt05_Alignment == .@faction) {
			mes "- prt_fild05";
			.@count++;
		}
	}

	if (.@count == 0)
		mes "Aucun territoire enregistre pour le moment.";

	next;
	if (select("Voir une autre categorie:Retour au menu principal") == 1)
		callsub S_SelectCategory, .@faction;
	return;

S_ShowInfo:
	mes "[Gardienne des Territoires]";
	mes "Chaque territoire possede un Cristal de Puissance.";
	mes "Quand un camp defend ce cristal, son influence grandit sur la zone.";
	next;

	mes "[Gardienne des Territoires]";
	mes "Soutiens les Anges ou les Demons en protegeant le cristal de la carte.";
	mes "Si ton camp tient la defense, le territoire peut passer sous son controle.";
	next;

	mes "[Gardienne des Territoires]";
	mes "Territoires actuels :";
	mes "- Les listes sont mises a jour quand un Cristal de Puissance est conquis.";
	mes "- Un cristal neutre peut etre revendique par les Anges ou les Demons.";
	return;
}

function	script	F_PowerCrystalTalk	{
	.@map$ = getarg(0);
	.@npc$ = getarg(1);
	.@alignment_var$ = getarg(2);
	.@battle_var$ = getarg(3);
	.@kills_var$ = getarg(4);
	.@pending_var$ = getarg(5);
	.@need_var$ = getarg(6);
	.@x = getarg(7);
	.@y = getarg(8);
	.@event$ = getarg(9);
	.@alignment = getd(.@alignment_var$);

	mes "[Cristal de Puissance]";
	if (.@alignment == 1) {
		mes "Ce territoire est actuellement affilie aux Anges.";
		return;
	}
	if (.@alignment == 2) {
		mes "Ce territoire est actuellement affilie aux Demons.";
		return;
	}

	if (getd(.@battle_var$)) {
		mes "La conquete du territoire est deja en cours.";
		mes "Elimine les gardiens invoques autour du cristal.";
		mes "Progression : " + getd(.@kills_var$) + "/" + getd(.@need_var$) + ".";
		return;
	}

	mes "Ce territoire est neutre.";
	if (Faction != 1 && Faction != 2) {
		mes "Je ne ressens aucune faction en toi.";
		mes "Va d'abord choisir ton camp aupres du NPC Ange ou Demon.";
		return;
	}

	mes "Souhaites-tu attaquer le Cristal de Puissance ?";
	next;

	if (select("Oui:Non") == 2) {
		mes "[Cristal de Puissance]";
		mes "Reviens lorsque tu seras pret.";
		return;
	}

	.@alignment = Faction;
	setd .@battle_var$, 1;
	setd .@kills_var$, 0;
	setd .@pending_var$, .@alignment;
	setd .@need_var$, 5;

	mes "[Cristal de Puissance]";
	if (.@alignment == 1) {
		mes "Tu soutiens les Anges.";
		mes "D'accord, tres bien.";
		mes "Le cristal invoque des Poring. Terrasse-les pour revendiquer ce territoire.";
		callfunc "F_PowerCrystalSpawnFive", .@map$, .@x, .@y, PORING, "Poring du Cristal", .@event$;
	} else {
		mes "Tu soutiens les Demons.";
		mes "D'accord, tres bien.";
		mes "Le cristal invoque des Lunatic. Terrasse-les pour revendiquer ce territoire.";
		callfunc "F_PowerCrystalSpawnFive", .@map$, .@x, .@y, 1063, "Lunatic du Cristal", .@event$;
	}
	specialeffect(EF_BEGINSPELL6, AREA, .@npc$);
	return;
}

function	script	F_PowerCrystalSpawnFive	{
	.@map$ = getarg(0);
	.@x = getarg(1);
	.@y = getarg(2);
	.@mob_id = getarg(3);
	.@mob_name$ = getarg(4);
	.@event$ = getarg(5);

	monster .@map$, .@x - 2, .@y, .@mob_name$, .@mob_id, 1, .@event$;
	monster .@map$, .@x + 2, .@y, .@mob_name$, .@mob_id, 1, .@event$;
	monster .@map$, .@x, .@y - 2, .@mob_name$, .@mob_id, 1, .@event$;
	monster .@map$, .@x, .@y + 2, .@mob_name$, .@mob_id, 1, .@event$;
	monster .@map$, .@x + 1, .@y + 1, .@mob_name$, .@mob_id, 1, .@event$;
	return;
}

function	script	F_PowerCrystalMonsterDead	{
	.@map$ = getarg(0);
	.@npc$ = getarg(1);
	.@alignment_var$ = getarg(2);
	.@battle_var$ = getarg(3);
	.@kills_var$ = getarg(4);
	.@pending_var$ = getarg(5);
	.@need_var$ = getarg(6);

	if (!getd(.@battle_var$))
		return;

	.@kills = getd(.@kills_var$) + 1;
	setd .@kills_var$, .@kills;

	if (.@kills < getd(.@need_var$))
		return;

	.@alignment = getd(.@pending_var$);
	setd .@alignment_var$, .@alignment;
	setd .@battle_var$, 0;
	setd .@kills_var$, 0;
	setd .@pending_var$, 0;
	setd .@need_var$, 0;

	specialeffect(EF_LEVEL99, AREA, .@npc$);
	specialeffect(EF_ANGEL2, AREA, .@npc$);
	if (.@alignment == 1)
		mapannounce .@map$, "Le Cristal de Puissance est maintenant affilie aux Anges.", bc_map, 0x66CCFF;
	else
		mapannounce .@map$, "Le Cristal de Puissance est maintenant affilie aux Demons.", bc_map, 0xCC66FF;
	return;
}
