prontera,152,169,4	script	ClassContracts	4_F_VALKYRIE,{
	mes("[Class Contracts]");

	if (BaseJob == Job_Swordman) {
		.@mob = 1002; .@mob$ = "Poring"; .@reward = 1101; .@reward$ = "Sword";
	} else if (BaseJob == Job_Archer) {
		.@mob = 1063; .@mob$ = "Lunatic"; .@reward = 1701; .@reward$ = "Bow";
	} else if (BaseJob == Job_Acolyte) {
		.@mob = 1076; .@mob$ = "Skeleton"; .@reward = 1501; .@reward$ = "Mace";
	} else if (BaseJob == Job_Thief) {
		.@mob = 1113; .@mob$ = "Drops"; .@reward = 1201; .@reward$ = "Knife";
	} else if (BaseJob == Job_Merchant) {
		.@mob = 1031; .@mob$ = "Poporing"; .@reward = 1301; .@reward$ = "Axe";
	} else if (BaseJob == Job_Mage) {
		.@mob = 1008; .@mob$ = "Pupa"; .@reward = 1601; .@reward$ = "Rod";
	} else {
		mes("Your class has no contract available.");
		close();
	}

	mes("Class points: " + #ClassContract_Points + "/2");

	if (#ClassContract_Active == 0) {
		mes("Your class contract is:");
		mes("Hunt ^FF00001 " + .@mob$ + "^000000.");
		next();

		if (select("Accept contract:Leave") == 1) {
			#ClassContract_Active = 1;
			#ClassContract_Target = .@mob;
			#ClassContract_Kill = 0;

			mes("[Class Contracts]");
			mes("Contract accepted.");
			mes("Target: ^FF00001 " + .@mob$ + "^000000.");
			close();
		}
		close();
	}

	if (#ClassContract_Kill < 1) {
		mes("You already have an active contract.");
		mes("Target: ^FF00001 " + .@mob$ + "^000000.");
		mes("Progress: " + #ClassContract_Kill + "/1");
		close();
	}

	mes("Your contract is complete.");
	mes("Validate it now?");
	next();

	if (select("Validate contract:Leave") == 1) {
		#ClassContract_Active = 0;
		#ClassContract_Target = 0;
		#ClassContract_Kill = 0;
		#ClassContract_Points++;

		getitem .@reward,1;

		mes("[Class Contracts]");
		mes("Contract validated.");
		mes("You received: ^00FF001 " + .@reward$ + "^000000.");
		mes("Class points: " + #ClassContract_Points + "/2");

		if (#ClassContract_Points == 2) {
			next();
			mes("[Class Contracts]");
			mes("You have unlocked your class equipment vendor.");
			mes("Class equipment is now available for ^00FF001 zeny^000000.");
		}
		close();
	}
	close();

OnNPCKillEvent:
	if (#ClassContract_Active == 0)
		end;

	if (#ClassContract_Kill >= 1)
		end;

	if (killedrid == #ClassContract_Target) {
		#ClassContract_Kill = 1;
		dispbottom "Class contract completed. Return to the Class Contracts NPC.";
	}
	end;
}

// =====================================================
// Buy Function
// =====================================================

function	script	F_ClassGearBuy	{
	.@item = getarg(0);
	.@name$ = getarg(1);

	mes("[Class Gear Vendor]");
	mes("Item: ^00FF00" + .@name$ + "^000000");
	mes("Price: ^00FF001 zeny^000000");
	next();

	if (select("Buy:Cancel") == 1) {
		if (Zeny < 1) {
			mes("[Class Gear Vendor]");
			mes("You do not have enough zeny.");
			close();
		}

		Zeny -= 1;
		getitem .@item,1;

		mes("[Class Gear Vendor]");
		mes("Purchase complete.");
		mes("You received ^00FF001 " + .@name$ + "^000000.");
		close();
	}
	close();
}
