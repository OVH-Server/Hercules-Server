prontera,163,152,4	script	Harvest Job	10451,{
	callfunc "F_Harvest_MigrateProgress";
	mes "[Harvest Master]";
	mes "Bienvenue dans les metiers de recolte.";
	next;
	switch(select("Choisir/changer mon metier:Voir mes points:Voir mon compteur:Explication des metiers:Fermer")) {
	case 1:
		mes "[Harvest Master]";
		if (HarvestJob) {
			if (HarvestJob == 1)
				mes "Metier actuel : Herboriste.";
			else if (HarvestJob == 2)
				mes "Metier actuel : Mineur.";
			else
				mes "Metier actuel : Archeologue.";
			mes "Changer de metier coute ^0000FF1 Log^000000.";
		}
		else {
			mes "Tu n'as pas encore de metier.";
			mes "Le premier choix est gratuit.";
		}
		mes "Ce choix determine les ressources que tu peux recolter.";
		next;
		.@job = select("Herboriste:Mineur:Archeologue:Annuler");
		if (.@job == 4)
			close;
		if (HarvestJob == .@job) {
			mes "[Harvest Master]";
			mes "C'est deja ton metier actuel.";
			close;
		}

		mes "[Harvest Master]";
		if (.@job == 1)
			mes "Tu veux devenir Herboriste ?";
		else if (.@job == 2)
			mes "Tu veux devenir Mineur ?";
		else
			mes "Tu veux devenir Archeologue ?";
		next;
		if (select("Confirmer:Annuler") == 2)
			close;

		if (HarvestJob) {
			if (countitem(7201) < 1) {
				mes "[Harvest Master]";
				mes "Il te faut ^0000FF1 Log^000000 pour changer de metier.";
				close;
			}
			delitem 7201,1;
		}

		HarvestJob = .@job;
		if (HarvestJob == 1 && !HarvestHerbLevel)
			HarvestHerbLevel = 1;
		else if (HarvestJob == 2 && !HarvestMiningLevel)
			HarvestMiningLevel = 1;
		else if (HarvestJob == 3 && !HarvestArchaeoLevel)
			HarvestArchaeoLevel = 1;
		mes "[Harvest Master]";
		mes "C'est note.";
		if (HarvestJob == 1)
			mes "Tu peux maintenant recolter les plants.";
		else if (HarvestJob == 2)
			mes "Tu peux maintenant recolter les minerais.";
		else
			mes "Tu peux maintenant recuperer les reliques archeologiques.";
		close;

	case 2:
		mes "[Harvest Master]";
		if (!HarvestJob) {
			mes "Tu n'as pas encore choisi de metier.";
			mes "Va dans ^0000FFChoisir/changer mon metier^000000 pour commencer.";
			close;
		}
		.@herb_level = HarvestHerbLevel;
		.@mining_level = HarvestMiningLevel;
		.@archaeo_level = HarvestArchaeoLevel;
		if (!.@herb_level)
			.@herb_level = 1;
		if (!.@mining_level)
			.@mining_level = 1;
		if (!.@archaeo_level)
			.@archaeo_level = 1;
		if (HarvestJob == 1)
			mes "Metier : Herboriste.";
		else if (HarvestJob == 2)
			mes "Metier : Mineur.";
		else
			mes "Metier : Archeologue.";
		next;
		mes "[Harvest Master]";
		mes "Herboriste : niveau " + .@herb_level + ", " + HarvestHerbPoints + "/100 points.";
		mes "Mineur : niveau " + .@mining_level + ", " + HarvestMiningPoints + "/100 points.";
		mes "Archeologue : niveau " + .@archaeo_level + ", " + HarvestArchaeoPoints + "/100 points.";
		close;

	case 3:
		mes "[Harvest Master]";
		mes "Compteur total de recoltes :";
		mes "^0000FF" + HarvestGatherCount + "^000000 ressources recuperees.";
		next;
		mes "[Harvest Master]";
		mes "Plants : " + HarvestHerbGatherCount + ".";
		mes "Mining : " + HarvestMiningGatherCount + ".";
		mes "Archeology : " + HarvestArchaeoGatherCount + ".";
		mes "Box : " + HarvestBoxGatherCount + ".";
		close;

	case 4:
		mes "[Harvest Master]";
		mes "Il existe trois metiers de recolte :";
		mes "^228B22Herboriste^000000 : recolte uniquement les plants.";
		mes "^8B4513Mineur^000000 : recolte uniquement les minerais.";
		mes "^4169E1Archeologue^000000 : recupere uniquement les traces archeologiques.";
		next;
		mes "[Harvest Master]";
		mes "Chaque recolte valide donne 1 point.";
		mes "A 10 points tu passes niveau 2.";
		mes "A 20 points tu passes niveau 3.";
		mes "Au niveau 3, tu peux continuer jusqu'a 100 points.";
		mes "Tu peux changer de metier contre ^0000FF1 Log^000000.";
		mes "Les box sont a part : tout le monde peut les recolter, meme sans metier.";
		close;

	case 5:
		close;
	}
	close;
}

function	script	F_Harvest_MigrateProgress	{
	if (!HarvestJob || !HarvestLevel || HarvestProgressMigrated)
		return;

	if (HarvestJob == 1 && !HarvestHerbLevel) {
		HarvestHerbLevel = HarvestLevel;
		HarvestHerbPoints = HarvestPoints;
	}
	else if (HarvestJob == 2 && !HarvestMiningLevel) {
		HarvestMiningLevel = HarvestLevel;
		HarvestMiningPoints = HarvestPoints;
	}
	else if (HarvestJob == 3 && !HarvestArchaeoLevel) {
		HarvestArchaeoLevel = HarvestLevel;
		HarvestArchaeoPoints = HarvestPoints;
	}

	HarvestProgressMigrated = 1;
	return;
}

function	script	F_FarmPrototype_Click	{
	callfunc "F_Harvest_MigrateProgress";
	.@x = getarg(0);
	.@y = getarg(1);
	.@stand_x = getarg(2);
	.@stand_y = getarg(3);
	.@need = getarg(4);
	.@resource_type = getarg(5,0);
	.@drop_item = getarg(6,7201);

	getmapxy .@map$, .@pc_x, .@pc_y, UNITTYPE_PC;
	if (.@map$ != "prontera" || distance(.@pc_x,.@pc_y,.@x,.@y) > 4)
		end;

	if (.Busy)
		end;

	if (.@resource_type) {
		if (!HarvestJob) {
			mes "[Harvest Master]";
			mes "Tu dois d'abord choisir un metier de recolte.";
			close;
		}
		if (HarvestJob != .@resource_type) {
			mes "[Harvest Master]";
			if (.@resource_type == 1)
				mes "Cette ressource est reservee aux Herboristes.";
			else if (.@resource_type == 2)
				mes "Cette ressource est reservee aux Mineurs.";
			else
				mes "Cette ressource est reservee aux Archeologues.";
			close;
		}
	}

	.Busy = 1;

	if (distance(.@pc_x,.@pc_y,.@x,.@y) > 1) {
		unitwalk(getcharid(CHAR_ID_ACCOUNT), .@stand_x, .@stand_y);
		.@walk_wait = 0;
		while (.@walk_wait < 6) {
			sleep2 500;
			getmapxy .@map$, .@pc_x, .@pc_y, UNITTYPE_PC;
			if (.@map$ == "prontera" && distance(.@pc_x,.@pc_y,.@x,.@y) <= 1)
				break;
			.@walk_wait++;
		}
		if (.@map$ != "prontera" || distance(.@pc_x,.@pc_y,.@x,.@y) > 1) {
			.Busy = 0;
			end;
		}
	}

	.Hits++;
	if (.Hits < .@need) {
		.Busy = 0;
		end;
	}

	.Hits = 0;
	specialeffect EF_SONIC_CLAW965;
	makeitem .@drop_item, 1, "prontera", .@x, .@y, 1;
	HarvestGatherCount++;
	if (.@resource_type == 1)
		HarvestHerbGatherCount++;
	else if (.@resource_type == 2)
		HarvestMiningGatherCount++;
	else if (.@resource_type == 3)
		HarvestArchaeoGatherCount++;
	else
		HarvestBoxGatherCount++;

	if (.@resource_type) {
		if (.@resource_type == 1) {
			.@points = HarvestHerbPoints;
			.@level = HarvestHerbLevel;
			.@job_name$ = "Herboriste";
		}
		else if (.@resource_type == 2) {
			.@points = HarvestMiningPoints;
			.@level = HarvestMiningLevel;
			.@job_name$ = "Mineur";
		}
		else {
			.@points = HarvestArchaeoPoints;
			.@level = HarvestArchaeoLevel;
			.@job_name$ = "Archeologue";
		}
		if (!.@level)
			.@level = 1;

		if (.@points < 100)
			.@points++;

		if (.@level < 2 && .@points >= 10) {
			.@level = 2;
			dispbottom .@job_name$ + " passe niveau 2.";
		}
		else if (.@level < 3 && .@points >= 20) {
			.@level = 3;
			dispbottom .@job_name$ + " passe niveau 3.";
		}
		else
			dispbottom .@job_name$ + " +1 point. Total : " + .@points + "/100.";

		if (.@resource_type == 1) {
			HarvestHerbPoints = .@points;
			HarvestHerbLevel = .@level;
		}
		else if (.@resource_type == 2) {
			HarvestMiningPoints = .@points;
			HarvestMiningLevel = .@level;
		}
		else {
			HarvestArchaeoPoints = .@points;
			HarvestArchaeoLevel = .@level;
		}
	}

	movenpc(strnpcinfo(NPC_NAME),1,1,4);
	disablenpc(strnpcinfo(NPC_NAME));
	sleep 10000;
	enablenpc(strnpcinfo(NPC_NAME));
	movenpc(strnpcinfo(NPC_NAME),.@x,.@y,4);
	.Busy = 0;
	end;
}
