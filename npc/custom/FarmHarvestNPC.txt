prontera,159,154,4	script	Harvest Master	4_F_VALKYRIE,{
	mes "[Harvest Master]";
	mes "The prototypes keep their visual test skins.";
	close;
}

prontera,163,149,4	script	Farm Proto Creeper#farm01	4_CREEPER,{
	getmapxy .@map$, .@x, .@y, UNITTYPE_PC;
	if (.@map$ != "prontera" || distance(.@x,.@y,163,149) > 2)
		close;

	.Hits++;
	mes "[Farm Prototype]";
	mes "Progress: " + .Hits + "/" + .NeedClicks;

	if (.Hits < .NeedClicks)
		close;

	.Hits = 0;
	makeitem 7201, 1, "prontera", 163, 149, 1;
	close2;
	disablenpc strnpcinfo(NPC_NAME);
	sleep 10000;
	enablenpc strnpcinfo(NPC_NAME);
	end;

OnInit:
	.Hits = 0;
	.NeedClicks = 1;
	setnpcdistance 2;
	end;
}
