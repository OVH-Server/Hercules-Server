//===== Hercules Script ======================================
// Farming and lumberjack prototype resources.
//============================================================

prontera,159,154,4	script	Harvest Master	4_F_VALKYRIE,{
	mes "[Harvest Master]";
	mes "The prototypes nearby are attackable test resources.";
	mes "No profession or distance check is required.";
	close;
}

-	script	Farm Resources	FAKE_NPC,{
	end;

OnInit:
	.TreeMap$ = "prontera";
	.TreeX = 163;
	.TreeY = 154;
	.PlantMap$ = "prontera";
	.PlantX = 163;
	.PlantY = 152;
	.TreeMob = 35000;
	.PlantMob = 35001;
	.RespawnDelay = 3000;
	.ProtoRespawnDelay = 3000;

	killmonster .TreeMap$, strnpcinfo(NPC_NAME) + "::OnTreeDead";
	killmonster .PlantMap$, strnpcinfo(NPC_NAME) + "::OnPlantDead";
	killmonster .TreeMap$, strnpcinfo(NPC_NAME) + "::OnProto01Dead";
	killmonster .TreeMap$, strnpcinfo(NPC_NAME) + "::OnProto02Dead";
	killmonster .TreeMap$, strnpcinfo(NPC_NAME) + "::OnProto03Dead";
	killmonster .TreeMap$, strnpcinfo(NPC_NAME) + "::OnProto04Dead";
	killmonster .TreeMap$, strnpcinfo(NPC_NAME) + "::OnProto05Dead";

	callsub S_SpawnTree;
	callsub S_SpawnPlant;
	callsub S_SpawnProto01;
	callsub S_SpawnProto02;
	callsub S_SpawnProto03;
	callsub S_SpawnProto04;
	callsub S_SpawnProto05;
	end;

S_SpawnTree:
	monster .TreeMap$, .TreeX, .TreeY, "Harvest Tree", .TreeMob, 1, strnpcinfo(NPC_NAME) + "::OnTreeDead";
	return;

S_SpawnPlant:
	monster .PlantMap$, .PlantX, .PlantY, "Harvest Plant", .PlantMob, 1, strnpcinfo(NPC_NAME) + "::OnPlantDead";
	return;

OnTreeDead:
	makeitem 7201, 1, .TreeMap$, .TreeX, .TreeY, 1;
	sleep .RespawnDelay;
	callsub S_SpawnTree;
	end;

OnPlantDead:
	makeitem 7872, 1, .PlantMap$, .PlantX, .PlantY, 1;
	sleep .RespawnDelay;
	callsub S_SpawnPlant;
	end;

S_SpawnProto01:
	monster .TreeMap$, 163, 149, "Farm Proto Mandragora", 1020, 1, strnpcinfo(NPC_NAME) + "::OnProto01Dead";
	return;

S_SpawnProto02:
	monster .TreeMap$, 163, 146, "Farm Proto Green Plant", 1080, 1, strnpcinfo(NPC_NAME) + "::OnProto02Dead";
	return;

S_SpawnProto03:
	monster .TreeMap$, 163, 143, "Farm Proto Blue Plant", 1079, 1, strnpcinfo(NPC_NAME) + "::OnProto03Dead";
	return;

S_SpawnProto04:
	monster .TreeMap$, 159, 143, "Farm Proto Wooden Golem", 1497, 1, strnpcinfo(NPC_NAME) + "::OnProto04Dead";
	return;

S_SpawnProto05:
	monster .TreeMap$, 159, 146, "Farm Proto Flora", 1118, 1, strnpcinfo(NPC_NAME) + "::OnProto05Dead";
	return;

OnProto01Dead:
	sleep .ProtoRespawnDelay;
	callsub S_SpawnProto01;
	end;

OnProto02Dead:
	sleep .ProtoRespawnDelay;
	callsub S_SpawnProto02;
	end;

OnProto03Dead:
	sleep .ProtoRespawnDelay;
	callsub S_SpawnProto03;
	end;

OnProto04Dead:
	sleep .ProtoRespawnDelay;
	callsub S_SpawnProto04;
	end;

OnProto05Dead:
	sleep .ProtoRespawnDelay;
	callsub S_SpawnProto05;
	end;
}
