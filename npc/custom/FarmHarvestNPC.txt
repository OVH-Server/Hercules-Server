	mes "[Harvest Master]";
	mes "The prototypes keep their visual test skins.";
	close;
}

function	script	F_FarmPrototype_Click	{
	.@x = getarg(0);
	.@y = getarg(1);

	getmapxy .@map$, .@pc_x, .@pc_y, UNITTYPE_PC;
	if (.@map$ != "prontera" || distance(.@pc_x,.@pc_y,.@x,.@y) > 2)
		end;

	makeitem 7201, 1, "prontera", .@x, .@y, 1;
	disablenpc strnpcinfo(NPC_NAME);
	sleep 10000;
	enablenpc strnpcinfo(NPC_NAME);
	end;
}
