prontera,159,157,4	script	Joueurs en ligne	4_MACHINE_DEVICE,{
	mes "[Joueurs en ligne]";
	mes "Joueurs actuellement entrain de game :";
	next;

	.@count = getunits(BL_PC, .@players);

	if (.@count <= 0) {
		mes "Aucun joueur en ligne.";
		close;
	}

	for (.@i = 0; .@i < .@count; .@i++) {

		// On attache le joueur pour lire ses infos
		if (attachrid(.@players[.@i])) {

			mes "- " + strcharinfo(0) 
			    + " | Classe : " + jobname(Class) 
			    + " | Levels : " + BaseLevel;

			detachrid;
		}
	}

	next;
	mes "Total : " + .@count + " joueur(s) en ligne.";
	close;
}
