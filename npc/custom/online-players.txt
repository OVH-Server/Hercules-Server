//===== Hercules Script ======================================

prontera,146,197,4	script	Online Players	4_MACHINE_DEVICE,{
	mes "[Online Players]";
	mes "Players currently online:";
	next;

	.@count = getunits(BL_PC, .@players, false);

	if (.@count <= 0) {
		mes "No players online.";
		close;
	}

	for (.@i = 0; .@i < .@count; .@i++) {
		mes "- " + rid2name(.@players[.@i]);
	}

	next;

	mes "Total: " + .@count + " player(s) online.";
	close;
}

//Duplicate
gon_in,148,20,4	duplicate(Online Players)	Online Players	4_MACHINE_DEVICE
