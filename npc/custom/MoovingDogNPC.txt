//===== Hercules Script ======================================
// Walking dog that appears on reload, walks, then disappears.
//============================================================

prontera,161,170,4	script	Walking Dog#move	4_DOG01,{
	end;

OnInit:
	npcspeed 100;
	disablenpc strnpcinfo(NPC_NAME);
	initnpctimer;
	end;

OnTimer1000:
	enablenpc strnpcinfo(NPC_NAME);
	npcwalkto 161,191;
	end;

OnTimer3900:
	npcwalkto 157,194;
	end;

OnTimer4800:
	npcwalkto 152,192;
	end;

OnTimer5600:
	npcwalkto 152,152;
	end;

OnTimer12800:
	npcwalkto 126,143;
	end;

OnTimer18500:
	disablenpc strnpcinfo(NPC_NAME);
	stopnpctimer;
	end;
}
