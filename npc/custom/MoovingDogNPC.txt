//===== Hercules Script ======================================
// Walking dog that appears on reload, walks, then disappears.
//============================================================

prontera,161,170,4	script	Walking Dog#move	4_DOG01,{
	end;

OnInit:
	npcspeed 100;
	disablenpc strnpcinfo(NPC_NAME);
	initnpctimer;
	end;

OnTimer1000:
	enablenpc strnpcinfo(NPC_NAME);
	npcwalkto 161,191;
	end;

OnTimer3900:
	npcwalkto 157,194;
	end;

OnTimer4800:
	npcwalkto 152,192;
	end;

OnTimer6000:
	npcwalkto 152,176;
	end;

OnTimer8400:
	npcwalkto 152,160;
	end;

OnTimer10400:
	npcwalkto 152,152;
	end;

OnTimer11800:
	npcwalkto 126,143;
	end;

OnTimer19000:
	disablenpc strnpcinfo(NPC_NAME);
	stopnpctimer;
	end;
}
