//===== Hercules Script ======================================
// Walking dog that appears on reload, walks, then disappears.
//============================================================

prontera,161,170,4	script	Walking Dog#move	4_DOG01,{
	end;

OnInit:
	npcspeed 120;
	disablenpc strnpcinfo(NPC_NAME);
	initnpctimer;
	end;

OnTimer1000:
	enablenpc strnpcinfo(NPC_NAME);
	npcwalkto 157,194;
	end;

OnTimer7000:
	npcwalkto 152,192;
	end;

OnTimer12000:
	npcwalkto 126,143;
	end;

OnTimer26000:
	npcwalkto 104,118;
	end;

OnTimer42000:
	disablenpc strnpcinfo(NPC_NAME);
	stopnpctimer;
	end;
}
