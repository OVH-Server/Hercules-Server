//===== Hercules Script ======================================
// Walking dog that appears on reload, walks, then disappears.
//============================================================

prontera,161,170,4	script	Walking Dog#move	4_DOG01,{
	end;

OnInit:
	npcspeed 120;
	disablenpc strnpcinfo(NPC_NAME);
	initnpctimer;
	end;

OnTimer1000:
	enablenpc strnpcinfo(NPC_NAME);
	callsub S_DogMoveTo, 161,191, 6000;
	sleep 150;
	callsub S_DogMoveTo, 157,194, 3000;
	sleep 150;
	callsub S_DogMoveTo, 152,192, 3000;
	sleep 150;
	callsub S_DogMoveTo, 152,152, 9000;
	disablenpc strnpcinfo(NPC_NAME);
	stopnpctimer;
	end;

S_DogMoveTo:
	npcwalkto getarg(0),getarg(1);
	.@timeout = getarg(2);
	.@elapsed = 0;
	while (.@elapsed < .@timeout) {
		getmapxy(.@map$, .@x, .@y, UNITTYPE_NPC);
		.@dx = getarg(0) - .@x;
		.@dy = getarg(1) - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@dx <= 1 && .@dy <= 1)
			return;
		sleep 100;
		.@elapsed += 100;
	}
	return;
}
