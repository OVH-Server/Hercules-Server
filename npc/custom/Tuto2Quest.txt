//===== Hercules Script ======================================
// Main Quest Tuto 2
// Valkyrie starts a repeatable quest and calls a merchant taxi.
// The taxi waits beside her, picks up the player, carries them along
// its route, then returns beside the Valkyrie.
//============================================================

// Quest ID:
// 65010 = taxi introduction, repeatable

// Character variables:
// Tuto2_State: ride counter / last state marker only

//============================================================
// NPC 1 - Valkyrie
//============================================================

prontera,159,167,4	script	Valkyrie Tuto 2	4_F_VALKYRIE,{
	callfunc "F_DebugConsole","Valkyrie Tuto 2","Click. State=" + Tuto2_State;

	mes "[Valkyrie]";

	mes "Welcome to the next step of your training.";
	mes "My merchant taxi can take you farther into Prontera.";
	mes "Are you ready to leave?";
	next;

	if (select("Yes, I am ready:Not yet") == 2) {
		mes "[Valkyrie]";
		mes "Come back when you are ready.";
		close;
	}

	if ($Tuto2_TaxiBusy) {
		mes "[Valkyrie]";
		mes "The taxi is already on the road.";
		mes "Wait a few seconds, then try again.";
		close;
	}

	if (questprogress(65010))
		erasequest 65010;

	$Tuto2_TaxiBusy = getcharid(CHAR_ID_ACCOUNT);
	setquest 65010;
	getmapxy(.@pc_map$, .@pc_x, .@pc_y, UNITTYPE_PC);

	mes "[Valkyrie]";
	mes "Perfect. The merchant is coming to pick you up.";
	close2;

	setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND|PCBLOCK_NPC, true);
	movenpc "Merchant Taxi#tuto2",160,167,4;
	unittalk(getnpcid("Merchant Taxi#tuto2"), "Hop in, follow me!");
	sleep2 600;

	callsub S_RideStep, 160,168, 160,167, 350;
	callsub S_RideStep, 160,169, 160,168, 350;
	callsub S_RideStep, 160,170, 160,169, 350;
	callsub S_RideStep, 159,171, 160,170, 350;
	callsub S_RideStep, 159,172, 159,171, 350;
	callsub S_RideStep, 159,173, 159,172, 350;
	callsub S_RideStep, 158,174, 159,173, 350;
	callsub S_RideStep, 158,175, 158,174, 350;
	callsub S_RideStep, 158,176, 158,175, 350;
	callsub S_RideStep, 157,177, 158,176, 350;
	callsub S_RideStep, 157,178, 157,177, 350;
	callsub S_RideStep, 157,179, 157,178, 350;
	callsub S_RideStep, 156,180, 157,179, 350;
	callsub S_RideStep, 156,181, 156,180, 350;
	callsub S_RideStep, 156,182, 156,181, 350;
	callsub S_RideStep, 155,183, 156,182, 350;
	callsub S_RideStep, 155,184, 155,183, 350;
	callsub S_RideStep, 155,185, 155,184, 350;
	callsub S_RideStep, 155,186, 155,185, 350;
	callsub S_RideStep, 155,187, 155,186, 350;
	callsub S_RideStep, 155,188, 155,187, 350;
	callsub S_RideStep, 155,189, 155,188, 350;
	callsub S_RideStep, 155,190, 155,189, 350;
	unitwalk(getcharid(CHAR_ID_ACCOUNT), 155, 190);
	sleep2 500;
	unittalk(getnpcid("Merchant Taxi#tuto2"), "We are here!");

	Tuto2_State++;
	completequest 65010;
	erasequest 65010;
	message strcharinfo(PC_NAME), "Main Quest Tuto 2: taxi ride complete. You can restart the quest whenever you want.";

	movenpc "Merchant Taxi#tuto2",160,167,4;
	setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND|PCBLOCK_NPC, false);
	$Tuto2_TaxiBusy = 0;
	end;

S_RideStep:
	.@taxi_x = getarg(0);
	.@taxi_y = getarg(1);
	.@player_x = getarg(2);
	.@player_y = getarg(3);
	.@delay = getarg(4);
	movenpc "Merchant Taxi#tuto2",.@taxi_x,.@taxi_y,4;
	unitwalk(getcharid(CHAR_ID_ACCOUNT), .@player_x, .@player_y);
	sleep2 .@delay;
	return;

OnPCLogoutEvent:
	if ($Tuto2_TaxiBusy == getcharid(CHAR_ID_ACCOUNT)) {
		movenpc "Merchant Taxi#tuto2",160,167,4;
		$Tuto2_TaxiBusy = 0;
	}
	end;
}

//============================================================
// Taxi merchant
//============================================================

prontera,160,167,4	script	Merchant Taxi#tuto2	5,{
	mes "[Merchant Taxi]";
	mes "I am the Valkyrie's taxi.";
	mes "Talk to her to start the ride.";
	close;

OnInit:
	movenpc "Merchant Taxi#tuto2",160,167,4;
	setunitdata(getnpcid("Merchant Taxi#tuto2"), UDT_SPEED, 70);
	end;
}
