//===== Hercules Script ======================================
// Main Quest Tuto 2
// Valkyrie starts a repeatable quest and calls a merchant taxi.
// The taxi waits beside her, picks up the player, carries them along
// its route, then returns beside the Valkyrie.
//============================================================

// Quest ID:
// 65010 = taxi introduction, repeatable

// Character variables:
// Tuto2_State: ride counter / last state marker only

//============================================================
// NPC 1 - Valkyrie
//============================================================

prontera,159,167,4	script	Valkyrie Tuto 2	4_F_VALKYRIE,{
	callfunc "F_DebugConsole","Valkyrie Tuto 2","Click. State=" + Tuto2_State;

	mes "[Valkyrie]";

	mes "Bienvenue dans la suite de ton entrainement.";
	mes "Mon taxi marchand peut t'emmener plus loin dans Prontera.";
	mes "Es-tu pret a partir ?";
	next;

	if (select("Oui je suis pret:Pas encore") == 2) {
		mes "[Valkyrie]";
		mes "Reviens me voir quand tu seras pret.";
		close;
	}

	if ($Tuto2_TaxiBusy) {
		mes "[Valkyrie]";
		mes "Le taxi est deja en route.";
		mes "Attends quelques secondes puis reessaie.";
		close;
	}

	if (questprogress(65010))
		erasequest 65010;

	$Tuto2_TaxiBusy = getcharid(CHAR_ID_ACCOUNT);
	setquest 65010;
	getmapxy(.@pc_map$, .@pc_x, .@pc_y, UNITTYPE_PC);

	mes "[Valkyrie]";
	mes "Parfait. Le marchand vient te chercher.";
	close2;

	.@taxi_gid = getnpcid("Taxi Marchand#tuto2");
	setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND|PCBLOCK_NPC, true);
	setunitdata(.@taxi_gid, UDT_SPEED, 70);
	unitwalk(.@taxi_gid, .@pc_x, .@pc_y);
	unittalk(.@taxi_gid, "J'arrive, ne bouge pas !");
	sleep2 1200;
	unittalk(.@taxi_gid, "Installe-toi, depart immediat !");

	callsub S_RideStep, 160,170, 160,169, 650;
	callsub S_RideStep, 159,173, 159,172, 650;
	callsub S_RideStep, 158,176, 158,175, 650;
	callsub S_RideStep, 157,179, 157,178, 650;
	callsub S_RideStep, 156,182, 156,181, 650;
	callsub S_RideStep, 155,185, 155,184, 650;
	callsub S_RideStep, 155,188, 155,187, 650;
	callsub S_RideStep, 155,190, 155,189, 650;
	unitwalk(getcharid(CHAR_ID_ACCOUNT), 155, 190);
	sleep2 650;

	Tuto2_State++;
	completequest 65010;
	erasequest 65010;
	message strcharinfo(PC_NAME), "Main Quest Tuto 2: taxi ride complete. Tu peux relancer la quete quand tu veux.";

	unitwarp(.@taxi_gid, "prontera", 160, 167);
	setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND|PCBLOCK_NPC, false);
	$Tuto2_TaxiBusy = 0;
	end;

S_RideStep:
	.@taxi_x = getarg(0);
	.@taxi_y = getarg(1);
	.@player_x = getarg(2);
	.@player_y = getarg(3);
	.@delay = getarg(4);
	.@taxi_gid = getnpcid("Taxi Marchand#tuto2");

	unitwalk(.@taxi_gid, .@taxi_x, .@taxi_y);
	unitwalk(getcharid(CHAR_ID_ACCOUNT), .@player_x, .@player_y);
	sleep2 .@delay;
	return;

OnPCLogoutEvent:
	if ($Tuto2_TaxiBusy == getcharid(CHAR_ID_ACCOUNT)) {
		unitwarp(getnpcid("Taxi Marchand#tuto2"), "prontera", 160, 167);
		$Tuto2_TaxiBusy = 0;
	}
	end;
}

//============================================================
// Taxi merchant
//============================================================

prontera,160,167,4	script	Taxi Marchand#tuto2	1_M_MERCHANT,{
	mes "[Taxi Marchand]";
	mes "Je suis le taxi de la Valkyrie.";
	mes "Parle-lui pour lancer le trajet.";
	close;

OnInit:
	setunitdata(getnpcid("Taxi Marchand#tuto2"), UDT_SPEED, 70);
	end;
}
