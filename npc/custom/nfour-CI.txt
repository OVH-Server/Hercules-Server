// The global configutation of the CI, can be edited in the OnInit section of this NPC
prontera,150,185,0	script	InstantCbt	4_M_ROGUE,{

    .@party_id = 4444;
    .@md_name$ = _("Combat Instance");

	if ($ci_is_open == 0) {
		mes("[InstantCbt]");
        mes("Instance combat is not open");
		close();
	}


    donpcevent("InstantCbt::OnStart");

    .@str$ = sprintf(_$("Enter to the %s"), .@md_name$);
    switch (select(.@str$, "Cancel")) {
    case 1:
        mes("[InstantCbt]");
        mesf("Current instance with id: %d", $ci_instance_id);
        next();
        break;
    case 2:
        close();
    }

    // Dungeon access
    if (has_instance($ci_instance_map_name$) == "") {
        mes("[InstantCbt]");
        mesf("Instance %s does not exist.", .@md_name$);
        close();
    }

    mapannounce("prontera", sprintf(_$("%s of the party, %s is entering the %s."), strcharinfo(PC_NAME), .@p_name$, .@md_name$), bc_all, C_SPRINGGREEN);
    .@map$ = instance_mapname("moc_pryd04", $ci_instance_id);
    announce("[DEBUG] Warping to: " + .@map$, bc_self, 0x00FF00);
    warp(.@map$, $ci_player_starting_pos_x, $ci_player_starting_pos_y);
    end;

OnStart:
    if ($ci_instance_id != -1 && has_instance($ci_instance_map_name$, $ci_instance_id) != "") {
        return ;
    }

    $ci_instance_id = instance_create(.@md_name$, .@party_id, IOT_NONE);
    if ($ci_instance_id < 0) {
        return ;
    }
    if (instance_attachmap($ci_instance_map_name$, $ci_instance_id, true, "combat") == "") {
        return ;
    }
    // 600 seconds instance timeout
    // 120 seconds without player timeout
    instance_set_timeout(600, 120, $ci_instance_id);
    instance_init($ci_instance_id);
    end;

OnInit:

    /*
     Monster waves configuration
     Format: [Mob_ID, Quantity]
     Need to have a comma ',' between line exept for the last line example:

        setarray($ci_mob_wave_array[0],
            PORING, 1,      // ',' at the end
            PORING, 2,      // ',' at the end
            PORING, 3       // no ',' at the end because this is the last one
        );
    
    */

    setarray($ci_mob_wave_array[0],
        PORING, 1,      // ',' at the end
        PORING, 2,      // ',' at the end
        PORING, 3       // no ',' at the end because this is the last one
    );

    /* Total number of wave */
    $ci_total_mob_wave = 3;

    /* Starting position of player on the map */
    $ci_player_starting_pos_x = 60;
    $ci_player_starting_pos_y = 100;

    /*
        $ci_monster_spawn_offset
        
        This value are used to spawn monster arround the player groups
        
        For exemple with :
            $ci_player_starting_pos_x = 60;
            $ci_player_starting_pos_y = 100;
        and 
            $ci_monster_spawn_offset = 10;
        
        Monster will spawn: 
            between 50 and 70 in x
            between 90 and 110 in y
    */
    $ci_monster_spawn_offset = 10;

    /*
        Number of seconds between each wave spawn
        This time is use for the end of the last wave to
    */
    $ci_nb_second_between_wave = 5;

    /* Number second before starting */
    $ci_nb_second_before_start = 10;

    /* Instance map from */
    $ci_instance_map_name$ = "moc_pryd04";

    if ($ci_instance_id != -1 && has_instance($ci_instance_map_name$, $ci_instance_id) != "") {
        announce("[CI] On init destroy instance" , bc_all, 0xFF0000);
        instance_destroy($ci_instance_id);
    }
    announce("[CI] On init NO destroy current id : " + $ci_instance_id , bc_all, 0xFF0000);
    
    /* Reset instance id */
    $ci_instance_id = -1;
    end;

}

moc_pryd04,70,110,0	script	Combat_Controller	FAKE_NPC,{

OnInstanceInit:
    announce("[DEBUG] OnInstanceInit started", bc_all, 0x0000FF);

    .current_wave = 0;

    // Load array from OnInit block, load total wawe number to   
	copyarray(.mob_waves[0], $ci_mob_wave_array[0], getarraysize($ci_mob_wave_array));
	.total_waves = $ci_total_mob_wave;

    announce(sprintf("Waves configured. Starting in %d seconds...", $ci_nb_second_before_start), bc_all, 0x0000FF);

    announce(sprintf("Battle starts in %d seconds!", $ci_nb_second_before_start), bc_all, 0xFF0000);
    
    sleep($ci_nb_second_before_start * 1000);
    announce("[DEBUG] Calling OnStartCombat", bc_all, 0x0000FF);
    donpcevent(instance_npcname("Combat_Controller")+"::OnStartCombat");
    end;

OnStartCombat:
    announce("[DEBUG] OnStartCombat triggered", bc_all, 0x0000FF);
    announce("The battle begins! First wave!", bc_all, 0xFF0000);
    donpcevent(instance_npcname("Combat_Controller")+"::OnSpawnWave");
    end;

OnSpawnWave:
    announce("[DEBUG] OnSpawnWave - Current wave: " + .current_wave, bc_all, 0x0000FF);
    
    if (.current_wave >= .total_waves) {
        announce("[DEBUG] All waves completed", bc_all, 0x0000FF);
        end;
    }
    
    // Get current wave info
    .@mob_id = .mob_waves[.current_wave * 2];
    .@mob_count = .mob_waves[.current_wave * 2 + 1];
    .@map$ = instance_mapname("moc_pryd04");
    
    announce("[DEBUG] Map name: " + .@map$, bc_all, 0x0000FF);
    announce("[DEBUG] Spawning " + .@mob_count + " mobs (ID: " + .@mob_id + ")", bc_all, 0x0000FF);
    
    // Spawn mobs
    .@spawned = 0;
    for (.@i = 0; .@i < .@mob_count; ++.@i) {
        .@x = $ci_player_starting_pos_x + rand(-$ci_monster_spawn_offset, $ci_monster_spawn_offset);
        .@y = $ci_player_starting_pos_y + rand(-$ci_monster_spawn_offset, $ci_monster_spawn_offset);
        .@gid = monster(.@map$, .@x, .@y, "--ja--", .@mob_id, 1, instance_npcname("Combat_Controller")+"::OnMobKilled");
        if (.@gid > 0) {
            ++.@spawned;
        }
    }
    
    .mobs_remaining = .@spawned;
    announce("[DEBUG] Successfully spawned: " + .@spawned + " mobs", bc_all, 0x0000FF);
    announce("Wave " + (.current_wave + 1) + ": " + .@mob_count + " " + getmonsterinfo(.@mob_id, MOB_NAME) + " appear!", bc_all, 0xFFFF00);
    end;

OnMobKilled:
    --.mobs_remaining;
    announce("[DEBUG] Mob killed. Remaining: " + .mobs_remaining, bc_all, 0x0000FF);
    
    if (.mobs_remaining <= 0) {
        ++.current_wave;
        announce("[DEBUG] Wave completed. Moving to wave: " + .current_wave, bc_all, 0x0000FF);
        

        announce(sprintf("Wave completed! Next wave in %d seconds...", $ci_nb_second_between_wave), bc_all, 0x00FF00);
        sleep($ci_nb_second_between_wave * 1000);

        if (.current_wave >= .total_waves) {
            // All waves completed
            announce("[DEBUG] Calling OnCombatEnd", bc_all, 0x0000FF);
            donpcevent(instance_npcname("Combat_Controller")+"::OnCombatEnd");
        } else {
            donpcevent(instance_npcname("Combat_Controller")+"::OnSpawnWave");
        }
    }
    end;

OnCombatEnd:
    announce("[DEBUG] OnCombatEnd started", bc_all, 0x0000FF);
    .@map$ = instance_mapname("moc_pryd04");
    
    announce("Combat complete! Teleporting to center...", bc_all, 0x00FF00);
    
    // Teleport all players to center (below the central square)
    instance_warpall(.@map$, 100, 50);
    
    announce("20 secondes remaining before the end of the instant battle!", bc_all, 0xFF0000);

    // wait 5 sec
    sleep(5000);
    
    // Spawn 1 Yggdrasil Berries each seconde around players, spawn 10 Yggdrasil in 10 secondes
    for (.@i = 0; .@i < 10; ++.@i) {
        makeitem("Yggdrasilberry", 1, .@map$, 100 + rand(10), 50 + rand(10));
        sleep(1000);
    }
    
    // wait 5 sec with announce number...
    for (.@i = 5; .@i > 0; --.@i) {
        announce(.@i + "...", bc_all, 0xFF0000);
        sleep(1000);
    }
    
    // Teleport everyone to Prontera
    announce("[DEBUG] Warping all to Prontera", bc_all, 0x0000FF);
    instance_warpall("prontera", 155, 185);
    
    // Destroy instance after a delay
    sleep(5000);
    if ($ci_instance_id != -1 && has_instance($ci_instance_map_name$, $ci_instance_id) != "") {
        announce("[CI] OnCombatEnd destroy instance" , bc_all, 0xFF0000);
        instance_destroy($ci_instance_id);
        $ci_instance_id = -1;
    }
    end;
}

-	script	CI_Timer_Test	-1,{

// Cette section crée une commande utilisable en jeu : @testci
OnCommand:
    // On simule exactement ce qui se passera à 12h, 14h, etc.
    goto L_StartEvent;

OnClock0000: OnClock0200: OnClock0400: OnClock0600:
OnClock0800: OnClock1000: OnClock1200: OnClock1400:
OnClock1600: OnClock1800: OnClock2000: OnClock2200:
L_StartEvent:
    $ci_is_open = 1;
    announce "[DEBUG] Arena OPEN", bc_all, 0x00FF00;
    
    // Wait 2 minutes
    sleep(120 * 1000);
    
    $ci_is_open = 0;
    announce "[DEBUG] Arena close", bc_all, 0xFF0000;
    end;

OnInit:
    $ci_is_open = 0;
	bindatcmd "testci", "CI_Timer_Test::OnCommand", 99, 99, true;
	announce "[DEBUG] Script CI_Timer loaded.", bc_all, 0xFFFF00;
    end;
}
