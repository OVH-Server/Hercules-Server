//===== Hercules Script ======================================
// Hourly account reward
// Gives 1 Yggdrasil Berry every hour per account.
//============================================================

prontera,159,173,4	script	Hourly Bonus	4_VENDING_MACHINE,{
	mes "[Hourly Bonus]";

	.@now = gettimetick(2);
	if (#HourlyBonus_LastClaim && .@now < #HourlyBonus_LastClaim + 3600) {
		.@left = (#HourlyBonus_LastClaim + 3600) - .@now;
		mes "Ta recompense de l'heure n'est pas encore disponible.";
		mes "Temps restant : " + (.@left / 60) + " minute(s).";
		close;
	}

	if (!checkweight(607,1)) {
		mes "Ton inventaire est trop lourd.";
		mes "Reviens avec un peu de place pour recevoir ta Yggdrasil Berry.";
		close;
	}

	getitem 607,1;
	#HourlyBonus_LastClaim = .@now;

	mes "Voici ta recompense de l'heure :";
	mes "^00AAFF" + getitemname(607) + "^000000 x1";
	dispbottom "[Hourly Bonus] Tu as recu 1 " + getitemname(607) + ".";
	close;

OnPCLoginEvent:
	callsub S_NotifyPlayer;
	end;

OnInit:
	initnpctimer;
	end;

OnTimer60000:
	callsub S_NotifyOnlinePlayers;
	setnpctimer 0;
	end;

S_NotifyOnlinePlayers:
	.@count = getunits(BL_PC, .@players, false);
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@players[.@i]))
			continue;

		callsub S_NotifyPlayer;
	}
	detachrid;
	return;

S_NotifyPlayer:
	.@now = gettimetick(2);
	if (!#HourlyBonus_LastClaim || .@now >= #HourlyBonus_LastClaim + 3600) {
		.@msg$ = "[Hourly Bonus] Ta recompense de l'heure est disponible a Prontera 159 173.";
		message getcharid(CHAR_ID_ACCOUNT), .@msg$;
		dispbottom .@msg$;
	}
	return;
}
