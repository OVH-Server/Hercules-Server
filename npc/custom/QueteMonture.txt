//===== Hercules Script ======================================
//============================================================
// Quête Halter Lead
// Prontera 161,185
//============================================================

prontera,161,185,4	script	Chasseuse de Monture	4_F_DOMINO,{
	mes "[Chasseuse de Monture]";

	if (#HalterQuest_Done) {
		mes "Tu as déjà reçu ta récompense spéciale.";
		close;
	}

	if (!#HalterQuest_Start) {
		mes "Tu veux obtenir une récompense spéciale ?";
		mes "Alors va tuer :";
		mes "- 10 Poring";
		mes "- 10 Archer Skeleton";
		mes "- 10 Les";
		next;

		if (select("Accepter:Refuser") == 2) {
			mes "[Chasseuse de Monture]";
			mes "Reviens quand tu seras prêt.";
			close;
		}

		#HalterQuest_Start = 1;
		#HalterQuest_Poring = 0;
		#HalterQuest_Archer = 0;
		#HalterQuest_Les = 0;

		mes "[Chasseuse de Monture]";
		mes "Très bien. Reviens quand tu auras terminé.";
		close;
	}

	mes "Progression actuelle :";
	mes "Poring : " + #HalterQuest_Poring + "/10";
	mes "Archer Skeleton : " + #HalterQuest_Archer + "/10";
	mes "Les : " + #HalterQuest_Les + "/10";
	next;

	if (#HalterQuest_Poring >= 10 && #HalterQuest_Archer >= 10 && #HalterQuest_Les >= 10) {
		mes "[Chasseuse de Monture]";
		mes "Tu as rempli les conditions.";
		mes "Voici ta récompense spéciale.";
		getitem 12622,1;

		#HalterQuest_Done = 1;
		close;
	}

	mes "[Chasseuse de Monture]";
	mes "Il te manque encore quelques monstres.";
	close;


OnNPCKillEvent:
	if (!#HalterQuest_Start || #HalterQuest_Done)
		end;

	if (killedrid == 1002 && #HalterQuest_Poring < 10) {
		#HalterQuest_Poring++;
		message strcharinfo(PC_NAME), "Quête Halter Lead : Poring " + #HalterQuest_Poring + "/10";
		end;
	}

	if (killedrid == 1016 && #HalterQuest_Archer < 10) {
		#HalterQuest_Archer++;
		message strcharinfo(PC_NAME), "Quête Halter Lead : Archer Skeleton " + #HalterQuest_Archer + "/10";
		end;
	}

	if (killedrid == 1881 && #HalterQuest_Les < 10) {
		#HalterQuest_Les++;
		message strcharinfo(PC_NAME), "Quête Halter Lead : Les " + #HalterQuest_Les + "/10";
		end;
	}

	end;
}
