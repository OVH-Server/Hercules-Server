//===== Hercules Script ======================================
//============================================================
// Halter Lead Quest
// Prontera 161,185
//============================================================

prontera,159,170,4	script	Mount Quest	4_F_DOMINO,{
	mes "[Mount Hunter]";

	// Checks whether this character has already received the reward.
	if (HalterQuest_Done) {
		mes "You have already received your special reward.";
		close;
	}

	// Quest start.
	if (!HalterQuest_Start) {
		mes "Do you want to obtain a special reward?";
		mes "Then go kill:";
		mes "- 10 Poring";
		mes "- 10 Archer Skeleton";
		mes "- 10 Les";
		next;

		if (select("Accept:Decline") == 2) {
			mes "[Mount Hunter]";
			mes "Come back when you are ready.";
			close;
		}

		// Initializes character-bound progress variables.
		HalterQuest_Start = 1;
		HalterQuest_Poring = 0;
		HalterQuest_Archer = 0;
		HalterQuest_Les = 0;

		mes "[Mount Hunter]";
		mes "Very well. Come back when you have finished.";
		close;
	}

	// Displays current progress.
	mes "Current progress:";
	mes "Poring: " + HalterQuest_Poring + "/10";
	mes "Archer Skeleton: " + HalterQuest_Archer + "/10";
	mes "Les: " + HalterQuest_Les + "/10";
	next;

	// Checks whether all objectives are complete.
	if (HalterQuest_Poring >= 10 && HalterQuest_Archer >= 10 && HalterQuest_Les >= 10) {
		mes "[Mount Hunter]";
		mes "You have completed the challenge.";
		mes "Here is your special reward.";

		// Reward: Halter Lead.
		getitem 12622,1;

		HalterQuest_Done = 1;
		close;
	}

	mes "[Mount Hunter]";
	mes "You still have monsters left to kill.";
	close;


OnNPCKillEvent:

	// Prevents counting if the quest has not started
	// or if it is already complete.
	if (!HalterQuest_Start || HalterQuest_Done)
		end;

	// Poring
	if (killedrid == 1002 && HalterQuest_Poring < 10) {
		HalterQuest_Poring++;

		message strcharinfo(PC_NAME),
		"Halter Lead Quest: Poring " + HalterQuest_Poring + "/10";
		end;
	}

	// Archer Skeleton
	if (killedrid == 1016 && HalterQuest_Archer < 10) {
		HalterQuest_Archer++;

		message strcharinfo(PC_NAME),
		"Halter Lead Quest: Archer Skeleton " + HalterQuest_Archer + "/10";
		end;
	}

	// Les
	if (killedrid == 1881 && HalterQuest_Les < 10) {
		HalterQuest_Les++;

		message strcharinfo(PC_NAME),
		"Halter Lead Quest: Les " + HalterQuest_Les + "/10";
		end;
	}

	end;
}
