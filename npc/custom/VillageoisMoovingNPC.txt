// Villageois mobiles - Prontera
// Mobs visibles pilotes par unitwalk, interaction par zones OnTouch invisibles.

-	script	VillageoisMoovingController	-1,{
	end;

OnInit:
	killmonster "prontera","VillageoisMoovingController::OnMobDead";
	.villager1_gid = monster("prontera",163,164,"Villageois Marchand",35002,1,"VillageoisMoovingController::OnMobDead");
	.villager2_gid = monster("prontera",163,160,"Villageois Marchand",35003,1,"VillageoisMoovingController::OnMobDead");

	setunitdata(.villager1_gid, UDT_SPEED, 120);
	setunitdata(.villager2_gid, UDT_SPEED, 120);
	setunitdata(.villager1_gid, UDT_MODE, MD_CANMOVE|MD_NOKNOCKBACK);
	setunitdata(.villager2_gid, UDT_MODE, MD_CANMOVE|MD_NOKNOCKBACK);

	initnpctimer;
	end;

OnTimer1000:
	unitwalk(.villager1_gid, 163, 185);
	unitwalk(.villager2_gid, 155, 160);
	end;

OnTimer6200:
	unitstop(.villager1_gid);
	end;

OnTimer7600:
	unitwalk(.villager1_gid, 163, 164);
	end;

OnTimer10100:
	unitstop(.villager2_gid);
	end;

OnTimer11500:
	unitwalk(.villager2_gid, 163, 160);
	end;

OnTimer12800:
	unitstop(.villager1_gid);
	end;

OnTimer14200:
	unitwalk(.villager1_gid, 163, 185);
	end;

OnTimer20100:
	unitstop(.villager2_gid);
	end;

OnTimer21500:
	unitwalk(.villager2_gid, 155, 160);
	end;

OnTimer23200:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnMobDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "VillageoisMoovingController::OnInit";
	end;
}

prontera,163,174,0	script	VillageoisTouch#prt1	HIDDEN_WARP_NPC,1,22,{
OnTouch:
	callfunc "F_VillageoisMoovingTalk", "VillageoisMoovingController", 1;
	end;
}

prontera,159,160,0	script	VillageoisTouch#prt2	HIDDEN_WARP_NPC,10,1,{
OnTouch:
	callfunc "F_VillageoisMoovingTalk", "VillageoisMoovingController", 2;
	end;
}

function	script	F_VillageoisMoovingTalk	{
	.@controller$ = getarg(0);
	.@which = getarg(1);

	if (.@which == 1)
		.@gid = getvariableofnpc(.villager1_gid, .@controller$);
	else
		.@gid = getvariableofnpc(.villager2_gid, .@controller$);

	if (.@gid <= 0)
		return;

	getunitdata(.@gid, UDT_MAPIDXY, .@mob_pos);
	getmapxy(.@map$, .@pc_x, .@pc_y, UNITTYPE_PC);

	if (.@map$ != "prontera" || distance(.@pc_x, .@pc_y, .@mob_pos[1], .@mob_pos[2]) > 2)
		return;

	if (gettimetick(2) < getvariableofnpc(.last_talk_tick, .@controller$))
		return;

	set getvariableofnpc(.last_talk_tick, .@controller$), gettimetick(2) + 2;
	unitstop(.@gid);
	setunitdata(.@gid, UDT_CANMOVETICK, gettimetick(0) + 1800);

	mes "[Villageois]";
	mes "Tu veux quoi ?";
	sleep2 1500;
	close2;

	if (.@which == 1) {
		if (distance(.@mob_pos[1], .@mob_pos[2], 163, 185) < distance(.@mob_pos[1], .@mob_pos[2], 163, 164))
			unitwalk(.@gid, 163, 164);
		else
			unitwalk(.@gid, 163, 185);
	} else {
		if (distance(.@mob_pos[1], .@mob_pos[2], 155, 160) < distance(.@mob_pos[1], .@mob_pos[2], 163, 160))
			unitwalk(.@gid, 163, 160);
		else
			unitwalk(.@gid, 155, 160);
	}
	return;
}
