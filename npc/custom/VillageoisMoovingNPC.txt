// Fait des allers-retours entre prt 163,164 et prt 163,185.

prontera,163,164,0	script	Villageois Marchand#prt	4028,{
	stopnpctimer;
	npcstop;

	mes "[Villageois]";
	mes "Tu veux quoi ?";
	sleep2 1500;
	close2;

	setnpctimer 0;
	initnpctimer;
	end;

	npcspeed 180;
	initnpctimer;
	end;

OnTimer1000:
	npcwalkto 163,185;
	end;

OnTimer7500:
	npcwalkto 163,164;
	end;

OnTimer14000:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;
}
