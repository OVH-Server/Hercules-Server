prontera,163,164,0	script	Villageois Marchand#prt	4028,{
	stopnpctimer;
	npcstop;

	mes "[Villageois]";
	mes "Tu veux quoi ?";
	sleep2 1500;
	close2;

	setnpctimer 0;
	initnpctimer;
	end;

	npcspeed 100;
	initnpctimer;
	end;

OnTimer1000:
	npcwalkto 163,185;
	end;

OnTimer6500:
	npcwalkto 163,185;
	end;

OnTimer8000:
	npcwalkto 163,164;
	end;

OnTimer13500:
	npcwalkto 163,164;
	end;

OnTimer15000:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;
}
