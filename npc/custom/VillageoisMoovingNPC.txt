//===== Hercules Script ======================================
// Villageois mobiles - Prontera
// Mobs visibles pilotes par unitwalk, interaction par zones OnTouch invisibles.

-	script	VillageoisMoovingController	-1,{
	end;

OnInit:
	killmonster "prontera","VillageoisMoovingController::OnMobDead";
	.villager1_gid = monster("prontera",163,164,"Villageois Marchand",20990,1,"VillageoisMoovingController::OnMobDead");
	.villager2_gid = monster("prontera",163,160,"Villageois Marchand",20991,1,"VillageoisMoovingController::OnMobDead");

	setunitdata(.villager1_gid, UDT_SPEED, 120);
	setunitdata(.villager2_gid, UDT_SPEED, 120);
	setunitdata(.villager1_gid, UDT_MODE, MD_NOKNOCKBACK);
	setunitdata(.villager2_gid, UDT_MODE, MD_NOKNOCKBACK);

	initnpctimer;
	end;

OnTimer1000:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	setunitdata(.villager2_gid, UDT_WALKTOXY, 155, 160);
	end;

OnTimer3000:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 185);
	end;

OnTimer5200:
	unitstop(.villager1_gid);
	end;

OnTimer6600:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	end;

OnTimer8600:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 164);
	end;

OnTimer10200:
	unitstop(.villager1_gid);
	end;

OnTimer10100:
	unitstop(.villager2_gid);
	end;

OnTimer11500:
	setunitdata(.villager2_gid, UDT_WALKTOXY, 163, 160);
	end;

OnTimer14200:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	end;

OnTimer16200:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 185);
	end;

OnTimer18400:
	unitstop(.villager1_gid);
	end;

OnTimer19800:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	end;

OnTimer20100:
	unitstop(.villager2_gid);
	end;

OnTimer21500:
	setunitdata(.villager2_gid, UDT_WALKTOXY, 155, 160);
	end;

OnTimer21800:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 164);
	end;

OnTimer23000:
	unitstop(.villager2_gid);
	end;

OnTimer24000:
	unitstop(.villager1_gid);
	end;

OnTimer25200:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnMobDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "VillageoisMoovingController::OnInit";
	end;
}
