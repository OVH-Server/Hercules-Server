-	script	VillageoisMoovingController	-1,{
	end;

OnInit:
	killmonster "prontera","VillageoisMoovingController::OnMobDead";
	.villager1_gid = monster("prontera",163,164,"Villageois Marchand",20990,1,"VillageoisMoovingController::OnMobDead");
	.villager2_gid = monster("prontera",163,160,"Villageois Marchand",20991,1,"VillageoisMoovingController::OnMobDead");

	setunitdata(.villager1_gid, UDT_SPEED, 120);
	setunitdata(.villager2_gid, UDT_SPEED, 120);
	setunitdata(.villager1_gid, UDT_MODE, MD_NOKNOCKBACK);
	setunitdata(.villager2_gid, UDT_MODE, MD_NOKNOCKBACK);

	initnpctimer;
	end;

OnTimer1000:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	setunitdata(.villager2_gid, UDT_WALKTOXY, 155, 160);
	end;

OnTimer3000:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 185);
	end;

OnTimer5200:
	unitstop(.villager1_gid);
	end;

OnTimer6600:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	end;

OnTimer8600:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 164);
	end;

OnTimer10200:
	unitstop(.villager1_gid);
	end;

OnTimer10100:
	unitstop(.villager2_gid);
	end;

OnTimer11500:
	setunitdata(.villager2_gid, UDT_WALKTOXY, 163, 160);
	end;

OnTimer14200:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	end;

OnTimer16200:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 185);
	end;

OnTimer18400:
	unitstop(.villager1_gid);
	end;

OnTimer19800:
	setunitdata(.villager1_gid, UDT_WALKTOXY, 163, 174);
	end;

OnTimer20100:
	unitstop(.villager2_gid);
	end;

OnTimer21500:

prontera,163,174,0	script	VillageoisTouch#prt1	HIDDEN_WARP_NPC,1,22,{

	end;

	end;
}

prontera,160,170,0	script	VillageoisMoovingScanner	HIDDEN_WARP_NPC,{
	end;

OnInit:
	mapannounce "prontera","[Villageois] Scanner charge.",bc_map,"0x77ff77";
	initnpctimer;
	end;

OnTimer1000:
	callfunc "F_VillageoisMoovingScan", "VillageoisMoovingController";
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;
}

function	script	F_VillageoisMoovingScan	{
	.@count = getunits(BL_PC, .@players, false, "prontera", 1, 1, 350, 350);
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!attachrid(.@players[.@i]))
			continue;

		if (@VillageoisMoovingDelay > gettimetick(2)) {
			detachrid;
			continue;
		}

		dispbottom "[Villageois] Scanner actif.";
		getmapxy(.@map$, .@pc_x, .@pc_y, UNITTYPE_PC);
		if (.@map$ == "prontera" && distance(.@pc_x, .@pc_y, 163, 174) <= 5) {
			@VillageoisMoovingDelay = gettimetick(2) + 8;
			dispbottom "[Villageois] Detection proche.";
			callfunc "F_VillageoisMoovingTalk", "VillageoisMoovingController", 1;
		} else if (.@map$ == "prontera" && distance(.@pc_x, .@pc_y, 159, 160) <= 5) {
			@VillageoisMoovingDelay = gettimetick(2) + 8;
			dispbottom "[Villageois] Detection proche.";
			callfunc "F_VillageoisMoovingTalk", "VillageoisMoovingController", 2;
		}
		detachrid;
	}
	return;
}

function	script	F_VillageoisMoovingTalk	{
	.@controller$ = getarg(0);
	.@which = getarg(1);

	if (.@which == 1)
		.@gid = getvariableofnpc(.villager1_gid, .@controller$);
	else
		.@gid = getvariableofnpc(.villager2_gid, .@controller$);

	if (.@gid > 0)
		getunitdata(.@gid, UDT_MAPIDXY, .@mob_pos);
	else {
		.@mob_pos[1] = (.@which == 1) ? 163 : 159;
		.@mob_pos[2] = (.@which == 1) ? 174 : 160;
	}
	getmapxy(.@map$, .@pc_x, .@pc_y, UNITTYPE_PC);

	if (.@map$ != "prontera")
		return;

	if (gettimetick(2) < getvariableofnpc(.last_talk_tick, .@controller$))
		return;

	set getvariableofnpc(.last_talk_tick, .@controller$), gettimetick(2) + 2;
	if (.@gid > 0) {
		unitstop(.@gid);
		setunitdata(.@gid, UDT_CANMOVETICK, gettimetick(0) + 1800);
	}

	mes "[Villageois]";
	mes "Tu veux quoi ?";
	next;
	.@choice = select("Rien:Echanger");
	if (.@choice == 2) {
		callshop "VillageoisCloneTradeShop",1;
		return;
	}
	close2;

	if (.@which == 1) {
		if (distance(.@mob_pos[1], .@mob_pos[2], 163, 185) < distance(.@mob_pos[1], .@mob_pos[2], 163, 164)) {
			if (.@gid > 0)
				setunitdata(.@gid, UDT_WALKTOXY, 163, 164);
		} else {
			if (.@gid > 0)
				setunitdata(.@gid, UDT_WALKTOXY, 163, 185);
		}
	} else {
		if (distance(.@mob_pos[1], .@mob_pos[2], 155, 160) < distance(.@mob_pos[1], .@mob_pos[2], 163, 160)) {
			if (.@gid > 0)
				setunitdata(.@gid, UDT_WALKTOXY, 163, 160);
		} else {
			if (.@gid > 0)
				setunitdata(.@gid, UDT_WALKTOXY, 155, 160);
		}
	}
	return;
}
