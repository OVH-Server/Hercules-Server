//===== Hercules Script ======================================
// Team Contracts
// Fixed board of 5 team contracts.
//
// Rules:
// - Everyone can view the 5 contracts.
// - Only a party leader can accept a team contract.
// - The leader must be one of the required base classes.
// - The party must contain all required base classes.
// - Progress is shared by the party.
// - The leader validates the contract and online party members receive reward.
//============================================================

prontera,152,166,4	script	Team Contracts	4_F_VALKYRIE,{
	mes "[Team Contracts]";
	mes "Welcome to the Team Contract Board.";
	mes "These contracts require a party with specific classes.";
	next;

	switch(select("View team contracts:Validate current contract:Cancel current contract:Leave")) {
	case 1:
		callsub S_ListContracts;
		close;

	case 2:
		callsub S_ValidateContract;
		close;

	case 3:
		callsub S_CancelContract;
		close;

	case 4:
		close;
	}

S_ListContracts:
	mes "[Team Contracts]";
	mes "Available team contracts:";
	next;

	for (.@i = 1; .@i <= 5; .@i++) {
		mes "[Contract #" + .@i + "]";
		mes "Name: ^00FF00" + .Title$[.@i] + "^000000";
		mes "Required team:";
		mes "^00FF00" + .Classes$[.@i] + "^000000";
		mes "Location: ^00FF00" + .Map$[.@i] + "^000000";
		mes "Objectives:";
		mes "- " + .Need[.@i] + " " + .Mob1Name$[.@i];
		mes "- " + .Need[.@i] + " " + .Mob2Name$[.@i];
		mes "Reward: " + .RewardExp[.@i] + " Base EXP and " + .RewardZeny[.@i] + " Zeny.";
		next;
	}

	mes "[Team Contracts]";
	mes "Which contract do you want to inspect?";
	next;

	.@choice = select("Contract 1:Contract 2:Contract 3:Contract 4:Contract 5:Leave");
	if (.@choice == 6)
		return;

	callsub S_ShowContract,.@choice;
	return;

S_ShowContract:
	.@quest = getarg(0);

	mes "[Team Contracts]";
	mes "Contract #" + .@quest + ": ^00FF00" + .Title$[.@quest] + "^000000";
	mes "Required team: " + .Classes$[.@quest];
	mes "Location: " + .Map$[.@quest];
	mes "Kill " + .Need[.@quest] + " " + .Mob1Name$[.@quest] + ".";
	mes "Kill " + .Need[.@quest] + " " + .Mob2Name$[.@quest] + ".";
	next;

	if (select("Accept this team contract:Back") == 2)
		return;

	callsub S_AcceptContract,.@quest;
	return;

S_AcceptContract:
	.@quest = getarg(0);

	.@party_id = getcharid(CHAR_ID_PARTY);
	if (!.@party_id) {
		mes "[Team Contracts]";
		mes "You need to create a party before accepting a team contract.";
		close;
	}

	if (getpartyleader(.@party_id,2) != getcharid(CHAR_ID_CHAR)) {
		mes "[Team Contracts]";
		mes "Only the party leader can accept a team contract.";
		close;
	}

	.@leader_ok = 0;
	.@baseclass = BaseClass;
	.@basejob = BaseJob;
	.@eac = eaclass();

	if (.ReqSwordman[.@quest] && (.@baseclass == Job_Swordman || .@basejob == Job_Swordman || (.@eac&EAJ_BASEMASK) == EAJ_SWORDMAN)) .@leader_ok = 1;
	if (.ReqArcher[.@quest] && (.@baseclass == Job_Archer || .@basejob == Job_Archer || (.@eac&EAJ_BASEMASK) == EAJ_ARCHER)) .@leader_ok = 1;
	if (.ReqMage[.@quest] && (.@baseclass == Job_Mage || .@basejob == Job_Mage || (.@eac&EAJ_BASEMASK) == EAJ_MAGE)) .@leader_ok = 1;
	if (.ReqAcolyte[.@quest] && (.@baseclass == Job_Acolyte || .@basejob == Job_Acolyte || (.@eac&EAJ_BASEMASK) == EAJ_ACOLYTE)) .@leader_ok = 1;
	if (.ReqThief[.@quest] && (.@baseclass == Job_Thief || .@basejob == Job_Thief || (.@eac&EAJ_BASEMASK) == EAJ_THIEF)) .@leader_ok = 1;
	if (.ReqMerchant[.@quest] && (.@baseclass == Job_Merchant || .@basejob == Job_Merchant || (.@eac&EAJ_BASEMASK) == EAJ_MERCHANT)) .@leader_ok = 1;

	if (!.@leader_ok) {
		mes "[Team Contracts]";
		mes "Your class cannot accept this contract.";
		mes "Leader class detected: ^0000FF" + jobname(Class) + "^000000";
		mes "The party leader must be one of:";
		mes "^FF0000" + .Classes$[.@quest] + "^000000";
		close;
	}

	callsub S_FindPartySlot,.@party_id;
	.@slot = .FindSlot;
	if (.@slot >= 0) {
		mes "[Team Contracts]";
		mes "Your party already has an active team contract.";
		callsub S_ShowProgress,.@slot;
		close;
	}

	callsub S_CheckPartyClasses,.@party_id,.ReqSwordman[.@quest],.ReqArcher[.@quest],.ReqMage[.@quest],.ReqAcolyte[.@quest],.ReqThief[.@quest],.ReqMerchant[.@quest];
	if (!.PartyOk) {
		mes "[Team Contracts]";
		mes "Your party does not contain all required classes.";
		mes "Required: ^FF0000" + .Classes$[.@quest] + "^000000";
		mes "Detected: ^0000FF" + .Detected$ + "^000000";
		mes "Missing: ^FF0000" + .Missing$ + "^000000";
		close;
	}

	callsub S_FindFreeSlot;
	.@slot = .FindSlot;
	if (.@slot < 0) {
		mes "[Team Contracts]";
		mes "The team contract board is full.";
		mes "Please wait until another party completes or cancels a contract.";
		close;
	}

	$TC_Active[.@slot] = 1;
	$TC_PartyId[.@slot] = .@party_id;
	$TC_LeaderCharId[.@slot] = getcharid(CHAR_ID_CHAR);
	$TC_LeaderName$[.@slot] = strcharinfo(PC_NAME);
	$TC_Quest[.@slot] = .@quest;
	$TC_Title$[.@slot] = .Title$[.@quest];
	$TC_Map$[.@slot] = .Map$[.@quest];
	$TC_Mob1[.@slot] = .Mob1[.@quest];
	$TC_Mob2[.@slot] = .Mob2[.@quest];
	$TC_Mob1_Name$[.@slot] = .Mob1Name$[.@quest];
	$TC_Mob2_Name$[.@slot] = .Mob2Name$[.@quest];
	$TC_Kill1[.@slot] = 0;
	$TC_Kill2[.@slot] = 0;
	$TC_Need[.@slot] = .Need[.@quest];
	$TC_RewardExp[.@slot] = .RewardExp[.@quest];
	$TC_RewardZeny[.@slot] = .RewardZeny[.@quest];
	$TC_CompleteNotice[.@slot] = 0;

	callsub S_MessageParty,.@party_id,"[Team Contract] " + .Title$[.@quest] + " accepted by " + strcharinfo(PC_NAME) + ".";

	mes "[Team Contracts]";
	mes "Contract accepted for your party.";
	mes "Go to ^00FF00" + .Map$[.@quest] + "^000000.";
	close;

S_ValidateContract:
	.@party_id = getcharid(CHAR_ID_PARTY);
	if (!.@party_id) {
		mes "[Team Contracts]";
		mes "You are not in a party.";
		return;
	}

	callsub S_FindPartySlot,.@party_id;
	.@slot = .FindSlot;
	if (.@slot < 0) {
		mes "[Team Contracts]";
		mes "Your party has no active team contract.";
		return;
	}

	if (getpartyleader(.@party_id,2) != getcharid(CHAR_ID_CHAR)) {
		mes "[Team Contracts]";
		mes "Only the party leader can validate the team contract.";
		callsub S_ShowProgress,.@slot;
		return;
	}

	mes "[Team Contracts]";
	callsub S_ShowProgress,.@slot;
	next;

	if ($TC_Kill1[.@slot] < $TC_Need[.@slot] || $TC_Kill2[.@slot] < $TC_Need[.@slot]) {
		mes "[Team Contracts]";
		mes "The contract is not complete yet.";
		return;
	}

	.@party_id = $TC_PartyId[.@slot];
	.@reward_exp = $TC_RewardExp[.@slot];
	.@reward_zeny = $TC_RewardZeny[.@slot];
	.@title$ = $TC_Title$[.@slot];

	getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);
	.@count = getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);

	.@original_aid = getcharid(CHAR_ID_ACCOUNT);
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@accid[.@i],.@charid[.@i]))
			continue;

		if (!attachrid(.@accid[.@i]))
			continue;

		if (getcharid(CHAR_ID_CHAR) != .@charid[.@i])
			continue;

		Zeny += .@reward_zeny;
		getexp .@reward_exp,0;
		message strcharinfo(PC_NAME), "[Team Contract] " + .@title$ + " completed. Reward received.";
	}
	attachrid .@original_aid;

	callsub S_ClearSlot,.@slot;

	mes "[Team Contracts]";
	mes "Contract completed.";
	mes "Online party members received the reward.";
	return;

S_CancelContract:
	.@party_id = getcharid(CHAR_ID_PARTY);
	if (!.@party_id) {
		mes "[Team Contracts]";
		mes "You are not in a party.";
		return;
	}

	callsub S_FindPartySlot,.@party_id;
	.@slot = .FindSlot;
	if (.@slot < 0) {
		mes "[Team Contracts]";
		mes "Your party has no active team contract.";
		return;
	}

	if (getpartyleader(.@party_id,2) != getcharid(CHAR_ID_CHAR)) {
		mes "[Team Contracts]";
		mes "Only the party leader can cancel the team contract.";
		return;
	}

	callsub S_ClearSlot,.@slot;
	callsub S_MessageParty,.@party_id,"[Team Contract] Current team contract cancelled by " + strcharinfo(PC_NAME) + ".";

	mes "[Team Contracts]";
	mes "Your party contract has been cancelled.";
	return;

S_ShowProgress:
	.@slot = getarg(0);
	mes "Contract: ^00FF00" + $TC_Title$[.@slot] + "^000000";
	mes $TC_Mob1_Name$[.@slot] + ": " + $TC_Kill1[.@slot] + "/" + $TC_Need[.@slot];
	mes $TC_Mob2_Name$[.@slot] + ": " + $TC_Kill2[.@slot] + "/" + $TC_Need[.@slot];
	return;

S_FindPartySlot:
	.FindSlot = -1;
	for (.@i = 0; .@i < 20; .@i++) {
		if ($TC_Active[.@i] && $TC_PartyId[.@i] == getarg(0)) {
			.FindSlot = .@i;
			return;
		}
	}
	return;

S_FindFreeSlot:
	.FindSlot = -1;
	for (.@i = 0; .@i < 20; .@i++) {
		if (!$TC_Active[.@i]) {
			.FindSlot = .@i;
			return;
		}
	}
	return;

S_ClearSlot:
	.@slot = getarg(0);
	$TC_Active[.@slot] = 0;
	$TC_PartyId[.@slot] = 0;
	$TC_LeaderCharId[.@slot] = 0;
	$TC_LeaderName$[.@slot] = "";
	$TC_Quest[.@slot] = 0;
	$TC_Title$[.@slot] = "";
	$TC_Map$[.@slot] = "";
	$TC_Mob1[.@slot] = 0;
	$TC_Mob2[.@slot] = 0;
	$TC_Mob1_Name$[.@slot] = "";
	$TC_Mob2_Name$[.@slot] = "";
	$TC_Kill1[.@slot] = 0;
	$TC_Kill2[.@slot] = 0;
	$TC_Need[.@slot] = 0;
	$TC_RewardExp[.@slot] = 0;
	$TC_RewardZeny[.@slot] = 0;
	$TC_CompleteNotice[.@slot] = 0;
	return;

S_MessageParty:
	.@party_id = getarg(0);
	.@msg$ = getarg(1);
	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	for (.@i = 0; .@i < .@count; .@i++) {
		if (isloggedin(.@accid[.@i],.@charid[.@i]))
			message .@accid[.@i], .@msg$;
	}
	return;

S_CheckPartyClasses:
	.@party_id = getarg(0);
	.@req_swordman = getarg(1);
	.@req_archer = getarg(2);
	.@req_mage = getarg(3);
	.@req_acolyte = getarg(4);
	.@req_thief = getarg(5);
	.@req_merchant = getarg(6);

	.@has_swordman = 0;
	.@has_archer = 0;
	.@has_mage = 0;
	.@has_acolyte = 0;
	.@has_thief = 0;
	.@has_merchant = 0;

	getpartymember(.@party_id, PT_MEMBER_CHARID, .@charid);
	.@count = getpartymember(.@party_id, PT_MEMBER_ACCID, .@accid);

	.@original_aid = getcharid(CHAR_ID_ACCOUNT);
	for (.@i = 0; .@i < .@count; .@i++) {
		if (!isloggedin(.@accid[.@i],.@charid[.@i]))
			continue;

		if (!attachrid(.@accid[.@i]))
			continue;

		.@baseclass = BaseClass;
		.@basejob = BaseJob;
		.@eac = eaclass();

		if (.@baseclass == Job_Swordman || .@basejob == Job_Swordman || (.@eac&EAJ_BASEMASK) == EAJ_SWORDMAN) .@has_swordman = 1;
		if (.@baseclass == Job_Archer || .@basejob == Job_Archer || (.@eac&EAJ_BASEMASK) == EAJ_ARCHER) .@has_archer = 1;
		if (.@baseclass == Job_Mage || .@basejob == Job_Mage || (.@eac&EAJ_BASEMASK) == EAJ_MAGE) .@has_mage = 1;
		if (.@baseclass == Job_Acolyte || .@basejob == Job_Acolyte || (.@eac&EAJ_BASEMASK) == EAJ_ACOLYTE) .@has_acolyte = 1;
		if (.@baseclass == Job_Thief || .@basejob == Job_Thief || (.@eac&EAJ_BASEMASK) == EAJ_THIEF) .@has_thief = 1;
		if (.@baseclass == Job_Merchant || .@basejob == Job_Merchant || (.@eac&EAJ_BASEMASK) == EAJ_MERCHANT) .@has_merchant = 1;
	}
	attachrid .@original_aid;

	.Detected$ = "";
	if (.@has_swordman) .Detected$ = .Detected$ + "Swordman ";
	if (.@has_archer) .Detected$ = .Detected$ + "Archer ";
	if (.@has_mage) .Detected$ = .Detected$ + "Mage ";
	if (.@has_acolyte) .Detected$ = .Detected$ + "Acolyte ";
	if (.@has_thief) .Detected$ = .Detected$ + "Thief ";
	if (.@has_merchant) .Detected$ = .Detected$ + "Merchant ";
	if (.Detected$ == "") .Detected$ = "None";

	.Missing$ = "";
	if (.@req_swordman && !.@has_swordman) .Missing$ = .Missing$ + "Swordman ";
	if (.@req_archer && !.@has_archer) .Missing$ = .Missing$ + "Archer ";
	if (.@req_mage && !.@has_mage) .Missing$ = .Missing$ + "Mage ";
	if (.@req_acolyte && !.@has_acolyte) .Missing$ = .Missing$ + "Acolyte ";
	if (.@req_thief && !.@has_thief) .Missing$ = .Missing$ + "Thief ";
	if (.@req_merchant && !.@has_merchant) .Missing$ = .Missing$ + "Merchant ";

	.PartyOk = (.Missing$ == "");
	return;

OnNPCKillEvent:
	.@party_id = getcharid(CHAR_ID_PARTY);
	if (!.@party_id)
		end;

	callsub S_FindPartySlot,.@party_id;
	.@slot = .FindSlot;
	if (.@slot < 0)
		end;

	if (killedrid == $TC_Mob1[.@slot] && $TC_Kill1[.@slot] < $TC_Need[.@slot]) {
		$TC_Kill1[.@slot]++;
		callsub S_MessageParty,.@party_id,"[Team Contract] " + $TC_Mob1_Name$[.@slot] + " " + $TC_Kill1[.@slot] + "/" + $TC_Need[.@slot] + ".";
	}

	if (killedrid == $TC_Mob2[.@slot] && $TC_Kill2[.@slot] < $TC_Need[.@slot]) {
		$TC_Kill2[.@slot]++;
		callsub S_MessageParty,.@party_id,"[Team Contract] " + $TC_Mob2_Name$[.@slot] + " " + $TC_Kill2[.@slot] + "/" + $TC_Need[.@slot] + ".";
	}

	if ($TC_Kill1[.@slot] >= $TC_Need[.@slot] && $TC_Kill2[.@slot] >= $TC_Need[.@slot] && !$TC_CompleteNotice[.@slot]) {
		$TC_CompleteNotice[.@slot] = 1;
		callsub S_MessageParty,.@party_id,"[Team Contract] Objectives complete. Party leader can validate at Team Contracts.";
	}

	end;

OnInit:
	for (.@i = 0; .@i < 20; .@i++)
		callsub S_ClearSlot,.@i;

	for (.@i = 1; .@i <= 5; .@i++) {
		.ReqSwordman[.@i] = 0;
		.ReqArcher[.@i] = 0;
		.ReqMage[.@i] = 0;
		.ReqAcolyte[.@i] = 0;
		.ReqThief[.@i] = 0;
		.ReqMerchant[.@i] = 0;
	}

	.Title$[1] = "Payon Dungeon 00";
	.Classes$[1] = "Swordman + Acolyte";
	.Map$[1] = "pay_dun00";
	.Mob1[1] = 1015;
	.Mob2[1] = 1031;
	.Mob1Name$[1] = "Zombie";
	.Mob2Name$[1] = "Poporing";
	.Need[1] = 10;
	.RewardExp[1] = 10000;
	.RewardZeny[1] = 10000;
	.ReqSwordman[1] = 1;
	.ReqAcolyte[1] = 1;

	.Title$[2] = "Moscovia Field 01";
	.Classes$[2] = "Archer + Mage";
	.Map$[2] = "mosk_fild01";
	.Mob1[2] = 1881;
	.Mob2[2] = 1880;
	.Mob1Name$[2] = "Les";
	.Mob2Name$[2] = "Wood Goblin";
	.Need[2] = 10;
	.RewardExp[2] = 10000;
	.RewardZeny[2] = 10000;
	.ReqArcher[2] = 1;
	.ReqMage[2] = 1;

	.Title$[3] = "Einbroch Field 04";
	.Classes$[3] = "Merchant + Thief";
	.Map$[3] = "ein_fild04";
	.Mob1[3] = 1613;
	.Mob2[3] = 1629;
	.Mob1Name$[3] = "Metaling";
	.Mob2Name$[3] = "Mol";
	.Need[3] = 10;
	.RewardExp[3] = 10000;
	.RewardZeny[3] = 10000;
	.ReqThief[3] = 1;
	.ReqMerchant[3] = 1;

	.Title$[4] = "Payon Dungeon 01";
	.Classes$[4] = "Swordman + Acolyte + Archer";
	.Map$[4] = "pay_dun01";
	.Mob1[4] = 1016;
	.Mob2[4] = 1028;
	.Mob1Name$[4] = "Archer Skeleton";
	.Mob2Name$[4] = "Soldier Skeleton";
	.Need[4] = 10;
	.RewardExp[4] = 15000;
	.RewardZeny[4] = 15000;
	.ReqSwordman[4] = 1;
	.ReqArcher[4] = 1;
	.ReqAcolyte[4] = 1;

	.Title$[5] = "Orc Field";
	.Classes$[5] = "Swordman + Archer + Mage + Acolyte + Thief + Merchant";
	.Map$[5] = "gef_fild10";
	.Mob1[5] = 1023;
	.Mob2[5] = 1273;
	.Mob1Name$[5] = "Orc Warrior";
	.Mob2Name$[5] = "Orc Lady";
	.Need[5] = 20;
	.RewardExp[5] = 25000;
	.RewardZeny[5] = 25000;
	.ReqSwordman[5] = 1;
	.ReqArcher[5] = 1;
	.ReqMage[5] = 1;
	.ReqAcolyte[5] = 1;
	.ReqThief[5] = 1;
	.ReqMerchant[5] = 1;
	end;
}
