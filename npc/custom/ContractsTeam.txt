//===== Hercules Script ======================================

// =====================================================
// Team Contracts NPC - Fixed Version
// Everyone can view missions
// Only required classes can accept
// =====================================================

prontera,152,166,4	script	TeamContracts#001	4_F_VALKYRIE,{

	mes("[Team Contracts]");
	mes("Welcome.");
	mes("Here you can view and accept team contracts.");
	mes("Everyone can view missions, but only required classes can accept them.");
	next();

	.@choice = select(
		"Quest 1 - Payon Dungeon 00:" +
		"Quest 2 - Moscovia Field 01:" +
		"Quest 3 - Einbroch Field 04:" +
		"Quest 4 - Payon Dungeon 01:" +
		"Quest 5 - Orc Field:" +
		"Validate current contract:" +
		"Cancel current contract:" +
		"Leave"
	);

	if (.@choice == 1) {
		callsub S_ShowQuest,1,"Swordman + Acolyte","Kill 10 Zombies and 10 Poporings in Payon Dungeon 00.",1015,1031,"Zombie","Poporing","pay_dun00",10;
		close();
	}

	if (.@choice == 2) {
		callsub S_ShowQuest,2,"Archer + Mage","Kill 10 Les and 10 Wood Goblins in Moscovia Field 01.",1881,1880,"Les","Wood Goblin","mosk_fild01",10;
		close();
	}

	if (.@choice == 3) {
		callsub S_ShowQuest,3,"Merchant + Thief","Kill 10 Metalings and 10 Mols in Einbroch Field 04.",1613,1629,"Metaling","Mol","ein_fild04",10;
		close();
	}

	if (.@choice == 4) {
		callsub S_ShowQuest,4,"Swordman + Acolyte + Archer","Kill 10 Archer Skeletons and 10 Soldier Skeletons in Payon Dungeon 01.",1016,1028,"Archer Skeleton","Soldier Skeleton","pay_dun01",10;
		close();
	}

	if (.@choice == 5) {
		callsub S_ShowQuest,5,"Swordman + Archer + Mage + Acolyte + Thief + Merchant","Kill 20 Orc Warriors and 20 Orc Ladies in Gefen Field 10.",1023,1273,"Orc Warrior","Orc Lady","gef_fild10",20;
		close();
	}

	if (.@choice == 6) {
		callsub S_Validate;
		close();
	}

	if (.@choice == 7) {
		callsub S_Cancel;
		close();
	}

	close();

S_ShowQuest:
	.@qid = getarg(0);
	.@classes$ = getarg(1);
	.@desc$ = getarg(2);
	.@mob1 = getarg(3);
	.@mob2 = getarg(4);
	.@mob1$ = getarg(5);
	.@mob2$ = getarg(6);
	.@map$ = getarg(7);
	.@amount = getarg(8);

	mes("[Team Contracts]");
	mes("Contract #" + .@qid);
	mes("Required classes:");
	mes("^00FF00" + .@classes$ + "^000000");
	mes("");
	mes("Objective:");
	mes(.@desc$);
	mes("");
	mes("Map:");
	mes("^00FF00" + .@map$ + "^000000");
	mes("");
	mes("Reward:");
	mes("^00FF0010,000 Base EXP");
	mes("10,000 Zeny^000000");
	next();

	if (#TeamContract_Active) {
		mes("[Team Contracts]");
		mes("You already have an active contract.");
		close();
	}

	.@class_ok = callsub(S_CheckClassRequirement, .@qid);

	if (!.@class_ok) {
		mes("[Team Contracts]");
		mes("You can view this contract, but your class cannot accept it.");
		close();
	}

	if (select("Accept contract:Leave") == 1) {
		#TeamContract_Active = 1;
		#TeamContract_ID = .@qid;
		#TeamContract_Mob1 = .@mob1;
		#TeamContract_Mob2 = .@mob2;
		#TeamContract_Mob1_Name$ = .@mob1$;
		#TeamContract_Mob2_Name$ = .@mob2$;
		#TeamContract_Mob1_Kill = 0;
		#TeamContract_Mob2_Kill = 0;
		#TeamContract_Amount = .@amount;

		mes("[Team Contracts]");
		mes("Contract accepted.");
		mes("Go to ^00FF00" + .@map$ + "^000000.");
		mes("Kill:");
		mes("- " + .@amount + " " + .@mob1$);
		mes("- " + .@amount + " " + .@mob2$);
		close();
	}

	close();

S_Validate:
	if (!#TeamContract_Active) {
		mes("[Team Contracts]");
		mes("You do not have an active contract.");
		close();
	}

	mes("[Team Contracts]");
	mes("Current contract progress:");
	mes(#TeamContract_Mob1_Name$ + ": " + #TeamContract_Mob1_Kill + "/" + #TeamContract_Amount);
	mes(#TeamContract_Mob2_Name$ + ": " + #TeamContract_Mob2_Kill + "/" + #TeamContract_Amount);
	next();

	if (#TeamContract_Mob1_Kill < #TeamContract_Amount || #TeamContract_Mob2_Kill < #TeamContract_Amount) {
		mes("[Team Contracts]");
		mes("The contract is not complete yet.");
		close();
	}

	#TeamContract_Active = 0;
	#TeamContract_ID = 0;
	#TeamContract_Mob1 = 0;
	#TeamContract_Mob2 = 0;
	#TeamContract_Mob1_Name$ = "";
	#TeamContract_Mob2_Name$ = "";
	#TeamContract_Mob1_Kill = 0;
	#TeamContract_Mob2_Kill = 0;
	#TeamContract_Amount = 0;

	Zeny += 10000;
	getexp 10000,0;

	specialeffect2 EF_LEVEL99_3;
	skilleffect AL_BLESSING,0;

	mes("[Team Contracts]");
	mes("Contract completed!");
	mes("You received:");
	mes("^00FF0010,000 Base EXP");
	mes("10,000 Zeny^000000");
	close();

S_Cancel:
	if (!#TeamContract_Active) {
		mes("[Team Contracts]");
		mes("You do not have an active contract.");
		close();
	}

	mes("[Team Contracts]");
	mes("Cancel your current contract?");
	next();

	if (select("Cancel contract:Keep contract") == 1) {
		#TeamContract_Active = 0;
		#TeamContract_ID = 0;
		#TeamContract_Mob1 = 0;
		#TeamContract_Mob2 = 0;
		#TeamContract_Mob1_Name$ = "";
		#TeamContract_Mob2_Name$ = "";
		#TeamContract_Mob1_Kill = 0;
		#TeamContract_Mob2_Kill = 0;
		#TeamContract_Amount = 0;

		mes("[Team Contracts]");
		mes("Your contract has been cancelled.");
		close();
	}

	close();

S_CheckClassRequirement:
	.@qid = getarg(0);

	if (.@qid == 1 && (BaseJob == Job_Swordman || BaseJob == Job_Acolyte))
		return 1;

	if (.@qid == 2 && (BaseJob == Job_Archer || BaseJob == Job_Mage))
		return 1;

	if (.@qid == 3 && (BaseJob == Job_Merchant || BaseJob == Job_Thief))
		return 1;

	if (.@qid == 4 && (BaseJob == Job_Swordman || BaseJob == Job_Acolyte || BaseJob == Job_Archer))
		return 1;

	if (.@qid == 5 && (BaseJob == Job_Swordman || BaseJob == Job_Archer || BaseJob == Job_Mage || BaseJob == Job_Acolyte || BaseJob == Job_Thief || BaseJob == Job_Merchant))
		return 1;

	return 0;

OnNPCKillEvent:
	if (!#TeamContract_Active)
		end;

	if (killedrid == #TeamContract_Mob1 && #TeamContract_Mob1_Kill < #TeamContract_Amount) {
		#TeamContract_Mob1_Kill++;
		dispbottom "Team Contract: " + #TeamContract_Mob1_Name$ + " " + #TeamContract_Mob1_Kill + "/" + #TeamContract_Amount;
	}

	if (killedrid == #TeamContract_Mob2 && #TeamContract_Mob2_Kill < #TeamContract_Amount) {
		#TeamContract_Mob2_Kill++;
		dispbottom "Team Contract: " + #TeamContract_Mob2_Name$ + " " + #TeamContract_Mob2_Kill + "/" + #TeamContract_Amount;
	}

	end;
}

1108:10000,2314:10000,\
1714:10000,2330:10000,\
1610:10000,2321:10000,\

1222:10000,2337:10000,\
