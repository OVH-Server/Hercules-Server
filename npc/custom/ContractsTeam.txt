//===== Hercules Script ======================================

prontera,152,166,4	script	TeamContractsFixed#TC01	4_F_VALKYRIE,{

	mes("[Team Contracts]");
	mes("Welcome.");
	mes("Choose a team contract.");
	next();

	.@s = select("Quest 1 - Zombie/Poporing:Quest 2 - Les/Wood Goblin:Quest 3 - Metaling/Mol:Quest 4 - Skeletons:Quest 5 - Orcs:Validate:Cancel:Leave");

	if (.@s == 8) close();

	if (.@s == 7) {
		#TC_Active = 0;
		#TC_Mob1 = 0;
		#TC_Mob2 = 0;
		#TC_Kill1 = 0;
		#TC_Kill2 = 0;
		#TC_Need = 0;
		mes("[Team Contracts]");
		mes("Your contract has been cancelled.");
		close();
	}

	if (.@s == 6) {
		if (!#TC_Active) {
			mes("[Team Contracts]");
			mes("You have no active contract.");
			close();
		}

		mes("[Team Contracts]");
		mes("Progress:");
		mes(#TC_Name1$ + ": " + #TC_Kill1 + "/" + #TC_Need);
		mes(#TC_Name2$ + ": " + #TC_Kill2 + "/" + #TC_Need);
		next();

		if (#TC_Kill1 < #TC_Need || #TC_Kill2 < #TC_Need) {
			mes("[Team Contracts]");
			mes("Contract not completed yet.");
			close();
		}

		#TC_Active = 0;
		#TC_Mob1 = 0;
		#TC_Mob2 = 0;
		#TC_Kill1 = 0;
		#TC_Kill2 = 0;
		#TC_Need = 0;

		Zeny += 10000;
		getexp 10000,0;
		specialeffect2 EF_LEVEL99_3;

		mes("[Team Contracts]");
		mes("Contract completed!");
		mes("Reward: 10,000 EXP and 10,000 Zeny.");
		close();
	}

	if (#TC_Active) {
		mes("[Team Contracts]");
		mes("You already have an active contract.");
		close();
	}

	if (.@s == 1) {
		.@classes$ = "Swordman + Acolyte";
		.@mob1 = 1015; .@mob2 = 1031;
		.@name1$ = "Zombie"; .@name2$ = "Poporing";
		.@map$ = "pay_dun00";
		.@need = 10;
	}
	if (.@s == 2) {
		.@classes$ = "Archer + Mage";
		.@mob1 = 1881; .@mob2 = 1880;
		.@name1$ = "Les"; .@name2$ = "Wood Goblin";
		.@map$ = "mosk_fild01";
		.@need = 10;
	}
	if (.@s == 3) {
		.@classes$ = "Merchant + Thief";
		.@mob1 = 1613; .@mob2 = 1629;
		.@name1$ = "Metaling"; .@name2$ = "Mol";
		.@map$ = "ein_fild04";
		.@need = 10;
	}
	if (.@s == 4) {
		.@classes$ = "Swordman + Acolyte + Archer";
		.@mob1 = 1016; .@mob2 = 1028;
		.@name1$ = "Archer Skeleton"; .@name2$ = "Soldier Skeleton";
		.@map$ = "pay_dun01";
		.@need = 10;
	}
	if (.@s == 5) {
		.@classes$ = "Swordman + Archer + Mage + Acolyte + Thief + Merchant";
		.@mob1 = 1023; .@mob2 = 1273;
		.@name1$ = "Orc Warrior"; .@name2$ = "Orc Lady";
		.@map$ = "gef_fild10";
		.@need = 20;
	}

	mes("[Team Contracts]");
	mes("Required classes:");
	mes("^00FF00" + .@classes$ + "^000000");
	mes("Map: ^00FF00" + .@map$ + "^000000");
	mes("Objective:");
	mes("Kill " + .@need + " " + .@name1$ + ".");
	mes("Kill " + .@need + " " + .@name2$ + ".");
	mes("Reward: 10,000 EXP and 10,000 Zeny.");
	next();

	if (select("Accept:Leave") == 2) close();

	#TC_Active = 1;
	#TC_Mob1 = .@mob1;
	#TC_Mob2 = .@mob2;
	#TC_Name1$ = .@name1$;
	#TC_Name2$ = .@name2$;
	#TC_Kill1 = 0;
	#TC_Kill2 = 0;
	#TC_Need = .@need;

	mes("[Team Contracts]");
	mes("Contract accepted.");
	close();

OnNPCKillEvent:
	if (!#TC_Active) end;

	if (killedrid == #TC_Mob1 && #TC_Kill1 < #TC_Need) {
		#TC_Kill1++;
		dispbottom "Team Contract: " + #TC_Name1$ + " " + #TC_Kill1 + "/" + #TC_Need;
	}

	if (killedrid == #TC_Mob2 && #TC_Kill2 < #TC_Need) {
		#TC_Kill2++;
		dispbottom "Team Contract: " + #TC_Name2$ + " " + #TC_Kill2 + "/" + #TC_Need;
	}

	end;
}
