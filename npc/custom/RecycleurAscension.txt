//===== Hercules Script ======================================

prontera,159,163,4	script	Valkyrie Recycler::ValkRecycler	4_F_VALKYRIE,{

	if (.Charge >= 20) {
		mes "[Valkyrie Recycler]";
		mes "The recycler is fully charged.";
		mes "Destroy it to release the stored energy.";
		close;
	}

	mes "[Valkyrie Recycler]";
	mes "Insert an item into the recycler.";
	next;

	getinventorylist;

	.@menu$ = "";
	deletearray .@items[0], getarraysize(.@items);

	for (.@i = 0; .@i < @inventorylist_count; .@i++) {

		if (@inventorylist_amount[.@i] <= 0)
			continue;

		.@menu$ += getitemname(@inventorylist_id[.@i]) + ":";
		.@items[getarraysize(.@items)] = @inventorylist_id[.@i];
	}

	if (.@menu$ == "") {
		mes "You have no items.";
		close;
	}

	.@sel = select(.@menu$);
	.@item = .@items[.@sel-1];

	if (#RecycleCount[.@item] >= 10
		&& #RecycleDay[.@item] == gettime(DT_DAYOFYEAR)) {
		mes "This item has already been recycled 10 times today.";
		close;
	}

	if (#RecycleDay[.@item] != gettime(DT_DAYOFYEAR)) {
		#RecycleDay[.@item] = gettime(DT_DAYOFYEAR);
		#RecycleCount[.@item] = 0;
	}

	if (countitem(.@item) < 1)
		close;

	if (getiteminfo(.@item, ITEMINFO_TYPE) == IT_CARD)
		.@value = 10;
	else if (@inventorylist_equip[.@sel-1])
		.@value = 5;
	else if (
		getiteminfo(.@item, ITEMINFO_TYPE) == IT_HEALING ||
		getiteminfo(.@item, ITEMINFO_TYPE) == IT_USABLE
	)
		.@value = 2;
	else
		.@value = 1;

	if (.Charge + .@value > 20) {
		mes "Not enough remaining capacity.";
		close;
	}

	delitem .@item,1;

	#RecycleCount[.@item]++;

	.Charge += .@value;

	if (.Charge >= 20) {
		.Charge = 20;
		.Full = 1;

		specialeffect EF_LEVEL99;

		disablenpc "Valkyrie Recycler";

		monster "prontera",150,180,"Recycler Core",1905,1,"ValkRecycler::OnCoreDead";
	}

	mes "[Valkyrie Recycler]";
	mes "Current charge: "+.Charge+"/20";
	close;

OnInit:
	.Charge = 0;
	.Full = 0;
	end;

OnCoreDead:

	.Charge = 0;
	.Full = 0;

	enablenpc "Valkyrie Recycler";

	end;
}
