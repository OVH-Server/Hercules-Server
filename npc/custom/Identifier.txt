//----Identifier sans Magnifier----

prontera,145,168,4	script	Identifier	4_M_KID2,{
	getinventorylist;
	set .@identified, 0;

	for (set .@i, 0; .@i < @inventorylist_count; set .@i, .@i + 1) {
		if (@inventorylist_identify[.@i] == 1) continue;

		delitem2 @inventorylist_id[.@i],1,0,0,0,0,0,0,0;
		getitem @inventorylist_id[.@i],1;

		set .@identified, .@identified + 1;
	}

	mes "[Identifier]";

	if (.@identified > 0) {
		mes "J'ai identifié " + .@identified + " objet(s).";
	} else {
		mes "Tu n'as aucun objet à identifier.";
	}

	close;
}
