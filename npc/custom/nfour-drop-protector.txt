function	script	item_type_id_to_item_type_name	{
    .@item_type = getarg(0);

    switch (.@item_type) {
        case 0: 
            return ("HEALING");
        case 1:
            return ("UNKNOWN");
        case 2:
            return ("USABLE");
        case 3:
            return ("ETC");
        case 4:
            return ("WEAPON");
        case 5:
            return ("ARMOR");
        case 6:
            return ("CARD");
        case 7:
            return ("PETEGG");
        case 8:
            return ("PETARMOR8");
        case 9:
            return ("UNKNOWN29");
        case 10:
            return ("AMMO");
        case 11:
            return ("DELAYCONSUME11");
        case 12:
            return ("SELECTPACKAGE12");
        case 13:
            return ("CASH");
        end;
    }
    return ("INVALID");

}
