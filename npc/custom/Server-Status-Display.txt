//------Server Status Display------
//--NPC displaying online players and the server status--


-  script ServerStatusDisplay {
    mes "Server Status:";
    mes "Online Players: " + getusers(1);
    mes "Server Time: " + gettimestr("%Y-%m-%d %H:%M:%S", 4);
    close;
}

// NPC Example
prontera,164,162,4	script	Server Status	151,{
    callfunc "ServerStatusDisplay";
    close;
}
