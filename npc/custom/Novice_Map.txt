	Tuto1_RockNode2 = 0;
	Tuto1_RockNode3 = 0;

	end;

S_MoveToSagard:
	.@pc_gid = getcharid(CHAR_ID_ACCOUNT);
	.@timeout = getarg(2);
	.@elapsed = 0;

	unitwalk(.@pc_gid, getarg(0), getarg(1));
	sleep2 75;
	.@elapsed += 75;

	while (.@elapsed < .@timeout) {
		getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
		.@dx = getarg(0) - .@x;
		.@dy = getarg(1) - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@dx <= 1 && .@dy <= 1)
			return;

		if (unitiswalking(.@pc_gid) != 1 && .@elapsed > 250)
			unitwalk(.@pc_gid, getarg(0), getarg(1));

		sleep2 75;
		.@elapsed += 75;
	}

	return;
}

// DARK APPARITION PORTALS - HAYATA DEFENSE
// Hayata se buff toutes les minutes, puis lance des bolts sur les Ygnizem Clone.
// Ygnizem Clone est un vrai clone ennemi, attire vers Hayata, avec l'impact Lightning force au bon timing.
// Un seul Ygnizem Clone apparait toutes les 20 secondes depuis le Portal 1 et reste force sur Hayata.
-	script	CastleProfessorClone#new11	-1,{
	end;

OnInit:
	killmonster "new_1-1", "CastleProfessorClone#new11::OnCloneDead";
	killmonster "new_1-1", "CastleProfessorClone#new11::OnYgnizemDead";

	.CloneSlot = 7;
	.StartX = 130;
	.StartY = 112;
	.CloneMobID = 1905;
	.CombatRange = 20;
	.MaxYgnizem = 1;
	.Tick = 0;
	.NextSpawn = 0;
	.CastingBolt = 0;
	setarray .SpawnX[0], 124;
	setarray .SpawnY[0], 110;

	.CloneGID = monster("new_1-1", .StartX, .StartY, "Hayata", .CloneMobID, 1, "CastleProfessorClone#new11::OnCloneDead");

	if ($CloneStyle_Saved[.CloneSlot])
		callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	else
		setunitdata(.CloneGID, UDT_CLASS, Job_Mage);

	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK | MD_CANATTACK | MD_BOSS);
	setunitdata(.CloneGID, UDT_AI, AI_NONE);
	setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	setunitdata(.CloneGID, UDT_ATKRANGE, 9);
	setunitdata(.CloneGID, UDT_MAXHP, 2000000);
	setunitdata(.CloneGID, UDT_HP, 2000000);
	setunitdata(.CloneGID, UDT_DEX, 90);
	setunitdata(.CloneGID, UDT_MATKMIN, 5000);
	setunitdata(.CloneGID, UDT_MATKMAX, 6500);
	setunitdata(.CloneGID, UDT_MAXSP, 2000);
	setunitdata(.CloneGID, UDT_SP, 2000);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer1000:
	.Tick++;

	if (.Tick == 1)
		callsub S_CastEnergyCoat;
	else if (.Tick == 4)
		callsub S_CastDoubleCasting;
	else if (.Tick == 7)
		callsub S_CastMemorize;

	if (.Tick % 20 == 0)
		callsub S_SpawnYgnizem;

	callsub S_RetargetYgnizem;

	if (.Tick >= 17 && .Tick % 3 == 0)
		callsub S_AttackNearbyYgnizem;

	if (.Tick >= 60)
		.Tick = 0;

	setnpctimer 0;
	end;

S_CastEnergyCoat:
	if (.CloneGID <= 0)
		return;
	setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	unitskilluseid(.CloneGID, MG_ENERGYCOAT, 1, .CloneGID);
	sc_start(SC_ENERGYCOAT, 600000, 1, 10000, SCFLAG_NOAVOID|SCFLAG_FIXEDTICK, .CloneGID);
	return;

S_CastDoubleCasting:
	if (.CloneGID <= 0)
		return;
	setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	unitskilluseid(.CloneGID, PF_DOUBLECASTING, 5, .CloneGID);
	sc_start(SC_DOUBLECASTING, 600000, 5, 10000, SCFLAG_NOAVOID|SCFLAG_FIXEDTICK, .CloneGID);
	return;

S_CastMemorize:
	if (.CloneGID <= 0)
		return;
	setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	unitskilluseid(.CloneGID, PF_MEMORIZE, 1, .CloneGID);
	sc_start(SC_MEMORIZE, 600000, 1, 10000, SCFLAG_NOAVOID|SCFLAG_FIXEDTICK, .CloneGID);
	return;

S_SpawnYgnizem:
	if (.CloneGID <= 0)
		return;
	if (mobcount("new_1-1", "CastleProfessorClone#new11::OnYgnizemDead") >= .MaxYgnizem)
		return;

	.@slot = -1;
	for (.@i = 0; .@i < .MaxYgnizem; .@i++) {
		if (.YgnizemGID[.@i] <= 0 || getunitdata(.YgnizemGID[.@i], UDT_HP) <= 0) {
			.@slot = .@i;
			break;
		}
	}
	if (.@slot < 0)
		return;

	for (.@tries = 0; .@tries < getarraysize(.SpawnX); .@tries++) {
		.@spawn = (.NextSpawn + .@tries) % getarraysize(.SpawnX);
		.@gid = monster("new_1-1", .SpawnX[.@spawn], .SpawnY[.@spawn], "Ygnizem Clone", 1652, 1, "CastleProfessorClone#new11::OnYgnizemDead");
		if (.@gid <= 0)
			continue;

		setunitdata(.@gid, UDT_MODE, MD_CANMOVE | MD_CANATTACK | MD_NOKNOCKBACK);
		setunitdata(.@gid, UDT_AI, AI_ATTACK);
		setunitdata(.@gid, UDT_SPEED, 160);
		setunitdata(.@gid, UDT_ATKRANGE, 1);
		setunitdata(.@gid, UDT_MAXHP, 2000000);
		setunitdata(.@gid, UDT_HP, 2000000);
		unitattack(.@gid, .CloneGID);
		.YgnizemGID[.@slot] = .@gid;
		.NextSpawn = (.@spawn + 1) % getarraysize(.SpawnX);
		return;
	}
	return;

S_RetargetYgnizem:
	if (.CloneGID <= 0)
		return;

	for (.@i = 0; .@i < .MaxYgnizem; .@i++) {
		.@gid = .YgnizemGID[.@i];
		if (.@gid <= 0)
			continue;
		if (getunitdata(.@gid, UDT_HP) <= 0) {
			.YgnizemGID[.@i] = 0;
			continue;
		}
		unitattack(.@gid, .CloneGID);
	}
	return;

S_AttackNearbyYgnizem:
	if (.CloneGID <= 0)
		return;
	if (.CastingBolt)
		return;
	getunitdata(.CloneGID, UDT_MAPIDXY, .@hayata);

	for (.@i = 0; .@i < .MaxYgnizem; .@i++) {
		.@gid = .YgnizemGID[.@i];
		if (.@gid <= 0)
			continue;

		.@hp = getunitdata(.@gid, UDT_HP);
		if (.@hp <= 0) {
			.YgnizemGID[.@i] = 0;
			continue;
		}

		getunitdata(.@gid, UDT_MAPIDXY, .@mob);
		.@dx = .@hayata[1] - .@mob[1];
		.@dy = .@hayata[2] - .@mob[2];
		if ((.@dx * .@dx) + (.@dy * .@dy) > (.CombatRange * .CombatRange))
			continue;

		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
		switch(rand(2)) {
		case 0:
			.@skill = MG_LIGHTNINGBOLT;
			.@effect = EF_LIGHTBOLT;
			break;
		default:
			.@skill = MG_COLDBOLT;
			.@effect = EF_COLDHIT;
			break;
		}
		.CastingBolt = 1;
		setunitdata(.@gid, UDT_HP, 2000000);
		unitattack(.CloneGID, .@gid);
		areamobuseskill("new_1-1", .StartX, .StartY, 0, .CloneMobID, .@skill, 10, 1800, 0, e_no1, 1);

		sleep 1800;
		.@hp = getunitdata(.@gid, UDT_HP);
		if (.@hp <= 0) {
			.YgnizemGID[.@i] = 0;
			.CastingBolt = 0;
			return;
		}

		setunitdata(.@gid, UDT_HP, 2000000);
		specialeffect(.@effect, AREA, .@gid);
		sleep 700;
		unitkill(.@gid);
		.YgnizemGID[.@i] = 0;
		.CastingBolt = 0;
		return;
	}
	return;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CastleProfessorClone#new11::OnInit";
	end;

OnYgnizemDead:
	.YgnizemGID[0] = 0;
	.CastingBolt = 0;
	end;
}

function	script	F_Tuto1_HarvestResource	{
	.@x = getarg(0);
	.@y = getarg(1);
	.@type = getarg(2);
	.@node = getarg(3);

	getmapxy .@map$, .@pc_x, .@pc_y, UNITTYPE_PC;
	if (.@map$ != "new_1-3" || distance(.@pc_x,.@pc_y,.@x,.@y) > 4)
		end;

	mes "[^0000FFTuto Resource^000000]";

	if (Tuto1_State < 20) {
		mes "You do not need this resource yet.";
		close;
	}

	if (Tuto1_State > 20) {
		mes "You already gathered the resources needed for this tutorial task.";
		close;
	}

	if (.@type == 1) {
		.@var$ = "Tuto1_PlantNode" + .@node;
		.@name$ = "plant";
	}
	else if (.@type == 2) {
		.@var$ = "Tuto1_RockNode" + .@node;
		.@name$ = "rock";
	}
	else {
		.@var$ = "Tuto1_ArchaeoNode" + .@node;
		.@name$ = "archaeological secret";
	}

	if (getd(.@var$)) {
		mes "You already gathered this " + .@name$ + ".";
		close;
	}

	if (.Busy)
		end;

	.Busy = 1;
	specialeffect EF_SONIC_CLAW965;
	setd .@var$, 1;

	if (.@type == 1) {
		Tuto1_Plant++;
		message strcharinfo(PC_NAME), "Tutorial Quest: plants gathered " + Tuto1_Plant + "/3.";
	}
	else if (.@type == 2) {
		Tuto1_Rock++;
		message strcharinfo(PC_NAME), "Tutorial Quest: rocks mined " + Tuto1_Rock + "/3.";
	}
	else {
		Tuto1_Archaeo++;
		message strcharinfo(PC_NAME), "Tutorial Quest: archaeological secret discovered " + Tuto1_Archaeo + "/1.";
	}

	if (Tuto1_Plant >= 3 && Tuto1_Rock >= 3 && Tuto1_Archaeo >= 1 && !Tuto1_Resource_Done) {
		Tuto1_Resource_Done = 1;
		message strcharinfo(PC_NAME), "Tutorial Quest: ressources terminees. Allez voir Olgi en 95 65 pour creer les pierres qui contiendront l'energie du pilier.";
	}

	mes "You gathered the " + .@name$ + ".";
	.Busy = 0;
	close;
}
