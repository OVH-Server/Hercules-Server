		getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
		.@dx = getarg(0) - .@x;
		.@dy = getarg(1) - .@y;
		if (.@dx < 0)
			.@dx = -.@dx;
		if (.@dy < 0)
			.@dy = -.@dy;
		if (.@dx <= 1 && .@dy <= 1)
			return;

		if (unitiswalking(.@pc_gid) != 1 && .@elapsed > 250)
			unitwalk(.@pc_gid, getarg(0), getarg(1));

		sleep2 75;
		.@elapsed += 75;
	}

	return;

OnInit:
	.Milon_DialogOpen = 0;
	initnpctimer;
	end;

OnTimer30000:
	if (!.Milon_DialogOpen)
		npctalk "Adventurer! Do not stay here! Quick! Come to me!";
	initnpctimer;
	end;

OnPCLogoutEvent:
	if ($Novice_Tuto_RiderBusy == getcharid(CHAR_ID_ACCOUNT)) {
		sc_end(SC_ALL_RIDING);
		if (countitem(12622) > 0)
			delitem 12622,1;
		setpcblock(PCBLOCK_ATTACK|PCBLOCK_SKILL|PCBLOCK_USEITEM|PCBLOCK_SITSTAND, false);
		$Novice_Tuto_RiderBusy = 0;
	}
	end;
}
