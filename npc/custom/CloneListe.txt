//===== Hercules Script ======================================
//= Clone Liste
//= Clones places depuis les styles enregistres in game.
//============================================================

-	script	CloneListeSlot1	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot1::OnCloneDead";

	.CloneSlot = 1;
	.StartX = 172;
	.StartY = 160;
	.EndX = 172;
	.EndY = 145;

	.CloneGID = monster("prontera", .StartX, .StartY, "Villagers Cesar", 1002, 1, "CloneListeSlot1::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_CANMOVE|MD_NOKNOCKBACK);
	setunitdata(.CloneGID, UDT_AI, AI_NONE);
	setunitdata(.CloneGID, UDT_LOOKDIR, 0);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .EndX, .EndY);
	end;

OnTimer9500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .StartX, .StartY);
	end;

OnTimer18500:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot1::OnInit";
	end;
}

prontera,174,145,0	script	CloneEndTalkZones	FAKE_NPC,6,5,{
OnTouch:
	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);

	.@cesar_dx = .@x - 172;
	if (.@cesar_dx < 0)
		.@cesar_dx = -.@cesar_dx;
	.@cesar_dy = .@y - 145;
	if (.@cesar_dy < 0)
		.@cesar_dy = -.@cesar_dy;

	.@moka_dx = .@x - 175;
	if (.@moka_dx < 0)
		.@moka_dx = -.@moka_dx;
	.@moka_dy = .@y - 145;
	if (.@moka_dy < 0)
		.@moka_dy = -.@moka_dy;

	if (.@cesar_dx <= 5 && .@cesar_dy <= 5 && (.@cesar_dx + .@cesar_dy) <= (.@moka_dx + .@moka_dy)) {
		mes "TEST Cesar";
		close;
	}
	if (.@moka_dx <= 5 && .@moka_dy <= 5) {
		mes "TEST Moka";
		close;
	}
	end;
}
