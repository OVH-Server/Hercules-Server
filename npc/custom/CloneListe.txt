//===== Hercules Script ======================================
//= Clone Liste
//= Clones places depuis les styles enregistres in game.
//============================================================

-	script	CloneListeSlot1	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot1::OnCloneDead";

	.CloneSlot = 1;
	.StartX = 172;
	.StartY = 160;
	.EndX = 172;
	.EndY = 152;

	.CloneGID = monster("prontera", .StartX, .StartY, "Villagers Cesar", 1002, 1, "CloneListeSlot1::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK);
	setunitdata(.CloneGID, UDT_LOOKDIR, 0);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .EndX, .EndY);
	end;

OnTimer4500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .StartX, .StartY);
	end;

OnTimer8500:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot1::OnInit";
	end;
}

-	script	CloneListeSlot2	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot2::OnCloneDead";

	.CloneSlot = 2;
	setarray .PathX[0], 175, 156, 156, 162, 162;
	setarray .PathY[0], 160, 160, 179, 185, 182;
	.PathCount = 5;
	.PathIndex = 0;
	.PathDir = 1;

	.CloneGID = monster("prontera", .PathX[0], .PathY[0], "Merchant Moka", 1002, 1, "CloneListeSlot2::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	if ($CloneStyle_SaveVersion[.CloneSlot] < 3)
		setunitdata(.CloneGID, UDT_OPTIONS, getunitdata(.CloneGID, UDT_OPTIONS) | 0x00000008);
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK);
	setunitdata(.CloneGID, UDT_LOOKDIR, 0);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer100:
	if (.CloneGID <= 0)
		end;
	if (!unitiswalking(.CloneGID)) {
		.PathIndex += .PathDir;
		if (.PathIndex >= .PathCount) {
			.PathIndex = .PathCount - 2;
			.PathDir = -1;
		} else if (.PathIndex < 0) {
			.PathIndex = 1;
			.PathDir = 1;
		}
		unitwalk(.CloneGID, .PathX[.PathIndex], .PathY[.PathIndex]);
	}
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot2::OnInit";
	end;
}
