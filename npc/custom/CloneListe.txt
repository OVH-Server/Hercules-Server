//===== Hercules Script ======================================
//= Clone Liste
//= Clones places depuis les styles enregistres in game.
//=
//= Principe pour reproduire ce systeme:
//= 1) Chaque clone est un mob cree avec monster(), puis habille avec
//=    F_ApplyCloneStyle(<GID>, <slot sauvegarde>).
//= 2) Le deplacement reste volontairement simple: deux points seulement,
//=    StartX/StartY et EndX/EndY, avec des timers fixes.
//= 3) Ne pas utiliser MD_CANMOVE ici: sur ces mobs, cela peut declencher
//=    du random walk. Le mode stable est MD_NOKNOCKBACK + AI_NONE.
//= 4) Lorsqu'une interaction doit forcer le clone a venir au point final,
//=    appeler OnInteractStart. Cela stoppe le timer et envoie le clone vers
//=    EndX/EndY.
//= 5) Apres fermeture de la fenetre de dialogue, appeler OnInteractEnd pour
//=    relancer la boucle normale.
//= 6) Pour ajouter un clone: dupliquer un bloc CloneListeSlotX, changer
//=    CloneSlot, nom, StartX/StartY, EndX/EndY, puis ajouter la logique dans
//=    CloneVillageTalkZones si une interaction de proximite est souhaitee.
//============================================================

-	script	CloneListeSlot1	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot1::OnCloneDead";

	.CloneSlot = 1;
	.StartX = 172;
	.StartY = 160;
	.EndX = 172;
	.EndY = 145;

	.CloneGID = monster("prontera", .StartX, .StartY, "Villagers Cesar", 1905, 1, "CloneListeSlot1::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK);
	setunitdata(.CloneGID, UDT_AI, AI_NONE);
	setunitdata(.CloneGID, UDT_LOOKDIR, 0);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .EndX, .EndY);
	end;

OnTimer9500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .StartX, .StartY);
	end;

OnTimer18500:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnInteractStart:
	stopnpctimer;
	setunitdata(.CloneGID, UDT_WALKTOXY, .EndX, .EndY);
	end;

OnInteractEnd:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot1::OnInit";
	end;
}

prontera,170,153,4	script	Mamoko#vendor	HIDDEN_NPC,{
	callshop "CloneMamokoVendorShop",1;
	npcshopattach "CloneMamokoVendorShop";
	end;

OnBuyItem:
	for (.@i = 0; .@i < getarraysize(@bought_nameid); .@i++) {
		if (@bought_quantity[.@i] <= 0) {
			mes "[Mamoko]";
			mes "Transaction invalide.";
			close;
		}

		if (@bought_nameid[.@i] == 1405) {
			.@stock = .SpearStock;
			.@price_each = 10000;
		}
		else if (@bought_nameid[.@i] == 2285) {
			.@stock = .Item2285Stock;
			.@price_each = 150999;
		}
		else if (@bought_nameid[.@i] == 4140) {
			.@stock = .Card4140Stock;
			.@price_each = 499999;
		}
		else {
			mes "[Mamoko]";
			mes "Transaction invalide.";
			close;
		}

		.@amount = @bought_quantity[.@i];
		if (.@stock <= 0) {
			mes "[Mamoko]";
			mes "Je n'ai plus cet item a vendre.";
			close;
		}
		if (.@amount > .@stock)
			.@amount = .@stock;

		.@price = .@amount * .@price_each;
		if (Zeny < .@price) {
			mes "[Mamoko]";
			mes "Tu n'as pas assez de Zeny.";
			close;
		}
		if (!checkweight(@bought_nameid[.@i],.@amount)) {
			mes "[Mamoko]";
			mes "Ton inventaire est trop lourd.";
			close;
		}

		Zeny -= .@price;
		getitem @bought_nameid[.@i],.@amount;
		if (@bought_nameid[.@i] == 1405)
			.SpearStock -= .@amount;
		else if (@bought_nameid[.@i] == 2285)
			.Item2285Stock -= .@amount;
		else
			.Card4140Stock -= .@amount;
	}
	deletearray @bought_quantity, getarraysize(@bought_quantity);
	deletearray @bought_nameid, getarraysize(@bought_nameid);
	callsub S_UpdateShop;
	mes "[Mamoko]";
	mes "Merci pour ton achat.";
	close;

OnInit:
	.SpearStock = 2;
	.Item2285Stock = 1;
	.Card4140Stock = 2;
	callsub S_UpdateShop;
	end;

S_UpdateShop:
	npcshopdelitem "CloneMamokoVendorShop",1405,2285,4140;
	if (.SpearStock >= 2)
		npcshopadditem "CloneMamokoVendorShop",1405,10000,1405,10000;
	else if (.SpearStock == 1)
		npcshopadditem "CloneMamokoVendorShop",1405,10000;
	if (.Item2285Stock > 0)
		npcshopadditem "CloneMamokoVendorShop",2285,150999;
	if (.Card4140Stock > 0)
		npcshopadditem "CloneMamokoVendorShop",4140,499999;
	return;
}

-	script	CloneListeSlot3	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot3::OnCloneDead";
	killmonster "prontera","CloneListeSlot3::OnDummyDead";

	.CloneSlot = 3;
	.CloneX = 174;
	.CloneY = 134;
	.DummyX = 179;
	.DummyY = 134;

	callsub S_SpawnDummy;

	.CloneGID = monster("prontera", .CloneX, .CloneY, "Magician Otto", 1905, 1, "CloneListeSlot3::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK | MD_CANATTACK);
	setunitdata(.CloneGID, UDT_AI, AI_NONE);
	setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	setunitdata(.CloneGID, UDT_ATKRANGE, 6);
	setunitdata(.CloneGID, UDT_MATKMIN, 500);
	setunitdata(.CloneGID, UDT_MATKMAX, 800);
	setunitdata(.CloneGID, UDT_SP, 2000);
	setunitdata(.CloneGID, UDT_MAXSP, 2000);
	unitattack(.CloneGID, .DummyGID);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer2500:
	if (.CloneGID > 0 && .DummyGID > 0) {
		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
		setunitdata(.DummyGID, UDT_HP, 2000000);
		unitattack(.CloneGID, .DummyGID);
		unitskilluseid(.CloneGID, MG_LIGHTNINGBOLT, 3, .DummyGID);
	}
	end;

OnTimer4000:
	if (.CloneGID > 0 && .DummyGID > 0) {
		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
		setunitdata(.DummyGID, UDT_HP, 2000000);
		specialeffect(EF_LIGHTBOLT, AREA, .DummyGID);
	}
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnInteractStart:
	stopnpctimer;
	if (.CloneGID > 0) {
		unitstop(.CloneGID);
		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	}
	end;

OnInteractEnd:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot3::OnInit";
	end;

OnDummyDead:
	callsub S_SpawnDummy;
	end;

S_SpawnDummy:
	.DummyGID = monster("prontera", .DummyX, .DummyY, "Training Post", 1915, 1, "CloneListeSlot3::OnDummyDead");
	setunitdata(.DummyGID, UDT_MAXHP, 2000000);
	setunitdata(.DummyGID, UDT_HP, 2000000);
	setunitdata(.DummyGID, UDT_MODE, MD_NOKNOCKBACK | MD_BOSS);
	setunitdata(.DummyGID, UDT_AI, AI_NONE);
	return;
}

	.CloneY = 141;
	.TargetAID = 0;
	.TargetX = 0;

	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;

	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK);

	setunitdata(.CloneGID, UDT_MATKMIN, 500);
	setunitdata(.CloneGID, UDT_MATKMAX, 800);
	setunitdata(.CloneGID, UDT_SP, 2000);
	setunitdata(.CloneGID, UDT_MAXSP, 2000);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer2500:
	if (.CloneGID > 0) {
		.@adam_gid = getvariableofnpc(.CloneGID, "CloneListeSlot4");
		if (.@adam_gid > 0) {
			setunitdata(.CloneGID, UDT_LOOKDIR, 2);
			unitskilluseid(.CloneGID, AL_HEAL, 10, .@adam_gid);
			specialeffect(EF_HEAL2, AREA, .@adam_gid);
		}
	}
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnInteractStart:
	stopnpctimer;
	if (.CloneGID > 0 && .TargetAID > 0) {
		if (.TargetX > .CloneX)
			setunitdata(.CloneGID, UDT_LOOKDIR, 6);
		else if (.TargetX < .CloneX)
			setunitdata(.CloneGID, UDT_LOOKDIR, 2);
		else if (.TargetY > .CloneY)
			setunitdata(.CloneGID, UDT_LOOKDIR, 0);
		else
			setunitdata(.CloneGID, UDT_LOOKDIR, 4);
		unitskilluseid(.CloneGID, AL_HEAL, 10, .TargetAID);
		specialeffect(EF_HEAL2, AREA, .TargetAID);
	}
	end;

OnInteractEnd:
	.TargetAID = 0;
	.TargetX = 0;
	.TargetY = 0;
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot5::OnInit";
	end;
}

prontera,172,143,0	script	CloneVillageTalkZones	FAKE_NPC,7,7,{
OnTouch:
	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);
	.@best = 0;
	.@best_dist = 999;

	.@cesar_dx = .@x - 172;
	if (.@cesar_dx < 0)
		.@cesar_dx = -.@cesar_dx;
	.@cesar_dy = .@y - 145;
	if (.@cesar_dy < 0)
		.@cesar_dy = -.@cesar_dy;

	.@moka_dx = .@x - 175;
	if (.@moka_dx < 0)
		.@moka_dx = -.@moka_dx;
	.@moka_dy = .@y - 145;
	if (.@moka_dy < 0)
		.@moka_dy = -.@moka_dy;

	.@adam_dx = .@x - 170;
	if (.@adam_dx < 0)
		.@adam_dx = -.@adam_dx;
	.@adam_dy = .@y - 141;
	if (.@adam_dy < 0)
		.@adam_dy = -.@adam_dy;

	.@risa_dx = .@x - 175;
	if (.@risa_dx < 0)
		.@risa_dx = -.@risa_dx;
	.@risa_dy = .@y - 141;
	if (.@risa_dy < 0)
		.@risa_dy = -.@risa_dy;

	if (.@cesar_dx <= 5 && .@cesar_dy <= 5) {
		.@best = 1;
		.@best_dist = .@cesar_dx + .@cesar_dy;
	}
	if (.@moka_dx <= 5 && .@moka_dy <= 5 && (.@moka_dx + .@moka_dy) < .@best_dist) {
		.@best = 2;
		.@best_dist = .@moka_dx + .@moka_dy;
	}
	if (.@adam_dx <= 3 && .@adam_dy <= 3 && (.@adam_dx + .@adam_dy) < .@best_dist) {
		.@best = 4;
		.@best_dist = .@adam_dx + .@adam_dy;
	}
	if (.@risa_dx <= 3 && .@risa_dy <= 3 && (.@risa_dx + .@risa_dy) < .@best_dist) {
		.@best = 5;
		.@best_dist = .@risa_dx + .@risa_dy;
	}

	if (.@best == 1) {
		donpcevent "CloneListeSlot1::OnInteractStart";
		mes "TEST Cesar";
		close2;
		donpcevent "CloneListeSlot1::OnInteractEnd";
		end;
	}
	if (.@best == 2) {
		donpcevent "CloneListeSlot2::OnInteractStart";
		mes "TEST Moka";
		close2;
		donpcevent "CloneListeSlot2::OnInteractEnd";
		end;
	}
	if (.@best == 4) {
		donpcevent "CloneListeSlot4::OnInteractStart";
		mes "Test ADAM";
		sleep2 1600;
		donpcevent "CloneListeSlot4::OnInteractCast";
		close2;
		donpcevent "CloneListeSlot4::OnInteractEnd";
		end;
	}
	if (.@best == 5) {
		set getvariableofnpc(.TargetAID, "CloneListeSlot5"), getcharid(CHAR_ID_ACCOUNT);
		set getvariableofnpc(.TargetX, "CloneListeSlot5"), .@x;
		set getvariableofnpc(.TargetY, "CloneListeSlot5"), .@y;
		donpcevent "CloneListeSlot5::OnInteractStart";
		mes "Test Risa";
		close2;
		donpcevent "CloneListeSlot5::OnInteractEnd";
		end;
	}
	end;
}
