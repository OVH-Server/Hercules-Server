//===== Hercules Script ======================================
//= Clone Liste
//= Clones places depuis les styles enregistres in game.
//=
//= Principe pour reproduire ce systeme:
//= 1) Chaque clone est un mob cree avec monster(), puis habille avec
//=    F_ApplyCloneStyle(<GID>, <slot sauvegarde>).
//= 2) Le deplacement reste volontairement simple: deux points seulement,
//=    StartX/StartY et EndX/EndY, avec des timers fixes.
//= 3) Ne pas utiliser MD_CANMOVE ici: sur ces mobs, cela peut declencher
//=    du random walk. Le mode stable est MD_NOKNOCKBACK + AI_NONE.
//= 4) Lorsqu'une interaction doit forcer le clone a venir au point final,
//=    appeler OnInteractStart. Cela stoppe le timer et envoie le clone vers
//=    EndX/EndY.
//= 5) Apres fermeture de la fenetre de dialogue, appeler OnInteractEnd pour
//=    relancer la boucle normale.
//= 6) Pour ajouter un clone: dupliquer un bloc CloneListeSlotX, changer
//=    CloneSlot, nom, StartX/StartY, EndX/EndY, puis ajouter la logique dans
//=    CloneEndTalkZones si une interaction de proximite est souhaitee.
//============================================================

-	script	CloneListeSlot1	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot1::OnCloneDead";

	.CloneSlot = 1;
	.StartX = 172;
	.StartY = 160;
	.EndX = 172;
	.EndY = 145;

	.CloneGID = monster("prontera", .StartX, .StartY, "Villagers Cesar", 1905, 1, "CloneListeSlot1::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK);
	setunitdata(.CloneGID, UDT_AI, AI_NONE);
	setunitdata(.CloneGID, UDT_LOOKDIR, 0);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .EndX, .EndY);
	end;

OnTimer9500:
	setunitdata(.CloneGID, UDT_WALKTOXY, .StartX, .StartY);
	end;

OnTimer18500:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnInteractStart:
	stopnpctimer;
	setunitdata(.CloneGID, UDT_WALKTOXY, .EndX, .EndY);
	end;

OnInteractEnd:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot1::OnInit";
	end;
}

-	script	CloneListeSlot3	-1,{
	end;

OnInit:
	killmonster "prontera","CloneListeSlot3::OnCloneDead";
	killmonster "prontera","CloneListeSlot3::OnDummyDead";

	.CloneSlot = 3;
	.CloneX = 174;
	.CloneY = 134;
	.DummyX = 179;
	.DummyY = 134;

	callsub S_SpawnDummy;

	.CloneGID = monster("prontera", .CloneX, .CloneY, "Magician Otto", 1905, 1, "CloneListeSlot3::OnCloneDead");
	callfunc "F_ApplyCloneStyle", .CloneGID, .CloneSlot;
	setunitdata(.CloneGID, UDT_SPEED, 120);
	setunitdata(.CloneGID, UDT_MODE, MD_NOKNOCKBACK | MD_CANATTACK);
	setunitdata(.CloneGID, UDT_AI, AI_NONE);
	setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	setunitdata(.CloneGID, UDT_ATKRANGE, 6);
	setunitdata(.CloneGID, UDT_MATKMIN, 500);
	setunitdata(.CloneGID, UDT_MATKMAX, 800);
	setunitdata(.CloneGID, UDT_SP, 2000);
	setunitdata(.CloneGID, UDT_MAXSP, 2000);
	unitattack(.CloneGID, .DummyGID);

	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnTimer2500:
	if (.CloneGID > 0 && .DummyGID > 0) {
		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
		setunitdata(.DummyGID, UDT_HP, 2000000);
		unitattack(.CloneGID, .DummyGID);
		unitskilluseid(.CloneGID, MG_LIGHTNINGBOLT, 3, .DummyGID);
	}
	end;

OnTimer4000:
	if (.CloneGID > 0 && .DummyGID > 0) {
		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
		setunitdata(.DummyGID, UDT_HP, 2000000);
		specialeffect(EF_LIGHTBOLT, AREA, .DummyGID);
	}
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnInteractStart:
	stopnpctimer;
	if (.CloneGID > 0) {
		unitstop(.CloneGID);
		setunitdata(.CloneGID, UDT_LOOKDIR, 6);
	}
	end;

OnInteractEnd:
	stopnpctimer;
	setnpctimer 0;
	initnpctimer;
	end;

OnCloneDead:
	stopnpctimer;
	setnpctimer 0;
	donpcevent "CloneListeSlot3::OnInit";
	end;

OnDummyDead:
	callsub S_SpawnDummy;
	end;

S_SpawnDummy:
	.DummyGID = monster("prontera", .DummyX, .DummyY, "Training Post", 1915, 1, "CloneListeSlot3::OnDummyDead");
	setunitdata(.DummyGID, UDT_MAXHP, 2000000);
	setunitdata(.DummyGID, UDT_HP, 2000000);
	setunitdata(.DummyGID, UDT_MODE, MD_NOKNOCKBACK | MD_BOSS);
	setunitdata(.DummyGID, UDT_AI, AI_NONE);
	return;
}

prontera,174,134,0	script	CloneOttoTalkZone	FAKE_NPC,3,3,{
OnTouch:
	donpcevent "CloneListeSlot3::OnInteractStart";
	mes "Test OTTO";
	close2;
	donpcevent "CloneListeSlot3::OnInteractEnd";
	end;
}

prontera,174,145,0	script	CloneEndTalkZones	FAKE_NPC,6,5,{
OnTouch:
	getmapxy(.@map$, .@x, .@y, UNITTYPE_PC);

	.@cesar_dx = .@x - 172;
	if (.@cesar_dx < 0)
		.@cesar_dx = -.@cesar_dx;
	.@cesar_dy = .@y - 145;
	if (.@cesar_dy < 0)
		.@cesar_dy = -.@cesar_dy;

	.@moka_dx = .@x - 175;
	if (.@moka_dx < 0)
		.@moka_dx = -.@moka_dx;
	.@moka_dy = .@y - 145;
	if (.@moka_dy < 0)
		.@moka_dy = -.@moka_dy;

	if (.@cesar_dx <= 5 && .@cesar_dy <= 5 && (.@cesar_dx + .@cesar_dy) <= (.@moka_dx + .@moka_dy)) {
		donpcevent "CloneListeSlot1::OnInteractStart";
		mes "TEST Cesar";
		close2;
		donpcevent "CloneListeSlot1::OnInteractEnd";
		end;
	}
	if (.@moka_dx <= 5 && .@moka_dy <= 5) {
		donpcevent "CloneListeSlot2::OnInteractStart";
		mes "TEST Moka";
		close2;
		donpcevent "CloneListeSlot2::OnInteractEnd";
		end;
	}
	end;
}
