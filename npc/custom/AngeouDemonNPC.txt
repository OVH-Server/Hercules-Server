//===== Hercules Script ======================================
// Angel / Demon faction choice.
//
// Character variable:
// Faction: 0 = none, 1 = Angel, 2 = Demon
//============================================================

prontera,161,160,4	script	Valkyrie of Destiny	4_F_VALKYRIE,{
	mes "[Valkyrie of Destiny]";

	if (Faction == 1) {
		mes "Your faction is Angel.";
		mes "I can send you to Yuno.";
		next;

		if (select("Go to Yuno:Cancel") == 1) {
			warp "yuno",157,51;
			end;
		}

		close;
	}

	if (Faction == 2) {
		mes "Your faction is Demon.";
		mes "I can send you to Niflheim.";
		next;

		if (select("Go to Niflheim:Cancel") == 1) {
			warp "niflheim",193,182;
			end;
		}

		close;
	}

	mes "Choose your faction.";
	mes "This choice is permanent.";
	next;

	.@choice = select("Angel:Demon:Cancel");

	if (.@choice == 3)
		close;

	mes "[Valkyrie of Destiny]";

	if (.@choice == 1) {
		mes "You chose the Angel faction.";
		next;

		if (select("Confirm Angel:Cancel") == 2)
			close;

		Faction = 1;
		mes "[Valkyrie of Destiny]";
		mes "Your faction is now Angel.";
		mes "Come back to travel to Yuno.";
		close;
	}

	mes "You chose the Demon faction.";
	next;

	if (select("Confirm Demon:Cancel") == 2)
		close;

	Faction = 2;
	mes "[Valkyrie of Destiny]";
	mes "Your faction is now Demon.";
	mes "Come back to travel to Niflheim.";
	close;
}
