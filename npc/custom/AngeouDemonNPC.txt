//===== Hercules Script ======================================

prontera,159,160,4	script	Valkyrie of Destiny::AngelDemon	4_F_VALKYRIE,{

	mes "[Valkyrie of Destiny]";

	if (Faction == 1) {
		mes "You have already chosen the path of the Angels.";
		close;
	}

	if (Faction == 2) {
		mes "You have already chosen the path of the Demons.";
		close;
	}

	mes "Mortal...";
	mes "Which path will you follow?";
	next;

	switch(select("Angel:Demon")) {

	case 1:

		mes "[Valkyrie of Destiny]";
		mes "This choice cannot be reversed.";
		next;

		if(select("Become an Angel:Cancel") == 2)
			close;

		Faction = 1;

		sc_start SC_ANGELUS,2147483647,10;

		specialeffect2 EF_ANGEL;

		mes "[Valkyrie of Destiny]";
		mes "The light embraces your soul.";
		close;

	case 2:

		mes "[Valkyrie of Destiny]";
		mes "This choice cannot be reversed.";
		next;

		if(select("Become a Demon:Cancel") == 2)
			close;

		Faction = 2;

		sc_start SC_DEVOTION,2147483647,10;

		specialeffect2 EF_CURSEATTACK;

		mes "[Valkyrie of Destiny]";
		mes "Darkness flows through your veins.";
		close;
	}

	close;

OnPCLoginEvent:

	if (Faction == 1) {
		sc_start SC_ANGELUS,2147483647,10;
	}

	if (Faction == 2) {
		sc_start SC_DEVOTION,2147483647,10;
	}

	end;
}
