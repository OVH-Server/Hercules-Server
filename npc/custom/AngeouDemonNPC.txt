//===== Hercules Script ======================================
// Angel / Demon alignment choice.
//
// Character variables:
// Faction: 0 = none, 1 = Angel, 2 = Demon
// FactionMascotGid: current visual mascot monster GID
//============================================================

prontera,161,160,4	script	Valkyrie of Destiny	4_F_VALKYRIE,{
	mes "[Valkyrie of Destiny]";

	if (Faction == 1) {
		callfunc "F_AngelDemon_ApplyIndicator";
		mes "You belong to the Angel alignment.";
		next;

		.@choice = select("Travel to Yuno:Refresh my alignment mark:Cancel");

		if (.@choice == 1) {
			warp "yuno", 157, 51;
			end;
		}

		if (.@choice == 2)
			callfunc "F_AngelDemon_ApplyIndicator";

		close;
	}

	if (Faction == 2) {
		callfunc "F_AngelDemon_ApplyIndicator";
		mes "You belong to the Demon alignment.";
		next;

		.@choice = select("Travel to Niflheim:Refresh my alignment mark:Cancel");

		if (.@choice == 1) {
			warp "niflheim", 193, 182;
			end;
		}

		if (.@choice == 2)
			callfunc "F_AngelDemon_ApplyIndicator";

		close;
	}

	mes "Choose your alignment.";
	mes "This choice is permanent and grants no bonus.";
	next;

	.@choice = select("Angel:Demon:Cancel");

	if (.@choice == 3)
		close;

	mes "[Valkyrie of Destiny]";

	if (.@choice == 1) {
		mes "The Angel alignment follows the white light.";
		next;

		if (select("Choose Angel alignment:Cancel") == 2)
			close;

		Faction = 1;
		specialeffect2 EF_CHAINGEHOLY;
		specialeffect2 EF_WHITEBODY;

		mes "[Valkyrie of Destiny]";
		mes "Your alignment is now Angel.";
	} else {
		mes "The Demon alignment follows the black light.";
		next;

		if (select("Choose Demon alignment:Cancel") == 2)
			close;

		Faction = 2;
		specialeffect2 EF_CHANGEDARK;
		specialeffect2 EF_HITDARK;

		mes "[Valkyrie of Destiny]";
		mes "Your alignment is now Demon.";
	}

	callfunc "F_AngelDemon_ApplyIndicator";
	close;
}

function	script	F_AngelDemon_ApplyIndicator	{
	if (Faction == 1) {
		specialeffect2 EF_CHAINGEHOLY;
		dispbottom "Alignment: Angel", 0xFFFFFF;
		return;
	}

	if (Faction == 2) {
		specialeffect2 EF_CHANGEDARK;
		dispbottom "Alignment: Demon", 0x555555;
		return;
	}

	return;
}

yuno,160,51,4	script	Angel Mascot Keeper	4_ANGELING,{
	mes "[Angel Mascot Keeper]";

	if (Faction != 1) {
		mes "Only Angel-aligned adventurers may call this mascot.";
		close;
	}

	mes "Your Angel alignment calls an Angeling to follow you.";
	next;

	if (select("Call Angeling:Cancel") == 2)
		close;

	callfunc "F_AngelDemon_SummonMascot";

	mes "[Angel Mascot Keeper]";
	mes "Your Angeling is now following you.";
	close;
}

niflheim,193,180,4	script	Demon Mascot Keeper	1582,{
	mes "[Demon Mascot Keeper]";

	if (Faction != 2) {
		mes "Only Demon-aligned adventurers may call this mascot.";
		close;
	}

	mes "Your Demon alignment calls a Deviling to follow you.";
	next;

	if (select("Call Deviling:Cancel") == 2)
		close;

	callfunc "F_AngelDemon_SummonMascot";

	mes "[Demon Mascot Keeper]";
	mes "Your Deviling is now following you.";
	close;
}

function	script	F_AngelDemon_SummonMascot	{
	getmapxy .@map$, .@x, .@y, UNITTYPE_PC;

	if (FactionMascotGid > 0)
		unitkill FactionMascotGid;

	if (Faction == 1) {
		.@mob = 1096;
		.@name$ = strcharinfo(PC_NAME) + "'s Angeling";
	} else if (Faction == 2) {
		.@mob = 1582;
		.@name$ = strcharinfo(PC_NAME) + "'s Deviling";
	} else {
		return;
	}

	FactionMascotGid = monster(.@map$, .@x, .@y, .@name$, .@mob, 1);
	setunitdata FactionMascotGid, UDT_MASTERAID, getcharid(CHAR_ID_ACCOUNT);
	setunitdata FactionMascotGid, UDT_MASTERCID, getcharid(CHAR_ID_CHAR);
	setunitdata FactionMascotGid, UDT_MODE, 0;
	setunitdata FactionMascotGid, UDT_ATKMIN, 0;
	setunitdata FactionMascotGid, UDT_ATKMAX, 0;
	setunitdata FactionMascotGid, UDT_MATKMIN, 0;
	setunitdata FactionMascotGid, UDT_MATKMAX, 0;
	setunitdata FactionMascotGid, UDT_SPEED, 160;
	unitwalk FactionMascotGid, getcharid(CHAR_ID_ACCOUNT);
	if (!FactionMascotTimer) {
		FactionMascotTimer = 1;
		addtimer 1500, "AngelDemon_FactionStatus::OnMascotFollow";
	}
	return;
}

-	script	AngelDemon_FactionStatus	FAKE_NPC,{
	end;

OnPCLoginEvent:
	if (Faction == 1 || Faction == 2)
		addtimer 1000, "AngelDemon_FactionStatus::OnRefresh";
	end;

OnPCLogoutEvent:
	if (FactionMascotGid > 0)
		unitkill FactionMascotGid;
	FactionMascotGid = 0;
	FactionMascotTimer = 0;
	end;

OnRefresh:
	callfunc "F_AngelDemon_ApplyIndicator";

	if (Faction == 1) {
		specialeffect2 EF_CHAINGEHOLY;
		end;
	}

	if (Faction == 2) {
		specialeffect2 EF_CHANGEDARK;
		end;
	}

	end;

OnMascotFollow:
	if (Faction != 1 && Faction != 2) {
		FactionMascotTimer = 0;
		end;
	}

	if (FactionMascotGid <= 0)
		callfunc "F_AngelDemon_SummonMascot";
	else
		unitwalk FactionMascotGid, getcharid(CHAR_ID_ACCOUNT);

	addtimer 1500, "AngelDemon_FactionStatus::OnMascotFollow";
	end;
}
