//===== Hercules Script ======================================
// Angel / Demon faction choice.
//
// Character variable:
// Faction: 0 = none, 1 = Angel, 2 = Demon
//============================================================

prontera,152,158,4	script	Alignment: Angel or Demon	4_F_VALKYRIE,{
	mes "[Valkyrie of Destiny]";

	if (Faction == 1) {
		mes "Your faction is Angel.";
		mes "I can send you to Yuno.";
		next;

		if (select("Go to Yuno:Cancel") == 1) {
			warp "yuno",157,51;
			end;
		}

		close;
	}

	if (Faction == 2) {
		mes "Your faction is Demon.";
		mes "I can send you to Niflheim.";
		next;

		if (select("Go to Niflheim:Cancel") == 1) {
			warp "niflheim",193,182;
			end;
		}

		close;
	}

	mes "Choose your faction.";
	mes "This choice is permanent.";
	next;

	.@choice = select("Angel:Demon:Cancel");

	if (.@choice == 3)
		close;

	mes "[Valkyrie of Destiny]";

	if (.@choice == 1) {
		mes "You chose the Angel faction.";
		next;

		if (select("Confirm Angel:Cancel") == 2)
			close;

		Faction = 1;
		mes "[Valkyrie of Destiny]";
		mes "Your faction is now Angel.";
		mes "I will send you to Yuno.";
		close2;
		warp "yuno",157,51;
		end;
	}

	mes "You chose the Demon faction.";
	next;

	if (select("Confirm Demon:Cancel") == 2)
		close;

	Faction = 2;
	mes "[Valkyrie of Destiny]";
	mes "Your faction is now Demon.";
	mes "I will send you to Niflheim.";
	close2;
	warp "niflheim",193,182;
	end;
}

yuno,160,53,4	script	Angel Mascot Valkyrie	4_F_VALKYRIE,{
	mes "[Angel Mascot Valkyrie]";

	if (Faction != 1) {
		mes "This mascot is reserved for the Angel faction.";
		close;
	}

	if (getpetinfo(PETINFO_ID) > 0) {
		mes "Your pet is already with you.";
		close;
	}

	mes "I can give you your Angel mascot.";
	next;

	if (select("Recover Angel mascot:Cancel") == 2)
		close;

	if (!AngelMascotClaimed) {
		makepet(1002);
		AngelMascotClaimed = 1;
	}

	mes "[Angel Mascot Valkyrie]";
	mes "Your mascot is ready.";
	mes "Choose the egg in the pet window to hatch it.";
	close2;
	bpet;
	end;
}

}
