//===== Hercules Script ======================================

prontera,159,160,4	script	Valkyrie of Destiny::AngelDemon	4_F_VALKYRIE,{
	mes "[Valkyrie of Destiny]";

	if (Faction == 1) {
		mes "You have already chosen the Angel path.";
		close;
	}

	if (Faction == 2) {
		mes "You have already chosen the Demon path.";
		close;
	}

	mes "Choose your destiny.";
	next;

	switch(select("Angel:Demon:Cancel")) {

	case 1:
		mes "[Valkyrie of Destiny]";
		mes "This choice is permanent.";
		next;

		if (select("Become an Angel:Cancel") == 2)
			close;

		Faction = 1;

		specialeffect2 EF_CHAINGEHOLY;

		mes "[Valkyrie of Destiny]";
		mes "You are now an Angel.";
		close;

	case 2:
		mes "[Valkyrie of Destiny]";
		mes "This choice is permanent.";
		next;

		if (select("Become a Demon:Cancel") == 2)
			close;

		Faction = 2;

		specialeffect2 EF_CHANGEDARK;

		mes "[Valkyrie of Destiny]";
		mes "You are now a Demon.";
		close;

	case 3:
		close;
	}

	end;
}
