//===== Hercules Script ======================================
// Angel / Demon alignment choice.
//
// Character variables:
// Faction: 0 = none, 1 = Angel, 2 = Demon
// FactionPetClaimed: 0 = not claimed, 1 = claimed
//============================================================

prontera,161,160,4	script	Valkyrie of Destiny::AngelDemon	4_F_VALKYRIE,{
	mes "[Valkyrie of Destiny]";

	if (Faction == 1) {
		callfunc "F_AngelDemon_ApplyIndicator";
		mes "You belong to the Angel alignment.";
		next;

		.@choice = select("Travel to Yuno:Refresh my alignment mark:Cancel");

		if (.@choice == 1) {
			warp "yuno", 157, 51;
			end;
		}

		if (.@choice == 2)
			callfunc "F_AngelDemon_ApplyIndicator";

		close;
	}

	if (Faction == 2) {
		callfunc "F_AngelDemon_ApplyIndicator";
		mes "You belong to the Demon alignment.";
		next;

		.@choice = select("Travel to Niflheim:Refresh my alignment mark:Cancel");

		if (.@choice == 1) {
			warp "niflheim", 193, 182;
			end;
		}

		if (.@choice == 2)
			callfunc "F_AngelDemon_ApplyIndicator";

		close;
	}

	mes "Choose your alignment.";
	mes "This choice is permanent and grants no bonus.";
	next;

	.@choice = select("Angel:Demon:Cancel");

	if (.@choice == 3)
		close;

	mes "[Valkyrie of Destiny]";

	if (.@choice == 1) {
		mes "The Angel alignment follows the white light.";
		next;

		if (select("Choose Angel alignment:Cancel") == 2)
			close;

		Faction = 1;
		specialeffect2 EF_CHAINGEHOLY;
		specialeffect2 EF_WHITEBODY;

		mes "[Valkyrie of Destiny]";
		mes "Your alignment is now Angel.";
	} else {
		mes "The Demon alignment follows the black light.";
		next;

		if (select("Choose Demon alignment:Cancel") == 2)
			close;

		Faction = 2;
		specialeffect2 EF_CHANGEDARK;
		specialeffect2 EF_HITDARK;

		mes "[Valkyrie of Destiny]";
		mes "Your alignment is now Demon.";
	}

	callfunc "F_AngelDemon_ApplyIndicator";
	close;
}

function	script	F_AngelDemon_ApplyIndicator	{
	if (Faction == 1) {
		sc_end SC_DEVIL1;
		sc_start SC_DRESS_UP, 2147483647, 1;
		dispbottom "Alignment: Angel", 0xFFFFFF;
		return;
	}

	if (Faction == 2) {
		sc_end SC_DRESS_UP;
		sc_start SC_DEVIL1, 2147483647, 1;
		dispbottom "Alignment: Demon", 0x555555;
		return;
	}

	sc_end SC_DRESS_UP;
	sc_end SC_DEVIL1;
	return;
}

yuno,160,51,4	script	Angel Mascot Keeper	4_ANGELING,{
	mes "[Angel Mascot Keeper]";

	if (Faction != 1) {
		mes "Only Angel-aligned adventurers may receive this mascot.";
		close;
	}

	if (FactionPetClaimed) {
		mes "You have already received your faction mascot.";
		close;
	}

	mes "Your Angel alignment allows you to receive an Angeling mascot egg.";
	next;

	if (select("Receive Angeling Egg:Cancel") == 2)
		close;

	makepet ANGELING;
	FactionPetClaimed = 1;

	mes "[Angel Mascot Keeper]";
	mes "Take good care of your Angeling.";
	next;
	bpet;
	close;
}

niflheim,193,180,4	script	Demon Mascot Keeper	DEVILING,{
	mes "[Demon Mascot Keeper]";

	if (Faction != 2) {
		mes "Only Demon-aligned adventurers may receive this mascot.";
		close;
	}

	if (FactionPetClaimed) {
		mes "You have already received your faction mascot.";
		close;
	}

	mes "Your Demon alignment allows you to receive a Deviling mascot egg.";
	next;

	if (select("Receive Deviling Egg:Cancel") == 2)
		close;

	makepet DEVILING;
	FactionPetClaimed = 1;

	mes "[Demon Mascot Keeper]";
	mes "Take good care of your Deviling.";
	next;
	bpet;
	close;
}

-	script	AngelDemon_FactionStatus	FAKE_NPC,{
	end;

OnPCLoginEvent:
	if (Faction == 1 || Faction == 2)
		addtimer 1000, "AngelDemon_FactionStatus::OnRefresh";
	end;

OnRefresh:
	callfunc "F_AngelDemon_ApplyIndicator";

	if (Faction == 1) {
		specialeffect2 EF_CHAINGEHOLY;
		end;
	}

	if (Faction == 2) {
		specialeffect2 EF_CHANGEDARK;
		end;
	}

	end;
}
