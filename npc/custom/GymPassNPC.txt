//===== Hercules Script ======================================
// Gym Pass trainer, based on the official Payon NPC.
// Custom limit: 50 Gym Passes, with multi-use support.
//============================================================

prontera,159,157,4	script	Valkyrie du Poids	4_F_VALKYRIE,{
	mes "[Valkyrie]";
	mes "Je peux utiliser tes ^FF0000Gym Pass^000000 pour augmenter ta capacite de poids.";
	mes "Progression actuelle: ^3131FF" + gympassmemory + "/50^000000.";
	next;

	switch(select("Utiliser des Gym Pass:Restaurer mon bonus:Annuler")) {
	case 1:
		if (gympassmemory >= 50) {
			mes "[Valkyrie]";
			mes "Ton entrainement est deja au maximum.";
			close;
		}

		.@passes = countitem(Max_Weight_Up_Scroll);
		if (.@passes <= 0) {
			mes "[Valkyrie]";
			mes "Tu dois apporter au moins un ^FF0000Gym Pass^000000.";
			close;
		}

		.@remaining = 50 - gympassmemory;
		if (.@passes > .@remaining)
			.@passes = .@remaining;

		mes "[Valkyrie]";
		mes "Tu peux utiliser jusqu'a ^3131FF" + .@passes + "^000000 Gym Pass maintenant.";
		mes "Combien veux-tu en utiliser ?";
		next;

		input .@use,1,.@passes;
		if (.@use < 1 || .@use > .@passes) {
			mes "[Valkyrie]";
			mes "Nombre invalide.";
			close;
		}

		mes "[Valkyrie]";
		mes "Tres bien. Que l'entrainement commence.";
		next;
		specialeffect(EF_EARTHSPIKE, AREA, playerattached());
		sleep2 250;
		specialeffect(EF_DEVIL, AREA, playerattached());
		sleep2 250;
		specialeffect(EF_COIN, AREA, playerattached());
		sleep2 250;
		specialeffect(EF_SIGHTRASHER, AREA, playerattached());
		next;

		delitem Max_Weight_Up_Scroll,.@use;
		gympassmemory += .@use;
		.@skill_lv = gympassmemory;
		if (.@skill_lv > 10)
			.@skill_lv = 10;
		skill ALL_INCCARRY,.@skill_lv,3;

		mes "[Valkyrie]";
		mes "C'est fait. Progression: ^3131FF" + gympassmemory + "/50^000000.";
		mes "Poids maximum actuel: " + (MaxWeight / 10) + ".";
		mes "Les Gym Pass au-dela de 10 demandent le map-server custom recompile.";
		close;
	case 2:
		if (gympassmemory <= 0) {
			mes "[Valkyrie]";
			mes "Tu n'as aucun entrainement a restaurer.";
			close;
		}

		.@skill_lv = gympassmemory;
		if (.@skill_lv > 10)
			.@skill_lv = 10;
		skill ALL_INCCARRY,.@skill_lv,3;
		mes "[Valkyrie]";
		mes "Ta memoire musculaire est restauree.";
		mes "Progression: ^3131FF" + gympassmemory + "/50^000000.";
		close;
	case 3:
		mes "[Valkyrie]";
		mes "Reviens quand tu voudras t'entrainer.";
		close;
	}
}
