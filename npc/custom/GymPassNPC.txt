//===== Hercules Script ======================================
// Gym Pass trainer, based on the official Payon NPC.
// Custom limit: 50 Gym Passes, with multi-use support.
//============================================================

prontera,159,157,4	script	Weight Valkyrie	4_F_VALKYRIE,{
	mes "[Valkyrie]";
	mes "I can use your ^FF0000Gym Passes^000000 to increase your weight capacity.";
	mes "Current progress: ^3131FF" + gympassmemory + "/50^000000.";
	next;

	switch(select("Use Gym Passes:Restore my bonus:Cancel")) {
	case 1:
		if (gympassmemory >= 50) {
			mes "[Valkyrie]";
			mes "Your training is already at its maximum.";
			close;
		}

		.@passes = countitem(Max_Weight_Up_Scroll);
		if (.@passes <= 0) {
			mes "[Valkyrie]";
			mes "You need to bring at least one ^FF0000Gym Pass^000000.";
			close;
		}

		.@remaining = 50 - gympassmemory;
		if (.@passes > .@remaining)
			.@passes = .@remaining;

		mes "[Valkyrie]";
		mes "You can use up to ^3131FF" + .@passes + "^000000 Gym Passes now.";
		mes "How many do you want to use?";
		next;

		input .@use,1,.@passes;
		if (.@use < 1 || .@use > .@passes) {
			mes "[Valkyrie]";
			mes "Invalid amount.";
			close;
		}

		mes "[Valkyrie]";
		mes "Very well. Let the training begin.";
		next;
		specialeffect(EF_EARTHSPIKE, AREA, playerattached());
		sleep2 250;
		specialeffect(EF_DEVIL, AREA, playerattached());
		sleep2 250;
		specialeffect(EF_COIN, AREA, playerattached());
		sleep2 250;
		specialeffect(EF_SIGHTRASHER, AREA, playerattached());
		next;

		delitem Max_Weight_Up_Scroll,.@use;
		gympassmemory += .@use;
		.@skill_lv = gympassmemory;
		if (.@skill_lv > 10)
			.@skill_lv = 10;
		skill ALL_INCCARRY,.@skill_lv,3;

		mes "[Valkyrie]";
		mes "Done. Progress: ^3131FF" + gympassmemory + "/50^000000.";
		mes "Current maximum weight: " + (MaxWeight / 10) + ".";
		mes "Gym Passes beyond 10 require the custom map-server to be recompiled.";
		close;
	case 2:
		if (gympassmemory <= 0) {
			mes "[Valkyrie]";
			mes "You have no training to restore.";
			close;
		}

		.@skill_lv = gympassmemory;
		if (.@skill_lv > 10)
			.@skill_lv = 10;
		skill ALL_INCCARRY,.@skill_lv,3;
		mes "[Valkyrie]";
		mes "Your muscle memory has been restored.";
		mes "Progress: ^3131FF" + gympassmemory + "/50^000000.";
		close;
	case 3:
		mes "[Valkyrie]";
		mes "Come back whenever you want to train.";
		close;
	}
}
