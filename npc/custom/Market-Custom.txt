prontera,163,205,4	script	Places marchandes	1_F_MERCHANT_02,{

OnInit: waitingroom "Places Marchandes",0;

	switch(select(
	"^FF0000Place marchande d'Alberta^000000:"
	+"^FF0000Place marchande d'Ayothaya^000000:"
	+"^FF0000Place marchande de Lighthalzen^000000:"
	+"Annuler")) {

	case 1:
		mes "[Transporteur]";
		mes "Direction ^FF0000la place marchande d'Alberta !^000000";
		close2;
		warp "alberta",212,51;
		end;

	case 2:
		mes "[Transporteur]";
		mes "Direction ^FF0000la place marchande d'Ayothaya !^000000";
		close2;
		warp "ayothaya",161,169;
		end;

	case 3:
		mes "[Transporteur]";
		mes "Direction ^FF0000la place marchande de Lighthalzen !^000000";
		close2;
		warp "lighthalzen",158,54;
		end;

	case 4:
		mes "[Transporteur]";
		mes "Tres bien.";
		close;
	}
}

/////--- Bot Marchand Alberta " Il est frais mon poisson ! "
alberta,214,38,6	script	Aladeen	4_M_ALADDIN,{

	mes "[Aladeen]";

	if (.stock_count <= 0) {
		callsub S_GenerateStock;
		mes "J'ai reçu une nouvelle cargaison.";
		next;
	}

	mes "Voilà ce que je vends :";
	next;

	.@menu$ = "";

	for (.@i = 0; .@i < .stock_count; .@i++) {
		.@menu$ += getitemname(.stock_id[.@i])+" x"+.stock_amount[.@i]+" - "+.stock_price[.@i]+"z:";
	}

	.@menu$ += "Quitter";

	.@choice = select(.@menu$) - 1;

	if (.@choice == .stock_count) close;

	if (Zeny < .stock_price[.@choice]) {
		mes "[Aladeen]";
		mes "Tu n'as pas assez de zeny.";
		close;
	}

	set Zeny, Zeny - .stock_price[.@choice];
	getitem .stock_id[.@choice],1;

	.stock_amount[.@choice]--;

	mes "[Aladeen]";
	mes "Marché conclu.";
	next;

	if (.stock_amount[.@choice] <= 0) {
		for (.@i = .@choice; .@i < .stock_count - 1; .@i++) {
			.stock_id[.@i] = .stock_id[.@i+1];
			.stock_price[.@i] = .stock_price[.@i+1];
			.stock_amount[.@i] = .stock_amount[.@i+1];
		}
		.stock_count--;
	}

	close;

OnInit:
	initnpctimer;
	end;

OnTimer15000:
	npctalk "Il est frais mon poisson !";
	initnpctimer;
	end;

S_GenerateStock:

	deletearray .stock_id[0], getarraysize(.stock_id);
	deletearray .stock_price[0], getarraysize(.stock_price);
	deletearray .stock_amount[0], getarraysize(.stock_amount);

	setarray .@pool[0],
		909,910,911,912,913,914,915,938,942,943;

	.stock_count = rand(5,10);

	for (.@i = 0; .@i < .stock_count; .@i++) {
		.stock_id[.@i] = .@pool[rand(getarraysize(.@pool))];
		.stock_price[.@i] = rand(1,10) * 1000;
		.stock_amount[.@i] = rand(1,99);
	}

	return;
}
