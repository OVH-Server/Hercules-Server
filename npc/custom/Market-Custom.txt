prontera,163,205,4	script	Market Places	1_F_MERCHANT_02,{

	+"^FF0000Lighthalzen Market Place^000000:"
	+"Cancel")) {

		mes "[Transporter]";
		mes "Heading to the ^FF0000Alberta Market Place!^000000";
		close2;

		end;

	case 2:
		mes "[Transporter]";
		mes "Heading to the ^FF0000Ayothaya Market Place!^000000";
		close2;
		warp "ayothaya",161,169;
		end;

	case 3:
		mes "[Transporter]";
		mes "Heading to the ^FF0000Lighthalzen Market Place!^000000";
		close2;
		warp "lighthalzen",158,54;
		end;

	case 4:
		mes "[Transporter]";
		mes "Very well.";
		close;
	}
}

// Alberta merchant bot.
alberta,214,38,6	script	Aladeen	4_M_ALADDIN,{

	mes "[Aladeen]";

	if (.stock_count <= 0) {
		callsub S_GenerateStock;
		mes "I received a new shipment.";
		next;
	}

	mes "Here is what I sell:";
	next;

	.@menu$ = "";

	for (.@i = 0; .@i < .stock_count; .@i++) {
		.@menu$ += getitemname(.stock_id[.@i])+" x"+.stock_amount[.@i]+" - "+.stock_price[.@i]+"z:";
	}

	.@menu$ += "Leave";

	.@choice = select(.@menu$) - 1;

	if (.@choice == .stock_count) close;

	if (Zeny < .stock_price[.@choice]) {
		mes "[Aladeen]";
		mes "You do not have enough zeny.";
		close;
	}

	set Zeny, Zeny - .stock_price[.@choice];
	getitem .stock_id[.@choice],1;

	.stock_amount[.@choice]--;

	mes "[Aladeen]";
	mes "Deal.";
	next;

	if (.stock_amount[.@choice] <= 0) {
		for (.@i = .@choice; .@i < .stock_count - 1; .@i++) {
			.stock_id[.@i] = .stock_id[.@i+1];
			.stock_price[.@i] = .stock_price[.@i+1];
			.stock_amount[.@i] = .stock_amount[.@i+1];
		}
		.stock_count--;
	}

	close;

OnInit:
	initnpctimer;
	end;

OnTimer15000:
	npctalk "Fresh fish for sale!";
	initnpctimer;
	end;

S_GenerateStock:

	deletearray .stock_id[0], getarraysize(.stock_id);
	deletearray .stock_price[0], getarraysize(.stock_price);
	deletearray .stock_amount[0], getarraysize(.stock_amount);

	setarray .@pool[0],
		909,910,911,912,913,914,915,938,942,943;

	.stock_count = rand(5,10);

	for (.@i = 0; .@i < .stock_count; .@i++) {
		.stock_id[.@i] = .@pool[rand(getarraysize(.@pool))];
		.stock_price[.@i] = rand(1,10) * 1000;
		.stock_amount[.@i] = rand(1,99);
	}

	return;
}
