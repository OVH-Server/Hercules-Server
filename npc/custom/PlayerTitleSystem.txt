//----PlayerTitleSystem----

function	script	PlayerTitleSystem	{
	mes "[Title Selector]";
	mes "Titre actuel : " + ((#PlayerTitle$ == "") ? "Aucun" : #PlayerTitle$);
	mes "Choisis ton titre :";
	next;

	switch(select("Newbie:GM:MacGyver:Retirer mon titre")) {
		case 1:
			set #PlayerTitle$, "Newbie";
			break;

		case 2:
			set #PlayerTitle$, "GM";
			break;

		case 3:
			set #PlayerTitle$, "MacGyver";
			break;

		case 4:
			set #PlayerTitle$, "";
			break;
	}

	mes "[Title Selector]";

	if (#PlayerTitle$ == "") {
		mes "Ton titre a été retiré.";
	} else {
		mes "Ton titre est maintenant : [" + #PlayerTitle$ + "]";
	}

	close;
}

// NPC
prontera,164,153,4	script	Title Selector	4_M_OLDFRIAR,{
	callfunc "PlayerTitleSystem";
	end;
}
