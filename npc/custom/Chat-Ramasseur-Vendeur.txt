//===== Hercules Script ======================================
// Collector Cat Vendor
//
// This NPC scans a controlled area around its home, collects accepted
// ground items into its stock, clears the scanned floor area, then returns
// home to sell what it found.
//
// Note: Hercules scripts can count accepted floor items, but cannot remove
// one specific floor item ID. The cleanup uses cleanarea(), so keep the
// scan range small and controlled.
//============================================================

prontera,159,170,4	script	Collector Cat	4_CAT_DOWN,{
	callsub S_CollectGroundItems;

	mes "[Collector Cat]";
	mes "Meow. I collect abandoned items around me, then sell them here.";
	next;

	if (.StockCount <= 0) {
		mes "[Collector Cat]";
		mes "I have nothing to sell right now.";
		mes "Leave a few items in my area and I will make my rounds.";
		close;
	}

	callsub S_ShowStock;
	close;

S_ShowStock:
	.@menu$ = "";
	for (.@i = 0; .@i < .StockCount; .@i++)
		.@menu$ = .@menu$ + getitemname(.StockId[.@i]) + " x" + .StockAmount[.@i] + " - " + .StockPrice[.@i] + "z:";

	.@menu$ = .@menu$ + "Leave";
	.@choice = select(.@menu$) - 1;

	if (.@choice == .StockCount)
		return;

	mes "[Collector Cat]";
	mes "Item: ^00AAFF" + getitemname(.StockId[.@choice]) + "^000000";
	mes "Price: ^00AAFF" + .StockPrice[.@choice] + " Zeny^000000";
	mes "Stock: " + .StockAmount[.@choice];
	next;

	if (select("Buy:Back") == 2) {
		callsub S_ShowStock;
		return;
	}

	if (Zeny < .StockPrice[.@choice]) {
		mes "[Collector Cat]";
		mes "You do not have enough Zeny.";
		return;
	}

	if (!checkweight(.StockId[.@choice],1)) {
		mes "[Collector Cat]";
		mes "Your inventory is too heavy.";
		return;
	}

	Zeny -= .StockPrice[.@choice];
	getitem .StockId[.@choice],1;
	.StockAmount[.@choice]--;

	if (.StockAmount[.@choice] <= 0)
		callsub S_RemoveStock,.@choice;

	mes "[Collector Cat]";
	mes "Meow. Deal.";
	return;

S_RemoveStock:
	.@index = getarg(0);
	for (.@i = .@index; .@i < .StockCount - 1; .@i++) {
		.StockId[.@i] = .StockId[.@i + 1];
		.StockAmount[.@i] = .StockAmount[.@i + 1];
		.StockPrice[.@i] = .StockPrice[.@i + 1];
	}

	.StockCount--;
	.StockId[.StockCount] = 0;
	.StockAmount[.StockCount] = 0;
	.StockPrice[.StockCount] = 0;
	return;

S_AddStock:
	.@item_id = getarg(0);
	.@amount = getarg(1);

	if (.@item_id <= 0 || .@amount <= 0)
		return;

	for (.@i = 0; .@i < .StockCount; .@i++) {
		if (.StockId[.@i] == .@item_id) {
			.StockAmount[.@i] += .@amount;
			return;
		}
	}

	if (.StockCount >= .MaxStock)
		return;

	.StockId[.StockCount] = .@item_id;
	.StockAmount[.StockCount] = .@amount;
	.@type = getiteminfo(.@item_id, ITEMINFO_TYPE);

	if (.@type == IT_WEAPON || .@type == IT_ARMOR)
		.StockPrice[.StockCount] = 500 + ((.@item_id * 37) % 501);
	else if (.@type == IT_HEALING || .@type == IT_USABLE || .@type == IT_DELAYCONSUME || .@type == IT_CASH)
		.StockPrice[.StockCount] = 300 + ((.@item_id * 31) % 201);
	else
		.StockPrice[.StockCount] = 100 + ((.@item_id * 17) % 101);

	.StockCount++;
	return;

S_ScanAcceptedItems:
	.FoundCount = 0;

	for (.@id = .ScanMin; .@id <= .ScanMax; .@id++) {
		.@type = getiteminfo(.@id, ITEMINFO_TYPE);
		.@accepted = 0;

		if (.@type == IT_ETC)
			.@accepted = 1;
		if (.@type == IT_HEALING || .@type == IT_USABLE || .@type == IT_DELAYCONSUME || .@type == IT_CASH)
			.@accepted = 1;
		if (.@type == IT_WEAPON || .@type == IT_ARMOR)
			.@accepted = 1;

		if (!.@accepted)
			continue;

		.@amount = getareadropitem(.HomeMap$,.ScanX1,.ScanY1,.ScanX2,.ScanY2,.@id);

		if (.@amount <= 0)
			continue;

		callsub S_AddStock,.@id,.@amount;
		.FoundCount += .@amount;
	}
	return;

OnInit:
	.HomeMap$ = "prontera";
	.HomeX = 159;
	.HomeY = 170;
	.ScanX1 = 145;
	.ScanY1 = 160;
	.ScanX2 = 175;
	.ScanY2 = 210;
	.MaxStock = 200;
	.StockCount = 0;
	.Busy = 0;
	.ScanMin = 501;
	.ScanMax = 32767;

	movenpc "Collector Cat",.HomeX,.HomeY,4;
	initnpctimer;
	end;

S_CollectGroundItems:
	if (.Busy)
		return;

	.Busy = 1;
	callsub S_ScanAcceptedItems;

	if (.FoundCount <= 0) {
		.Busy = 0;
		return;
	}

	specialeffect EF_STEAL;
	cleanarea .HomeMap$,.ScanX1,.ScanY1,.ScanX2,.ScanY2;
	movenpc "Collector Cat",.HomeX,.HomeY,4;
	.Busy = 0;
	return;

OnTimer10000:
	callsub S_CollectGroundItems;
	setnpctimer 0;
	initnpctimer;
	end;
}
