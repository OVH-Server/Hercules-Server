//===== Hercules Script ======================================
// Collector Cat Vendor
//
// This NPC scans a controlled area around its home, collects accepted
// ground items into its stock, clears the scanned floor area, then returns
// home to sell what it found.
//
// Note: Hercules scripts can count accepted floor items, but cannot remove
// one specific floor item ID. The cleanup uses cleanarea(), so keep the
// scan range small and controlled.
//============================================================

prontera,159,170,4	script	Collector Cat	4_CAT_DOWN,{
	mes "[Collector Cat]";
	mes "Meow. I collect abandoned items around me, then sell them here.";
	next;

	if (.StockCount <= 0) {
		mes "[Collector Cat]";
		mes "I have nothing to sell right now.";
		mes "Leave a few items in my area and I will make my rounds.";
		close;
	}

	callsub S_ShowStock;
	close;

S_ShowStock:
	.@menu$ = "";
	for (.@i = 0; .@i < .StockCount; .@i++)
		.@menu$ = .@menu$ + getitemname(.StockId[.@i]) + " x" + .StockAmount[.@i] + " - " + .StockPrice[.@i] + "z:";

	.@menu$ = .@menu$ + "Leave";
	.@choice = select(.@menu$) - 1;

	if (.@choice == .StockCount)
		return;

	mes "[Collector Cat]";
	mes "Item: ^00AAFF" + getitemname(.StockId[.@choice]) + "^000000";
	mes "Price: ^00AAFF" + .StockPrice[.@choice] + " Zeny^000000";
	mes "Stock: " + .StockAmount[.@choice];
	next;

	if (select("Buy:Back") == 2) {
		callsub S_ShowStock;
		return;
	}

	if (Zeny < .StockPrice[.@choice]) {
		mes "[Collector Cat]";
		mes "You do not have enough Zeny.";
		return;
	}

	if (!checkweight(.StockId[.@choice],1)) {
		mes "[Collector Cat]";
		mes "Your inventory is too heavy.";
		return;
	}

	Zeny -= .StockPrice[.@choice];
	getitem .StockId[.@choice],1;
	.StockAmount[.@choice]--;

	if (.StockAmount[.@choice] <= 0)
		callsub S_RemoveStock,.@choice;

	mes "[Collector Cat]";
	mes "Meow. Deal.";
	return;

S_RemoveStock:
	.@index = getarg(0);
	for (.@i = .@index; .@i < .StockCount - 1; .@i++) {
		.StockId[.@i] = .StockId[.@i + 1];
		.StockAmount[.@i] = .StockAmount[.@i + 1];
		.StockPrice[.@i] = .StockPrice[.@i + 1];
	}

	.StockCount--;
	.StockId[.StockCount] = 0;
	.StockAmount[.StockCount] = 0;
	.StockPrice[.StockCount] = 0;
	return;

S_AddStock:
	.@item_id = getarg(0);
	.@amount = getarg(1);

	if (.@item_id <= 0 || .@amount <= 0)
		return;

	for (.@i = 0; .@i < .StockCount; .@i++) {
		if (.StockId[.@i] == .@item_id) {
			.StockAmount[.@i] += .@amount;
			return;
		}
	}

	if (.StockCount >= .MaxStock)
		return;

	.StockId[.StockCount] = .@item_id;
	.StockAmount[.StockCount] = .@amount;
	.StockPrice[.StockCount] = callsub(S_GetPrice,.@item_id);
	.StockCount++;
	return;

S_GetPrice:
	.@item_id = getarg(0);
	.@type = getiteminfo(.@item_id, ITEMINFO_TYPE);
	.@sell = getiteminfo(.@item_id, ITEMINFO_SELLPRICE);

	if (.@sell < 1)
		.@sell = 50;

	if (.@type == IT_WEAPON || .@type == IT_ARMOR)
		return .@sell * 3;

	if (.@type == IT_HEALING || .@type == IT_USABLE)
		return .@sell * 2;

	return .@sell * 2;

S_ScanAcceptedItems:
	.FoundCount = 0;

	for (.@i = 0; .@i < .AcceptedCount; .@i++) {
		.@id = .AcceptedId[.@i];
		.@amount = getareadropitem(.HomeMap$,.ScanX1,.ScanY1,.ScanX2,.ScanY2,.@id);

		if (.@amount <= 0)
			continue;

		callsub S_AddStock,.@id,.@amount;
		.FoundCount += .@amount;
	}
	return;

OnInit:
	.HomeMap$ = "prontera";
	.HomeX = 155;
	.HomeY = 196;
	.ScanX1 = 145;
	.ScanY1 = 186;
	.ScanX2 = 175;
	.ScanY2 = 210;
	.ScanCenterX = 160;
	.ScanCenterY = 198;
	.MaxStock = 50;
	.StockCount = 0;
	.Busy = 0;

	setarray .AcceptedId[0],
		501,502,503,504,505,506,607,608,
		512,513,514,515,516,517,518,519,520,521,522,523,525,526,528,529,530,531,532,533,534,535,
		901,902,903,904,905,906,907,908,909,910,911,912,913,914,915,916,917,918,919,920,
		921,922,923,924,925,926,928,929,930,931,932,934,935,936,937,938,939,940,941,942,
		943,944,945,946,947,948,949,950,951,952,953,954,955,956,957,958,959,960,961,962,
		963,964,965,966,967,968,969,970,971,972,973,974,975,976,978,979,980,981,982,983,
		984,985,986,987,988,989,990,991,992,993,994,995,996,997,998,999,
		1000,1001,1002,1003,1004,1005,1006,1007,1008,1009,1010,1011,1012,1013,1014,1015,
		1016,1017,1018,1019,1020,1021,1022,1023,1024,1025,1026,1027,1028,1029,1030,1031,
		1032,1033,1034,1035,1036,1037,1038,1039,1040,1041,1042,1043,1044,1045,1046,1047,
		1048,1049,1050,1051,1052,1053,1054,1055,1056,1057,1058,1059,1060,1061,1062,1063,
		1064,1065,1066,1067,1068,1069,1070,1071,1072,1073,1074,1075,1076,1077,1078,1079,
		1080,1081,1082,1083,1084,1085,1086,1087,1088,1089,1090,1091,1092,1093,1094,1095,
		1096,1097,1098,1099,7001,7002,7003,7004,7005,7006,7007,7008,7009,7010,7011,7012,
		7013,7014,7015,7016,7017,7018,7019,7020,7021,7022,7023,7024,7025,7026,7027,7028,
		7029,7030,7031,7032,7033,7034,7035,7036,7037,7038,7039,7040,7041,7042,7043,7044,
		7045,7046,7047,7048,7049,7050,7051,7052,7053,7054,7055,7056,7057,7058,7059,7060,
		7061,7062,7063,7064,7065,7066,7067,7068,7069,7070,7071,7072,7073,7074,7075,7076,
		7077,7078,7079,7080,7081,7082,7083,7084,7085,7086,7087,7088,7089,7090,7091,7092,
		7093,7094,7095,7096,7097,7098,7099;
	.AcceptedCount = getarraysize(.AcceptedId);

	npcspeed 100;
	waitingroom "Found items for sale",0;
	initnpctimer;
	end;

OnTimer5000:
	if (.Busy) {
		setnpctimer 0;
		end;
	}

	.Busy = 1;
	callsub S_ScanAcceptedItems;

	if (.FoundCount <= 0) {
		.Busy = 0;
		setnpctimer 0;
		end;
	}

	delwaitingroom;
	waitingroom "Collecting items",0;
	npcwalkto .ScanCenterX,.ScanCenterY;
	end;

OnTimer8000:
	if (!.Busy) {
		setnpctimer 0;
		end;
	}

	specialeffect EF_STEAL;
	cleanarea .HomeMap$,.ScanX1,.ScanY1,.ScanX2,.ScanY2;
	npcwalkto .HomeX,.HomeY;
	end;

OnTimer12000:
	if (!.Busy) {
		setnpctimer 0;
		end;
	}

	movenpc "Collector Cat",.HomeX,.HomeY,4;
	delwaitingroom;
	waitingroom "Found items for sale",0;
	npctalk "Meow. I collected " + .FoundCount + " item(s).";
	.Busy = 0;
	setnpctimer 0;
	end;
}
