//===== Hercules Script ======================================

prontera,148,195,3	script	Megaphone	4_CAT_MERMASTER,{
	mes "[Megaphone]";
	mes "I can broadcast your message to every player on the server!";
	mes "Price: ^FF000010,000 zeny^000000.";
	next;

	if (Zeny < 10000) {
		mes "[Mayor]";
		mes "You do not have enough zeny.";
		mes "Come back later.";
		close;
	}

	mes "[Megaphone]";
	mes "Which channel do you want to use?";
	next;

	switch(select("Trade:Recruitment:Cancel")) {
		case 1:
			set .@channel$, "Trade";
			break;

		case 2:
			set .@channel$, "Recruitment";
			break;

		case 3:
			mes "[Megaphone]";
			mes "Cancelled.";
			close;
	}

	mes "[Megaphone]";
	mes "Write your message:";
	next;

	input .@message$;

	if (.@message$ == "") {
		mes "[Mayor]";
		mes "Empty message cancelled.";
		close;
	}

	mes "[Megaphone]";
	mes "Your message will be sent to the ^00AAFF" + .@channel$ + "^000000 channel:";
	mes "\"" + .@message$ + "\"";
	next;

	if (select("Send:Cancel") == 2) {
		mes "[Mayor]";
		mes "Cancelled.";
		close;
	}

	if (Zeny < 10000) {
		mes "[Megaphone]";
		mes "You no longer have enough zeny.";
		close;
	}

	set Zeny, Zeny - 10000;

	if (.@channel$ == "Trade") {
		announce "[Trade] " + strcharinfo(0) + ": " + .@message$, bc_all, 0xFFA500;
	} else {
		announce "[Recruitment] " + strcharinfo(0) + ": " + .@message$, bc_all, 0x00FF00;
	}

	mes "[Megaphone]";
	mes "Message received!";
	close;
}
