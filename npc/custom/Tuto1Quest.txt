//===== Hercules Script ======================================
// Main Quest Tuto 1
// Flow:
// Olga -> kill 2 Poring, kill 2 Lunatic, visit checkpoint
// Olgi -> validates the first part
// Olgu -> asks the player to return to Olga with 5 Jellopy
// Olga -> final reward
//============================================================

// Quest IDs:
// 65000 = monster training
// 65001 = speak with Olgu
// 65002 = bring 5 Jellopy

// Character variables:
// Tuto1_State: 0 none, 10 Olga task, 20 Olgi validated, 30 bring Jellopy, 40 done
// Tuto1_Poring, Tuto1_Lunatic: chat progress counters only
// Tuto1_Checkpoint: checkpoint progress
// Tuto1_Jellopy_Last, Tuto1_Jellopy_Timer: live Jellopy chat progress

//============================================================
// NPC 1 - Olga
//============================================================

new_1-3,99,30,4	script	Olga	4_F_KAFRA1,{
	if (#DebugConsole)
		dispbottom "[DEBUG][Olga] Click. State=" + Tuto1_State;

	mes "[Olga]";

	if (Tuto1_State >= 40 || questprogress(65002) == 2) {
		mes "You already completed Main Quest Tuto 1.";
		close;
	}

	if (Tuto1_State == 0) {
		mes "Welcome.";
		mes "This is Main Quest Tuto 1.";
		mes "Your first mission:";
		mes "- Kill 2 Poring";
		mes "- Kill 2 Lunatic";
		mes "- Visit new_1-3,87,32";
		next;

		if (select("Accept:Decline") == 2) {
			mes "[Olga]";
			mes "Come back when you are ready.";
			close;
		}

		if (questprogress(65000))
			erasequest 65000;
		if (questprogress(65001))
			erasequest 65001;
		if (questprogress(65002))
			erasequest 65002;

		Tuto1_State = 10;
		Tuto1_Poring = 0;
		Tuto1_Lunatic = 0;
		Tuto1_Checkpoint = 0;
		Tuto1_Jellopy_Last = -1;
		Tuto1_Jellopy_Timer = 0;

		setquest 65000;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Poring 0/2, Lunatic 0/2, Location 0/1.";

		mes "[Olga]";
		mes "Quest started.";
		mes "The monster kills are visible in your quest list.";
		mes "Once everything is done, go speak with Olgi.";
		close;
	}

	if (Tuto1_State == 10) {
		mes "Current task:";
		mes "Poring: " + Tuto1_Poring + "/2";
		mes "Lunatic: " + Tuto1_Lunatic + "/2";
		mes "Location: " + Tuto1_Checkpoint + "/1";
		mes "Olgi will validate this part.";
		close;
	}

	if (Tuto1_State == 20) {
		mes "Olgi validated your first task.";
		mes "Go speak with Olgu.";
		close;
	}

	if (Tuto1_State == 30) {
		.@jellopy = countitem(909);
		if (.@jellopy > 5)
			.@jellopy = 5;

		mes "Olgu sent you back to me.";
		mes "Bring me 5 Jellopy.";
		mes "Current progress: " + .@jellopy + "/5";
		next;

		if (countitem(909) < 5) {
			mes "[Olga]";
			mes "You do not have 5 Jellopy yet.";
			close;
		}

		delitem 909,5;
		Zeny += 100000;
		getitem 12622,1;

		Tuto1_State = 40;
		Tuto1_Jellopy_Timer = 0;
		Tuto1_Jellopy_Last = -1;

		completequest 65002;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: completed.";

		mes "[Olga]";
		mes "Quest complete.";
		mes "Reward:";
		mes "100,000 Zeny";
		mes "1 Halter Lead";
		close;
	}

	mes "Your tutorial state was invalid.";
	mes "I reset it. Please speak with me again.";
	Tuto1_State = 0;
	close;
}

//============================================================
// NPC 2 - Olgi
//============================================================

new_1-3,99,32,4	script	Olgi	4_M_03,{
	if (#DebugConsole)
		dispbottom "[DEBUG][Olgi] Click. State=" + Tuto1_State + " Poring=" + Tuto1_Poring + " Lunatic=" + Tuto1_Lunatic + " Checkpoint=" + Tuto1_Checkpoint;

	mes "[Olgi]";

	if (Tuto1_State == 0) {
		mes "Speak with Olga first.";
		close;
	}

	if (Tuto1_State == 10) {
		mes "I can validate Olga's first task.";
		mes "Let me check your progress.";
		next;

		if (Tuto1_Poring < 2 || Tuto1_Lunatic < 2 || Tuto1_Checkpoint < 1) {
			mes "[Olgi]";
			mes "You are not done yet.";
			mes "Poring: " + Tuto1_Poring + "/2";
			mes "Lunatic: " + Tuto1_Lunatic + "/2";
			mes "Location: " + Tuto1_Checkpoint + "/1";
			close;
		}

		Tuto1_State = 20;
		completequest 65000;
		setquest 65001;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: first task validated. Go speak with Olgu.";

		mes "[Olgi]";
		mes "Good. Everything is complete.";
		mes "Now go speak with Olgu.";
		close;
	}

	if (Tuto1_State == 20) {
		mes "Go speak with Olgu.";
		close;
	}

	if (Tuto1_State == 30) {
		mes "Olgu sent you back to Olga.";
		mes "Bring Olga 5 Jellopy.";
		close;
	}

	if (Tuto1_State >= 40) {
		mes "You already completed Main Quest Tuto 1.";
		close;
	}

	mes "Your tutorial state was invalid.";
	close;
}

//============================================================
// Checkpoint - invisible touch area
//============================================================

new_1-3,87,32,0	script	Tuto1_Checkpoint	HIDDEN_NPC,2,2,{
OnTouch:
	if (Tuto1_State == 10 && Tuto1_Checkpoint < 1) {
		Tuto1_Checkpoint = 1;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Location visited 1/1.";
	}
	end;
}

//============================================================
// NPC 3 - Olgu
//============================================================

new_1-3,116,45,4	script	Olgu	4_M_02,{
	if (#DebugConsole)
		dispbottom "[DEBUG][Olgu] Click. State=" + Tuto1_State;

	mes "[Olgu]";

	if (Tuto1_State < 20) {
		mes "You are not ready for me yet.";
		mes "Go follow Olga's instructions first.";
		close;
	}

	if (Tuto1_State == 20) {
		mes "Olgi sent you?";
		mes "Good.";
		mes "Return to Olga with 5 Jellopy in your inventory.";
		next;

		Tuto1_State = 30;
		Tuto1_Jellopy_Last = -1;
		Tuto1_Jellopy_Timer = 1;

		completequest 65001;
		setquest 65002;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: collect 5 Jellopy, then return to Olga.";
		addtimer 2000, "Tuto1_Tracker::OnJellopyCheck";

		mes "[Olgu]";
		mes "Olga will handle your final reward.";
		close;
	}

	if (Tuto1_State == 30) {
		.@jellopy = countitem(909);
		if (.@jellopy > 5)
			.@jellopy = 5;

		mes "Return to Olga with 5 Jellopy.";
		mes "Current progress: " + .@jellopy + "/5";
		close;
	}

	if (Tuto1_State >= 40) {
		mes "You already completed this tutorial.";
		close;
	}

	mes "Your tutorial state was invalid.";
	close;
}

//============================================================
// Progress tracker
//============================================================

-	script	Tuto1_Tracker	FAKE_NPC,{
OnNPCKillEvent:
	if (Tuto1_State != 10)
		end;

	if (killedrid == 1002 && Tuto1_Poring < 2) {
		Tuto1_Poring++;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Poring " + Tuto1_Poring + "/2.";
	}

	if (killedrid == 1063 && Tuto1_Lunatic < 2) {
		Tuto1_Lunatic++;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Lunatic " + Tuto1_Lunatic + "/2.";
	}

	if (Tuto1_Poring >= 2 && Tuto1_Lunatic >= 2 && Tuto1_Checkpoint >= 1)
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: objectives complete. Go validate with Olgi.";

	end;

OnJellopyCheck:
	if (Tuto1_State != 30) {
		Tuto1_Jellopy_Timer = 0;
		end;
	}

	.@jellopy = countitem(909);
	if (.@jellopy > 5)
		.@jellopy = 5;

	if (Tuto1_Jellopy_Last != .@jellopy) {
		Tuto1_Jellopy_Last = .@jellopy;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Jellopy " + .@jellopy + "/5.";
	}

	if (.@jellopy >= 5) {
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Jellopy complete. Return to Olga.";
		Tuto1_Jellopy_Timer = 0;
		end;
	}

	addtimer 2000, "Tuto1_Tracker::OnJellopyCheck";
	end;

OnPCLoginEvent:
	if (Tuto1_State == 30 && !Tuto1_Jellopy_Timer) {
		Tuto1_Jellopy_Timer = 1;
		addtimer 2000, "Tuto1_Tracker::OnJellopyCheck";
	}
	end;
}
