// ==========================
// NPC 1 - OLGA
// new_1-3,99,30
// ==========================

new_1-3,99,30,4	script	Olga#Tuto1	4_F_KAFRA1,{

	mes "[Olga]";

	if (Tuto1_State >= 4) {
		mes "You already completed Main Quest Tuto 1.";
		close;
	}

	if (Tuto1_State == 0) {
		mes "Welcome.";
		mes "This is Main Quest Tuto 1.";
		mes "Your first mission is simple:";
		mes "^FF00002 Poring^000000";
		mes "^FF00002 Lunatic^000000";
		mes "Then visit the marked location:";
		mes "^00FF00new_1-3,87,32^000000";
		next;

		if (select("Accept:Decline") == 2) {
			mes "[Olga]";
			mes "Come back when you are ready.";
			close;
		}

		Tuto1_State = 1;
		Tuto1_Poring = 0;
		Tuto1_Lunatic = 0;
		Tuto1_Checkpoint = 0;
		Tuto1_Jellopy_Last = -1;
		Tuto1_Jellopy_Timer = 0;

		mes "[Olga]";
		mes "Quest started:";
		mes "^00FF00Main Quest Tuto 1^000000";
		mes "Once everything is done, go validate it with Olgi.";
		close2;

		setquest 65000;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Poring 0/2, Lunatic 0/2, Location 0/1.";
		end;
	}

	if (Tuto1_State == 1) {
		mes "Current objective:";
		mes "Kill 2 Poring: " + Tuto1_Poring + "/2";
		mes "Kill 2 Lunatic: " + Tuto1_Lunatic + "/2";
		mes "Visit location new_1-3,87,32: " + Tuto1_Checkpoint + "/1";
		mes "Then go validate with Olgi.";
		close;
	}

	if (Tuto1_State == 2) {
		mes "You already validated your first task.";
		mes "Go speak with Olgu.";
		close;
	}

	if (Tuto1_State == 3) {
		if (!Tuto1_Jellopy_Timer) {
			Tuto1_Jellopy_Timer = 1;
			addtimer 2000, "Tuto1_KillTracker::OnJellopyCheck";
		}

		.@jellopy = countitem(909);
		if (.@jellopy > 5)
			.@jellopy = 5;

		mes "Olgu sent you back?";
		mes "Bring me ^FF00005 Jellopy^000000.";
		mes "Current progress: " + .@jellopy + "/5";
		next;

		if (countitem(909) < 5) {
			mes "[Olga]";
			mes "You do not have 5 Jellopy yet.";
			close;
		}

		delitem 909,5;
		Zeny += 100000;
		getitem 12622,1;

		Tuto1_State = 4;
		Tuto1_Jellopy_Last = -1;
		Tuto1_Jellopy_Timer = 0;

		completequest 65002;

		specialeffect2 EF_LEVEL99_3;
		skilleffect AL_BLESSING,0;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: completed.";

		mes "[Olga]";
		mes "Quest complete.";
		mes "You received:";
		mes "^00FF00100,000 Zeny^000000";
		mes "^00FF001 Halter Lead^000000";
		close;
	}

	mes "Your quest state was invalid.";
	mes "I reset it. Please speak with me again.";
	Tuto1_State = 0;
	close;
}


// ==========================
// NPC 2 - OLGI
// new_1-3,101,30
// ==========================

new_1-3,101,30,4	script	Olgi#Tuto1	4_F_KAFRA1,{

	mes("[Olgi]");

	if (Tuto1_State == 0) {
		mes("You should speak with Olga first.");
		close();
	}

	if (Tuto1_State == 1) {
		mes("Did you kill the monsters and visit the indicated place?");
		next();

		if (select("Yes:No") == 2) {
			mes("[Olgi]");
			mes("Finish Olga's task, then come back.");
			close();
		}

		if (Tuto1_Poring < 2 || Tuto1_Lunatic < 2 || Tuto1_Checkpoint < 1) {
			mes("[Olgi]");
			mes("No, you did not complete everything.");
			mes("Poring: " + Tuto1_Poring + "/2");
			mes("Lunatic: " + Tuto1_Lunatic + "/2");
			mes("Location: " + Tuto1_Checkpoint + "/1");
			close();
		}

		Tuto1_State = 2;

		completequest 65000;
		setquest 65001;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: first task validated. Go speak with Olgu.";

		mes("[Olgi]");
		mes("Good. Everything is done.");
		mes("Now go speak with Olgu.");
		close();
	}

	if (Tuto1_State == 2) {
		mes("Go speak with Olgu.");
		close();
	}

	if (Tuto1_State == 3) {
		if (!Tuto1_Jellopy_Timer) {
			Tuto1_Jellopy_Timer = 1;
			addtimer 2000, "Tuto1_KillTracker::OnJellopyCheck";
		}

		mes("Olgu sent you back to Olga.");
		mes("Bring Olga 5 Jellopy.");
		close();
	}

	if (Tuto1_State >= 4) {
		mes("You already completed Main Quest Tuto 1.");
		close();
	}
}


// ==========================
// CHECKPOINT
// new_1-3,87,32
// Invisible position trigger
// ==========================

new_1-3,87,32,0	script	Tuto1_Checkpoint#01	HIDDEN_NPC,2,2,{

OnTouch:
	if (Tuto1_State == 1 && Tuto1_Checkpoint < 1) {
		Tuto1_Checkpoint = 1;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Location visited 1/1.";
	}
	end;
}


// ==========================
// NPC 3 - OLGU
// new_1-3,116,45
// ==========================

new_1-3,116,45,4	script	Olgu#Tuto1	4_F_KAFRA1,{

	mes("[Olgu]");

	if (Tuto1_State < 2) {
		mes("You are not ready for me yet.");
		close();
	}

	if (Tuto1_State == 2) {
		mes("Olgi sent you?");
		mes("Good.");
		mes("Return to Olga with ^FF00005 Jellopy^000000 in your inventory.");
		next();

		Tuto1_State = 3;
		Tuto1_Jellopy_Last = -1;
		Tuto1_Jellopy_Timer = 1;

		completequest 65001;
		setquest 65002;

		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Collect Jellopy 0/5, then return to Olga.";
		addtimer 2000, "Tuto1_KillTracker::OnJellopyCheck";

		mes("[Olgu]");
		mes("I will let Olga handle the final reward.");
		close();
	}

	if (Tuto1_State == 3) {
		if (!Tuto1_Jellopy_Timer) {
			Tuto1_Jellopy_Timer = 1;
			addtimer 2000, "Tuto1_KillTracker::OnJellopyCheck";
		}

		.@jellopy = countitem(909);
		if (.@jellopy > 5)
			.@jellopy = 5;

		mes("Go back to Olga with 5 Jellopy.");
		mes("Current progress: " + .@jellopy + "/5");
		close();
	}

	if (Tuto1_State >= 4) {
		mes("You already completed this tutorial.");
		close();
	}
}


// ==========================
// PROGRESS TRACKER
// ==========================

-	script	Tuto1_KillTracker	FAKE_NPC,{

OnNPCKillEvent:
	if (Tuto1_State != 1)
		end;

	if (killedrid == 1002 && Tuto1_Poring < 2) {
		Tuto1_Poring++;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Poring " + Tuto1_Poring + "/2.";
	}

	if (killedrid == 1063 && Tuto1_Lunatic < 2) {
		Tuto1_Lunatic++;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Lunatic " + Tuto1_Lunatic + "/2.";
	}

	if (Tuto1_Poring >= 2 && Tuto1_Lunatic >= 2 && Tuto1_Checkpoint >= 1)
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: objectives complete. Go validate with Olgi.";

	end;

OnJellopyCheck:
	if (Tuto1_State != 3) {
		Tuto1_Jellopy_Timer = 0;
		end;
	}

	.@jellopy = countitem(909);
	if (.@jellopy > 5)
		.@jellopy = 5;

	if (Tuto1_Jellopy_Last != .@jellopy) {
		Tuto1_Jellopy_Last = .@jellopy;
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Jellopy " + .@jellopy + "/5.";
	}

	if (.@jellopy >= 5) {
		message strcharinfo(PC_NAME), "Main Quest Tuto 1: Jellopy complete. Return to Olga.";
		Tuto1_Jellopy_Timer = 0;
		end;
	}

	addtimer 2000, "Tuto1_KillTracker::OnJellopyCheck";
	end;

OnPCLoginEvent:
	if (Tuto1_State == 3 && !Tuto1_Jellopy_Timer) {
		Tuto1_Jellopy_Timer = 1;
		addtimer 2000, "Tuto1_KillTracker::OnJellopyCheck";
	}
	end;
}
