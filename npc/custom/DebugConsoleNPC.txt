//===== Hercules Script ======================================
// Generic in-game debug console.
//
// Limitation:
// This NPC cannot read the real map-server console and cannot catch a script
// before it starts executing. A future script must call F_DebugConsole to send
// debug lines to the player chat.
//
// Usage in any NPC script:
//   callfunc "F_DebugConsole","NPC_NAME","message";
//   callfunc "F_DebugConsole","Olga","State=" + Tuto1_State;
//
// Optional value helper:
//   callfunc "F_DebugValue","Olga","Tuto1_State",Tuto1_State;
//============================================================

prontera,164,162,4	script	Debug System	4_F_KAFRA1,{
	mes "[CONSOLE DEBUG]";
	mes "Generic debug console.";
	mes "Status: " + (#DebugConsole ? "enabled" : "disabled");
	next;

	switch(select((#DebugConsole ? "Disable" : "Enable"), "Show status", "How to use", "Reset custom tests", "Cancel")) {
	case 1:
		if (#DebugConsole) {
			#DebugConsole = 0;
			mes "[CONSOLE DEBUG]";
			mes "Debug console disabled.";
			dispbottom "[DEBUG] Console disabled.";
			close;
		}

		#DebugConsole = 1;
		mes "[CONSOLE DEBUG]";
		mes "Debug console enabled.";
		mes "Scripts that call F_DebugConsole will now write in your chat.";
		dispbottom "[DEBUG] Console enabled for " + strcharinfo(PC_NAME) + ".";
		callfunc "F_DebugStatus";
		close2;
		addtimer 10000, "DebugConsoleController::OnDebugPulse";
		end;

	case 2:
		callfunc "F_DebugStatus";
		mes "[CONSOLE DEBUG]";
		mes "Status printed in your chat.";
		close;

	case 3:
		mes "[CONSOLE DEBUG]";
		mes "Add this line inside any NPC you want to inspect:";
		mes "callfunc \"F_DebugConsole\",\"NPC\",\"step reached\";";
		next;
		mes "[CONSOLE DEBUG]";
		mes "For a variable:";
		mes "callfunc \"F_DebugValue\",\"NPC\",\"Var\",Var;";
		mes "Only players who enabled this console will see the lines.";
		close;

	case 4:
		callfunc "F_DebugResetMenu";
		close;

	case 5:
		close;
	}
}

-	script	DebugConsoleController	FAKE_NPC,{
OnDebugPulse:
	if (!#DebugConsole)
		end;

	callfunc "F_DebugStatus";
	addtimer 10000, "DebugConsoleController::OnDebugPulse";
	end;
}

function	script	F_DebugConsole	{
	if (!#DebugConsole)
		return;

	dispbottom "[DEBUG][" + getarg(0) + "] " + getarg(1);
	return;
}

function	script	F_DebugValue	{
	if (!#DebugConsole)
		return;

	dispbottom "[DEBUG][" + getarg(0) + "] " + getarg(1) + "=" + getarg(2);
	return;
}

function	script	F_DebugStatus	{
	if (!#DebugConsole)
		return;

	getmapxy(.@map$,.@x,.@y,UNITTYPE_PC);
	dispbottom "[DEBUG] Player=" + strcharinfo(PC_NAME) + " Map=" + .@map$ + " X=" + .@x + " Y=" + .@y;
	return;
}

function	script	F_DebugResetMenu	{
	while (true) {
		mes "[CONSOLE DEBUG]";
		mes "Reset des variables de test custom.";
		mes "Equivalent a des @set a 0, lance depuis ce NPC.";
		next;

		.@choice = select("All custom test vars:Faction + territoires:Eterites:Contracts:Tutos:Ascension:Mission contracts:Other custom:Show @set list:Back");

		if (.@choice == 1) {
			callfunc "F_DebugResetAllCustom";
			mes "[CONSOLE DEBUG]";
			mes "All custom test vars reset.";
			next;
		} else if (.@choice == 2) {
			callfunc "F_DebugResetFactionTerritories";
			mes "[CONSOLE DEBUG]";
			mes "Faction, mascottes et territoires reset.";
			next;
		} else if (.@choice == 3) {
			callfunc "F_DebugResetEterites";
			mes "[CONSOLE DEBUG]";
			mes "Eterites reset.";
			next;
		} else if (.@choice == 4) {
			callfunc "F_DebugResetContracts";
			mes "[CONSOLE DEBUG]";
			mes "Contracts reset.";
			next;
		} else if (.@choice == 5) {
			callfunc "F_DebugResetTutos";
			mes "[CONSOLE DEBUG]";
			mes "Tutos reset.";
			next;
		} else if (.@choice == 6) {
			callfunc "F_DebugResetAscension";
			mes "[CONSOLE DEBUG]";
			mes "Ascension reset.";
			next;
		} else if (.@choice == 7) {
			callfunc "F_DebugResetMissionContracts";
			mes "[CONSOLE DEBUG]";
			mes "Mission contracts reset.";
			next;
		} else if (.@choice == 8) {
			callfunc "F_DebugResetOtherCustom";
			mes "[CONSOLE DEBUG]";
			mes "Other custom vars reset.";
			next;
		} else if (.@choice == 9) {
			callfunc "F_DebugShowSetList";
			next;
		} else {
			return;
		}
	}
}

function	script	F_DebugResetAllCustom	{
	callfunc "F_DebugResetFactionTerritories";
	callfunc "F_DebugResetEterites";
	callfunc "F_DebugResetContracts";
	callfunc "F_DebugResetTutos";
	callfunc "F_DebugResetAscension";
	callfunc "F_DebugResetMissionContracts";
	callfunc "F_DebugResetOtherCustom";
	return;
}

	$Territory_Prt06_Kills = 0;
	$Territory_Prt06_Pending = 0;

}

function	script	F_DebugResetMissionContracts	{
	for (.@i = 0; .@i < 20; .@i++) {
		$MissionContract_Active[.@i] = 0;
		$MissionContract_Owner$[.@i] = "";
		$MissionContract_MobID[.@i] = 0;
		$MissionContract_MobName$[.@i] = "";
		$MissionContract_Message$[.@i] = "";
		$MissionContract_TimeText$[.@i] = "";
		$MissionContract_Reward$[.@i] = "";
		$MissionContract_Duration$[.@i] = "";
		$MissionContract_Expire[.@i] = 0;
	}
	return;
}
