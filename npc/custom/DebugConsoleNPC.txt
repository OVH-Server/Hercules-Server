//===== Hercules Script ======================================
// Generic in-game debug console.
//
// Limitation:
// This NPC cannot read the real map-server console and cannot catch a script
// before it starts executing. A future script must call F_DebugConsole to send
// debug lines to the player chat.
//
// Usage in any NPC script:
//   callfunc "F_DebugConsole","NPC_NAME","message";
//   callfunc "F_DebugConsole","Olga","State=" + Tuto1_State;
//
// Optional value helper:
//   callfunc "F_DebugValue","Olga","Tuto1_State",Tuto1_State;
//============================================================

prontera,164,162,4	script	CONSOLE DEBUG	4_F_KAFRA1,{
	mes "[CONSOLE DEBUG]";
	mes "Generic debug console.";
	mes "Status: " + (#DebugConsole ? "enabled" : "disabled");
	next;

	switch(select((#DebugConsole ? "Disable" : "Enable"), "Show status", "How to use", "Cancel")) {
	case 1:
		if (#DebugConsole) {
			#DebugConsole = 0;
			mes "[CONSOLE DEBUG]";
			mes "Debug console disabled.";
			dispbottom "[DEBUG] Console disabled.";
			close;
		}

		#DebugConsole = 1;
		mes "[CONSOLE DEBUG]";
		mes "Debug console enabled.";
		mes "Scripts that call F_DebugConsole will now write in your chat.";
		dispbottom "[DEBUG] Console enabled for " + strcharinfo(PC_NAME) + ".";
		callfunc "F_DebugStatus";
		close2;
		addtimer 10000, "DebugConsoleController::OnDebugPulse";
		end;

	case 2:
		callfunc "F_DebugStatus";
		mes "[CONSOLE DEBUG]";
		mes "Status printed in your chat.";
		close;

	case 3:
		mes "[CONSOLE DEBUG]";
		mes "Add this line inside any NPC you want to inspect:";
		mes "callfunc \"F_DebugConsole\",\"NPC\",\"step reached\";";
		next;
		mes "[CONSOLE DEBUG]";
		mes "For a variable:";
		mes "callfunc \"F_DebugValue\",\"NPC\",\"Var\",Var;";
		mes "Only players who enabled this console will see the lines.";
		close;

	case 4:
		close;
	}
}

-	script	DebugConsoleController	FAKE_NPC,{
OnDebugPulse:
	if (!#DebugConsole)
		end;

	callfunc "F_DebugStatus";
	addtimer 10000, "DebugConsoleController::OnDebugPulse";
	end;
}

function	script	F_DebugConsole	{
	if (!#DebugConsole)
		return;

	dispbottom "[DEBUG][" + getarg(0) + "] " + getarg(1);
	return;
}

function	script	F_DebugValue	{
	if (!#DebugConsole)
		return;

	dispbottom "[DEBUG][" + getarg(0) + "] " + getarg(1) + "=" + getarg(2);
	return;
}

function	script	F_DebugStatus	{
	if (!#DebugConsole)
		return;

	getmapxy(.@map$,.@x,.@y,UNITTYPE_PC);
	dispbottom "[DEBUG] Player=" + strcharinfo(PC_NAME) + " Map=" + .@map$ + " X=" + .@x + " Y=" + .@y;
	return;
}
