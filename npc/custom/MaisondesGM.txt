gonryun,211,118,3	script	Sacred Guard#1	4_M_CHNSOLDIER,{

	mes "[Sacred Guard]";
	mes "No one passes without the password.";
	next;

	input .@pass$;

	if (.@pass$ != "123") {
		mes "[Sacred Guard]";
		mes "Incorrect password.";
		close;
	}

	mes "[Sacred Guard]";
	mes "Very well. You have 10 seconds to enter.";
	close2;

	set #GonryunAccessTime, gettimetick(2) + 10;
	end;
}
